     1→/**
     2→ * Think Leo — Sound Engine v3 (Web Audio SFX + real background music)
     3→ * ────────────────────────────────────────────────────────
     4→ * SFX: a tiny synth studio. Design notes: docs/ALGORITHM.md §8.
     5→ * MUSIC: three licensed background tracks (CC-BY by Kevin MacLeod,
     6→ * incompetech.com — properly composed, properly mixed, kid-approved):
     7→ *
     8→ *   /audio/home.mp3   "Carefree"                  — hub & map
     9→ *   /audio/play.mp3   "Monkeys Spinning Monkeys"  — gameplay
    10→ *   /audio/rush.mp3   "Sneaky Snitch"             — LEO RUSH finale
    11→ *
    12→ * · One synthesized impulse-response gives the SFX the same "room".
    13→ * · Instruments are a fixed palette (marimba, pluck, chime, horn, kit,
    14→ *   riser, roar) — every cue is a recipe over that palette.
    15→ * · The music player is autoplay-policy-proof: the requested mood is
    16→ *   remembered and playback retries on the first gesture / on resume.
    17→ * · Music ducks under big cues (fanfare, level-up) like a real DJ.
    18→ *
    19→ * Signal chain:
    20→ *
    21→ *   voice ─► voiceGain ─► pan ─┬─► sfxBus ─────────────┬─► master ─► comp ─► out
    22→ *                              └─► reverbSend ─► conv ──┘        ▲
    23→ *   music (<audio> loops · crossfade · duck) ────────────┘
    24→ *
    25→ * · Master bus dynamics: a gentle compressor glues SFX layers so loud
    26→ *   hits never clip (kids crank volume; we protect their ears).
    27→ */
    28→
    29→type Osc = OscillatorType;
    30→
    31→type MusicMode = "home" | "play" | "rush";
    32→
    33→const MUSIC_SRC: Record<MusicMode, string> = {
    34→  home: "/audio/home.mp3",
    35→  play: "/audio/play.mp3",
    36→  rush: "/audio/rush.mp3",
    37→};
    38→
    39→/** Background level per mood — music sits UNDER the SFX, never on top. */
    40→const MUSIC_VOL: Record<MusicMode, number> = { home: 0.5, play: 0.42, rush: 0.52 };
    41→
    42→const NOTE = (semi: number, base = 440) => base * Math.pow(2, semi / 12);
    43→
    44→interface ToneOpts {
    45→  at?: number;
    46→  dur?: number;
    47→  type?: Osc;
    48→  vol?: number;
    49→  glide?: number;
    50→  attack?: number;
    51→  bus?: GainNode | null;
    52→  pan?: number;
    53→  wet?: number; // 0..1 reverb send
    54→  filter?: number; // lowpass cutoff (Hz)
    55→  detune?: number; // cents
    56→}
    57→
    58→class LeoSound {
    59→  private ctx: AudioContext | null = null;
    60→  private master: GainNode | null = null;
    61→  private comp: DynamicsCompressorNode | null = null;
    62→  private sfxBus: GainNode | null = null;
    63→  private reverb: ConvolverNode | null = null;
    64→  private reverbSend: GainNode | null = null;
    65→  private noiseBuf: AudioBuffer | null = null;
    66→  private duckTimer: number | null = null;
    67→
    68→  /* music: three real tracks, at most one playing */
    69→  private tracks: Partial<Record<MusicMode, HTMLAudioElement>> = {};
    70→  private currentTrack: MusicMode | null = null;
    71→  private desiredTrack: MusicMode | null = null;
    72→  private fadeRafs = new WeakMap<HTMLAudioElement, number>();
    73→
    74→  sfxOn = true;
    75→  musicOn = true;
    76→
    77→  /** Call on the first user gesture. Safe to call repeatedly. */
    78→  unlock() {
    79→    try {
    80→      if (!this.ctx) this.buildGraph();
    81→      if (this.ctx && this.ctx.state === "suspended") void this.ctx.resume();
    82→      if (typeof document !== "undefined") {
    83→        document.removeEventListener("visibilitychange", this.onVis);
    84→        document.addEventListener("visibilitychange", this.onVis);
    85→      }
    86→      // autoplay policy: the first gesture is our chance to start the band
    87→      this.tryStartMusic();
    88→    } catch {
    89→      /* audio is a bonus, never a blocker */
    90→    }
    91→  }
    92→
    93→  private onVis = () => {
    94→    if (document.visibilityState === "hidden") {
    95→      const cur = this.currentTrack ? this.tracks[this.currentTrack] : null;
    96→      cur?.pause(); // battery manners
    97→    } else {
    98→      if (this.ctx && this.ctx.state === "suspended") void this.ctx.resume();
    99→      this.tryStartMusic();
   100→    }
   101→  };
   102→
   103→  private buildGraph() {
   104→    const AC =
   105→      window.AudioContext ||
   106→      (window as unknown as { webkitAudioContext: typeof AudioContext }).webkitAudioContext;
   107→    if (!AC) return;
   108→    const ctx = new AC();
   109→    this.ctx = ctx;
   110→
   111→    this.comp = ctx.createDynamicsCompressor();
   112→    this.comp.threshold.value = -18;
   113→    this.comp.knee.value = 14;
   114→    this.comp.ratio.value = 4;
   115→    this.comp.attack.value = 0.004;
   116→    this.comp.release.value = 0.18;
   117→
   118→    this.master = ctx.createGain();
   119→    this.master.gain.value = 0.9;
   120→    this.master.connect(this.comp).connect(ctx.destination);
   121→
   122→    // one shared "room" — synthesized stereo impulse, exponential decay
   123→    this.reverb = ctx.createConvolver();
   124→    const len = Math.floor(ctx.sampleRate * 1.3);
   125→    const ir = ctx.createBuffer(2, len, ctx.sampleRate);
   126→    for (let ch = 0; ch < 2; ch++) {
   127→      const d = ir.getChannelData(ch);
   128→      for (let i = 0; i < len; i++) {
   129→        d[i] = (Math.random() * 2 - 1) * Math.pow(1 - i / len, 2.6);
   130→      }
   131→    }
   132→    this.reverb.buffer = ir;
   133→    this.reverbSend = ctx.createGain();
   134→    this.reverbSend.gain.value = 1;
   135→    const reverbOut = ctx.createGain();
   136→    reverbOut.gain.value = 0.32;
   137→    this.reverbSend.connect(this.reverb).connect(reverbOut).connect(this.master);
   138→
   139→    this.sfxBus = ctx.createGain();
   140→    this.sfxBus.gain.value = 1;
   141→    this.sfxBus.connect(this.master);
   142→
   143→    // reusable white noise (saves building a buffer per hat/snare)
   144→    const nlen = Math.floor(ctx.sampleRate * 1.0);
   145→    this.noiseBuf = ctx.createBuffer(1, nlen, ctx.sampleRate);
   146→    const nd = this.noiseBuf.getChannelData(0);
   147→    for (let i = 0; i < nlen; i++) nd[i] = Math.random() * 2 - 1;
   148→  }
   149→
   150→  private get t0(): number {
   151→    return this.ctx ? this.ctx.currentTime : 0;
   152→  }
   153→
   154→  /* ── instrument primitives ──────────────────────────────────── */
   155→
   156→  private tone(freq: number, o: ToneOpts = {}) {
   157→    if (!this.ctx || !this.master) return;
   158→    const {
   159→      at = this.t0,
   160→      dur = 0.15,
   161→      type = "sine",
   162→      vol = 0.2,
   163→      glide,
   164→      attack = 0.006,
   165→      bus,
   166→      pan = 0,
   167→      wet = 0,
   168→      filter,
   169→      detune = 0,
   170→    } = o;
   171→    const out = bus ?? this.sfxBus;
   172→    if (!out) return;
   173→
   174→    const osc = this.ctx.createOscillator();
   175→    const gain = this.ctx.createGain();
   176→    osc.type = type;
   177→    osc.frequency.setValueAtTime(freq, at);
   178→    if (detune) osc.detune.setValueAtTime(detune, at);
   179→    if (glide) osc.frequency.exponentialRampToValueAtTime(Math.max(30, glide), at + dur);
   180→
   181→    gain.gain.setValueAtTime(0.0001, at);
   182→    gain.gain.exponentialRampToValueAtTime(vol, at + attack);
   183→    gain.gain.exponentialRampToValueAtTime(0.0001, at + dur);
   184→
   185→    let node: AudioNode = osc;
   186→    if (filter) {
   187→      const f = this.ctx.createBiquadFilter();
   188→      f.type = "lowpass";
   189→      f.frequency.setValueAtTime(filter, at);
   190→      f.Q.value = 0.8;
   191→      node.connect(f);
   192→      node = f;
   193→    }
   194→    node.connect(gain);
   195→    this.route(gain, out, pan, wet);
   196→    osc.start(at);
   197→    osc.stop(at + dur + 0.08);
   198→  }
   199→
   200→  private noise(
   201→    o: {
   202→      at?: number;
   203→      dur?: number;
   204→      vol?: number;
   205→      f0?: number;
   206→      f1?: number;
   207→      q?: number;
   208→      type?: BiquadFilterType;
   209→      pan?: number;
   210→      wet?: number;
   211→      bus?: GainNode | null;
   212→    } = {}
   213→  ) {
   214→    if (!this.ctx || !this.noiseBuf) return;
   215→    const {
   216→      at = this.t0,
   217→      dur = 0.25,
   218→      vol = 0.12,
   219→      f0 = 800,
   220→      f1,
   221→      q = 1,
   222→      type = "bandpass",
   223→      pan = 0,
   224→      wet = 0,
   225→      bus,
   226→    } = o;
   227→    const out = bus ?? this.sfxBus;
   228→    if (!out) return;
   229→
   230→    const src = this.ctx.createBufferSource();
   231→    src.buffer = this.noiseBuf;
   232→    src.loop = true;
   233→    const filt = this.ctx.createBiquadFilter();
   234→    filt.type = type;
   235→    filt.frequency.setValueAtTime(f0, at);
   236→    if (f1) filt.frequency.exponentialRampToValueAtTime(Math.max(40, f1), at + dur);
   237→    filt.Q.value = q;
   238→    const gain = this.ctx.createGain();
   239→    gain.gain.setValueAtTime(vol, at);
   240→    gain.gain.exponentialRampToValueAtTime(0.0001, at + dur);
   241→    src.connect(filt).connect(gain);
   242→    this.route(gain, out, pan, wet);
   243→    src.start(at);
   244→    src.stop(at + dur + 0.05);
   245→  }
   246→
   247→  private route(input: AudioNode, bus: GainNode, pan: number, wet: number) {
   248→    if (!this.ctx) return;
   249→    let node = input;
   250→    if (pan !== 0 && this.ctx.createStereoPanner) {
   251→      const p = this.ctx.createStereoPanner();
   252→      p.pan.setValueAtTime(pan, this.t0);
   253→      node.connect(p);
   254→      node = p;
   255→    }
   256→    node.connect(bus);
   257→    if (wet > 0 && this.reverbSend) {
   258→      const w = this.ctx.createGain();
   259→      w.gain.value = wet;
   260→      node.connect(w).connect(this.reverbSend);
   261→    }
   262→  }
   263→
   264→  /* ── instruments ────────────────────────────────────────────── */
   265→
   266→  /** warm wooden hit — the game's "correct voice" */
   267→  private marimba(freq: number, at: number, vol = 0.18, pan = 0) {
   268→    this.tone(freq, { at, dur: 0.34, type: "sine", vol, pan, wet: 0.24 });
   269→    this.tone(freq * 4, { at, dur: 0.09, type: "sine", vol: vol * 0.34, pan });
   270→    this.tone(freq * 2.01, { at, dur: 0.16, type: "triangle", vol: vol * 0.22, pan });
   271→  }
   272→
   273→  /** guitar-ish pluck for arps and melodies */
   274→  private pluck(freq: number, at: number, vol = 0.13, pan = 0) {
   275→    this.tone(freq, { at, dur: 0.22, type: "triangle", vol, pan, filter: Math.min(9000, freq * 5), wet: 0.18 });
   276→    this.tone(freq * 2, { at, dur: 0.1, type: "sine", vol: vol * 0.4, pan });
   277→  }
   278→
   279→  /** bell — stars, achievements, magic */
   280→  private chime(freq: number, at: number, vol = 0.12, pan = 0) {
   281→    this.tone(freq, { at, dur: 0.6, type: "sine", vol, pan, wet: 0.5 });
   282→    this.tone(freq * 2.76, { at, dur: 0.28, type: "sine", vol: vol * 0.3, pan, wet: 0.4 });
   283→    this.tone(freq * 5.4, { at: at + 0.01, dur: 0.16, type: "sine", vol: vol * 0.14, pan });
   284→  }
   285→
   286→  /** soft rounded horn — fails, sad moments (never harsh) */
   287→  private horn(freq: number, at: number, dur = 0.3, vol = 0.08) {
   288→    this.tone(freq, { at, dur, type: "sawtooth", vol, filter: 900, attack: 0.03, wet: 0.2 });
   289→    this.tone(freq * 1.005, { at, dur, type: "triangle", vol: vol * 0.8, attack: 0.03 });
   290→  }
   291→
   292→  private kick(at: number, vol = 0.3) {
   293→    this.tone(150, { at, dur: 0.22, type: "sine", vol, glide: 44, attack: 0.002 });
   294→    this.noise({ at, dur: 0.03, vol: vol * 0.3, f0: 1200, type: "lowpass" });
   295→  }
   296→
   297→  private snare(at: number, vol = 0.16) {
   298→    this.noise({ at, dur: 0.14, vol, f0: 1900, q: 0.8, wet: 0.2 });
   299→    this.tone(196, { at, dur: 0.1, type: "triangle", vol: vol * 0.55 });
   300→  }
   301→
   302→  private hat(at: number, open = false, vol = 0.05) {
   303→    this.noise({ at, dur: open ? 0.14 : 0.035, vol, f0: 8200, type: "highpass", q: 0.7 });
   304→  }
   305→
   306→  /** rising energy sweep — intros, RUSH ignition */
   307→  private riser(at: number, dur = 0.7, vol = 0.1) {
   308→    this.noise({ at, dur, vol, f0: 350, f1: 3800, q: 1.6, wet: 0.3 });
   309→    this.tone(220, { at, dur, type: "triangle", vol: vol * 0.7, glide: 880, attack: dur * 0.7 });
   310→  }
   311→
   312→  private thud(at: number, vol = 0.2) {
   313→    this.tone(96, { at, dur: 0.24, type: "sine", vol, glide: 52, attack: 0.004 });
   314→  }
   315→
   316→  /** the one big "character" voice: a friendly lion-ish growl */
   317→  roar(at = this.t0) {
   318→    if (!this.guard()) return;
   319→    const vol = 0.16;
   320→    for (const [f, det] of [[104, 0], [131, 8], [78, -6]] as const) {
   321→      this.tone(f, {
   322→        at: at + 0.05,
   323→        dur: 0.85,
   324→        type: "sawtooth",
   325→        vol: vol * 0.5,
   326→        glide: f * 0.62,
   327→        attack: 0.12,
   328→        filter: 420,
   329→        wet: 0.35,
   330→        detune: det,
   331→      });
   332→    }
   333→    this.noise({ at: at + 0.05, dur: 0.8, vol: vol * 0.4, f0: 300, f1: 160, q: 0.6, wet: 0.3 });
   334→    this.noise({ at: at + 0.02, dur: 0.18, vol: vol * 0.5, f0: 220, f1: 900, q: 0.8 });
   335→  }
   336→
   337→  private guard(): boolean {
   338→    return this.sfxOn && !!this.ctx;
   339→  }
   340→
   341→  /**
   342→   * Brief music dip so a big cue always lands on top — a real DJ move,
   343→   * now with real music: fade down, let the fanfare land, ride back up.
   344→   */
   345→  private duck() {
   346→    if (!this.musicOn || !this.currentTrack) return;
   347→    const a = this.tracks[this.currentTrack];
   348→    if (!a || a.paused) return;
   349→    const base = MUSIC_VOL[this.currentTrack];
   350→    this.fadeTo(a, base * 0.22, 90);
   351→    if (this.duckTimer !== null) window.clearTimeout(this.duckTimer);
   352→    this.duckTimer = window.setTimeout(() => {
   353→      const cur = this.currentTrack ? this.tracks[this.currentTrack] : null;
   354→      if (cur && this.currentTrack && this.musicOn && !cur.paused) {
   355→        this.fadeTo(cur, MUSIC_VOL[this.currentTrack], 700);
   356→      }
   357→    }, 950);
   358→  }

   360→  /* ── SFX library ────────────────────────────────────────────── */
   361→
   362→  tap() {
   363→    if (!this.guard()) return;
   364→    this.pluck(660, this.t0, 0.1);
   365→  }
   366→
   367→  hover() {
   368→    if (!this.guard()) return;
   369→    this.tone(1180, { dur: 0.045, type: "sine", vol: 0.035 });
   370→  }
   371→
   372→  click() {
   373→    if (!this.guard()) return;
   374→    this.marimba(NOTE(3, 523.25), this.t0, 0.14);
   375→    this.noise({ dur: 0.03, vol: 0.025, f0: 3600, type: "highpass" });
   376→  }
   377→
   378→  pop() {
   379→    if (!this.guard()) return;
   380→    this.tone(340, { dur: 0.1, type: "sine", vol: 0.16, glide: 720 });
   381→    this.noise({ at: this.t0, dur: 0.05, vol: 0.05, f0: 1600, q: 1.4 });
   382→  }
   383→
   384→  scoreTick() {
   385→    if (!this.guard()) return;
   386→    this.tone(920 + Math.random() * 60, { dur: 0.04, type: "triangle", vol: 0.05 });
   387→  }
   388→
   389→  /** THE iconic cue — marimba major arp + air. Overheard-safe. */
   390→  correct() {
   391→    if (!this.guard()) return;
   392→    const t = this.t0;
   393→    [0, 4, 7].forEach((s, i) => this.marimba(NOTE(s, 523.25), t + i * 0.07, 0.17, i === 0 ? -0.2 : i === 2 ? 0.2 : 0));
   394→    this.noise({ at: t, dur: 0.09, vol: 0.02, f0: 5600, type: "highpass", wet: 0.3 });
   395→  }
   396→
   397→  /** Wrong: a soft horn fall — "hmm", never "NO!" */
   398→  wrong() {
   399→    if (!this.guard()) return;
   400→    const t = this.t0;
   401→    this.horn(NOTE(0, 233.08), t, 0.17, 0.07);
   402→    this.horn(NOTE(-3, 233.08), t + 0.13, 0.26, 0.07);
   403→  }
   404→
   405→  /**
   406→   * Combo escalation family — ONE motif (major-third leap), FOUR levels,
   407→   * root climbing a semitone per streak step (Duolingo principle, §8.3).
   408→   */
   409→  combo(n: number) {
   410→    if (!this.guard()) return;
   411→    const t = this.t0;
   412→    const step = Math.min(n, 12);
   413→    const root = NOTE(step, 523.25);
   414→    this.pluck(root, t, 0.11, -0.25);
   415→    this.pluck(root * 1.26, t + 0.06, 0.14, 0.1);
   416→    if (n >= 5) this.marimba(root * 1.5, t + 0.12, 0.12, 0.3);
   417→    if (n >= 7) {
   418→      this.chime(root * 2, t + 0.17, 0.08, 0.4);
   419→      this.noise({ at: t, dur: 0.22, vol: 0.025, f0: 6400, type: "highpass", wet: 0.4 });
   420→    }
   421→    if (n >= 10) {
   422→      [root * 2, root * 2.52, root * 3].forEach((f, i) =>
   423→        this.pluck(f, t + 0.24 + i * 0.055, 0.09, i === 1 ? 0 : -0.3)
   424→      );
   425→      this.snare(t + 0.24, 0.1);
   426→    }
   427→  }
   428→
   429→  heartLost() {
   430→    if (!this.guard()) return;
   431→    const t = this.t0;
   432→    this.tone(NOTE(3, 392), { at: t, dur: 0.18, type: "sine", vol: 0.1 });
   433→    this.tone(NOTE(0, 392), { at: t + 0.14, dur: 0.3, type: "sine", vol: 0.1 });
   434→    this.thud(t + 0.16, 0.12);
   435→  }
   436→
   437→  /** danger heartbeat — used under the last seconds of the clock */
   438→  pulse() {
   439→    if (!this.guard()) return;
   440→    const t = this.t0;
   441→    this.kick(t, 0.22);
   442→    this.kick(t + 0.22, 0.15);
   443→  }
   444→
   445→  tick(low = false) {
   446→    if (!this.guard()) return;
   447→    this.tone(low ? 880 : 1320, { dur: 0.035, type: "square", vol: 0.04 });
   448→  }
   449→
   450→  pad(i: number) {
   451→    if (!this.guard()) return;
   452→    const f = [261.63, 329.63, 392.0, 523.25][((i % 4) + 4) % 4];
   453→    this.marimba(f, this.t0, 0.2, i < 2 ? -0.3 : 0.3);
   454→  }
   455→
   456→  whoosh() {
   457→    if (!this.guard()) return;
   458→    this.noise({ dur: 0.42, vol: 0.13, f0: 320, f1: 2600, q: 2, wet: 0.25 });
   459→  }
   460→
   461→  star(i: number) {
   462→    if (!this.guard()) return;
   463→    const f = [1046.5, 1318.5, 1568.0][Math.min(i, 2)];
   464→    this.chime(f, this.t0, 0.14);
   465→    this.noise({ at: this.t0 + 0.02, dur: 0.3, vol: 0.02, f0: 7000, type: "highpass", wet: 0.5 });
   466→  }
   467→
   468→  /** the results parade */
   469→  fanfare() {
   470→    if (!this.guard()) return;
   471→    const t = this.t0;
   472→    this.duck();
   473→    const seq: [number, number][] = [[0, 0], [4, 0.12], [7, 0.24], [12, 0.36]];
   474→    seq.forEach(([s, at], i) => {
   475→      this.tone(NOTE(s, 523.25), { at: t + at, dur: 0.26, type: "triangle", vol: 0.15, wet: 0.3 });
   476→      this.tone(NOTE(s - 12, 523.25), { at: t + at, dur: 0.26, type: "sine", vol: 0.09 });
   477→      if (i === seq.length - 1) {
   478→        this.marimba(NOTE(s, 523.25), t + at + 0.16, 0.14);
   479→        this.marimba(NOTE(s + 4, 523.25), t + at + 0.26, 0.12);
   480→      }
   481→    });
   482→    this.snare(t + 0.36, 0.12);
   483→    this.noise({ at: t + 0.36, dur: 0.55, vol: 0.045, f0: 5200, type: "highpass", wet: 0.5 });
   484→  }
   485→
   486→  /** gentle sympathy, not a buzzer */
   487→  sadTrombone() {
   488→    if (!this.guard()) return;
   489→    const t = this.t0;

   716→  /**
   717→   * Request a mood. The request is REMEMBERED: if the browser blocks
   718→   * playback (autoplay policy) we retry on the next gesture / resume,
   719→   * so music reliably starts the moment it is allowed to.
   720→   */
   721→  startMusic(mode: MusicMode) {
   722→    this.desiredTrack = mode;
   723→    this.tryStartMusic();
   724→  }
   725→
   726→  private tryStartMusic() {
   727→    if (typeof window === "undefined" || typeof Audio === "undefined") return;
   728→    if (!this.musicOn || !this.desiredTrack) return;
   729→    const mode = this.desiredTrack;
   730→
   731→    // already on it? make sure it's actually audible
   732→    if (this.currentTrack === mode) {
   733→      const cur = this.tracks[mode];
   734→      if (cur && cur.paused) void cur.play().catch(() => {});
   735→      return;
   736→    }
   737→
   738→    // crossfade out whatever is playing
   739→    const prev = this.currentTrack ? this.tracks[this.currentTrack] : null;
   740→    if (prev) {
   741→      const old = prev;
   742→      this.fadeTo(old, 0, 500, () => {
   743→        old.pause();
   744→        old.currentTime = 0; // replay from the top next time
   745→      });
   746→    }
   747→
   748→    let a = this.tracks[mode];
   749→    if (!a) {
   750→      a = new Audio(MUSIC_SRC[mode]);
   751→      a.loop = true;
   752→      a.preload = "auto";
   753→      this.tracks[mode] = a;
   754→    }
   755→    this.currentTrack = mode;
   756→    a.volume = 0;
   757→    void a
   758→      .play()
   759→      .then(() => this.fadeTo(a!, MUSIC_VOL[mode], 900))
   760→      .catch(() => {
   761→        /* blocked — desiredTrack is remembered, we retry on the next gesture */
   762→      });
   763→  }
   764→
   765→  /** rAF volume ramp — one independent fade per <audio> element, so the
   766→   *  outgoing and incoming tracks of a crossfade never cancel each other. */
   767→  private fadeTo(a: HTMLAudioElement, to: number, ms: number, done?: () => void) {
   768→    const prev = this.fadeRafs.get(a);
   769→    if (prev !== undefined) cancelAnimationFrame(prev);
   770→    const from = a.volume;
   771→    const t0 = performance.now();
   772→    const step = (t: number) => {
   773→      const p = Math.min(1, (t - t0) / ms);
   774→      a.volume = Math.max(0, Math.min(1, from + (to - from) * p));
   775→      if (p < 1) this.fadeRafs.set(a, requestAnimationFrame(step));
   776→      else {
   777→        this.fadeRafs.delete(a);
   778→        done?.();
   779→      }
   780→    };
   781→    this.fadeRafs.set(a, requestAnimationFrame(step));
   782→  }
