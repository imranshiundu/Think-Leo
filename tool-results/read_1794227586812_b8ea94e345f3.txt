     1→     1→/**
     2→     2→ * Think Leo — Save State v4 (zustand + localStorage)
     3→     3→ * ─────────────────────────────────────────────────
     4→     4→ * Offline-first: the whole save lives on the kid's device.
     5→     5→ * Schema + hearts-regen, streak, quest and achievement math are
     6→     6→ * documented in docs/ALGORITHM.md §6–7 and §13.
     7→     7→ *
     8→     8→ * ENGINE RULES (v3 — after the great wipe-bug of v2):
     9→     9→ *  · Every helper is PURE: it takes a state snapshot and returns a new one.
    10→    10→ *    No helper ever re-reads `get()` mid-transaction — that is exactly what
    11→    11→ *    used to wipe stars the millisecond they were recorded.
    12→    12→ *  · The quest board is ALWAYS the length of questsForDay(today) (3 rows,
    13→    13→ *    distinct events) — a mismatch only happens on a genuine day rollover.
    14→    14→ *  · Any XP grant or finished run counts as "played today": the streak,
    15→    15→ *    playDays strip and daily ring can never disagree with each other.
    16→    16→ *
    17→    17→ * V4 — "levels must always end":
    18→    18→ *  · `done[levelIdx]` records ANY finished run (0 stars or practice too),
    19→    19→ *    so finishing a level always opens the next one;
    20→    20→ *  · `boostDay` powers Leo's courage boost (one free full refill per day),
    21→    21→ *    so a kid with 0 hearts is never stuck waiting;
    22→    22→ *  · heart regen is 8 min (was 15) — kids wait badly.
    23→    23→ *
    24→    24→ * V5 — Leo Pop (the arcade) joins the ONE economy:
    25→    25→ *  · merge XP flows through addXp like every other grant — leagues,
    26→    26→ *    streaks, the daily ring and the quest board all stay in sync;
    27→    27→ *  · the Zoo Book (mergeDiscoveries) is a collection, not a currency;
    28→    28→ *  · the daily egg (mergeEggDay) is the comeback hook for the arcade.
    29→    29→ */

   169→   169→/* ══════════════════ pure state helpers (no get(), ever) ══════════ */
   170→   170→
   171→   171→/** A fresh, empty quest board for today. */
   172→   172→function freshBoard(today: string): Pick<LeoSave, "questDay" | "questProgress" | "questClaimed"> {
   173→   173→  const board = questsForDay(today);
   174→   174→  return {
   175→   175→    questDay: today,
   176→   176→    questProgress: board.map(() => 0),
   177→   177→    questClaimed: board.map(() => false),
   178→   178→  };
   179→   179→}
   180→   180→
   181→   181→/** Rebase the snapshot onto today's board if (and only if) it is stale. */
   182→   182→function ensureQuestDay(s: LeoSave): LeoSave {
   183→   183→  const today = dayKey();
   184→   184→  const boardLen = questsForDay(today).length;
   185→   185→  if (s.questDay === today && s.questProgress.length === boardLen) return s;
   186→   186→  return { ...s, ...freshBoard(today) };
   187→   187→}
   188→   188→
   189→   189→/** Bump every quest listening to `event`. `max` mode keeps the highest value. */
   190→   190→function bumpQuests(
   191→   191→  sBase: LeoSave,
   192→   192→  event: string,
   193→   193→  n: number,
   194→   194→  mode: "add" | "max" = "add"
   195→   195→): LeoSave {
   196→   196→  const s = ensureQuestDay(sBase); // preserves every field of sBase
   197→   197→  const board = questsForDay(s.questDay);
   198→   198→  const progress = [...s.questProgress];
   199→   199→  board.forEach((q, i) => {
   200→   200→    if (q.event !== event) return;
   201→   201→    progress[i] = mode === "max" ? Math.max(progress[i] ?? 0, n) : (progress[i] ?? 0) + n;
   202→   202→  });
   203→   203→  return { ...s, questProgress: progress };
   204→   204→}
   205→   205→
   206→   206→/** The day counter: streak, best streak, freezes and the playDays strip. */
   207→   207→function bumpPlayDay(s0: LeoSave): LeoSave {
   208→   208→  const today = dayKey();
   209→   209→  let streak = s0.streak;
   210→   210→  let freezes = s0.freezes;
   211→   211→  let lastPlayDay = s0.lastPlayDay;
   212→   212→
   213→   213→  if (lastPlayDay !== today) {
   214→   214→    // 999 = first-ever play (no yesterday to chain from)
   215→   215→    const gap = lastPlayDay ? daysBetween(lastPlayDay, today) : 999;
   216→   216→    if (gap === 1) {
   217→   217→      streak += 1; // yesterday → today: the streak lives
   218→   218→    } else if (gap === 2 && freezes > 0 && streak > 0) {
   219→   219→      freezes -= 1; // a streak freeze silently saved the streak
   220→   220→      streak += 1;
   221→   221→    } else {
   222→   222→      streak = 1; // fresh start or first ever — kindly begin at 1
   223→   223→    }
   224→   224→    lastPlayDay = today;
   225→   225→  }
   226→   226→  if (streak === 3) freezes += 1; // milestone: earn a freeze
   227→   227→  if (streak === 7) freezes += 2;
   228→   228→
   229→   229→  const playDays = s0.playDays.includes(today)
   230→   230→    ? s0.playDays
   231→   231→    : [...s0.playDays, today].slice(-30);
   232→   232→
   233→   233→  return {
   234→   234→    ...s0,
   235→   235→    streak,
   236→   236→    bestStreak: Math.max(s0.bestStreak, streak),
   237→   237→    freezes,
   238→   238→    lastPlayDay,
   239→   239→    playDays,
   240→   240→    today,
   241→   241→  };
   242→   242→}
   243→   243→
   244→   244→/** XP plumbing without side effects on achievements (they call it too). */
   245→   245→function addXpPure(s0: LeoSave, n: number): { save: LeoSave; leveledTo: number | null } {
   246→   246→  const today = dayKey();
   247→   247→  const wk = weekKey();
   248→   248→  const played = bumpPlayDay(s0);
   249→   249→
   250→   250→  const xp = played.xp + n;
   251→   251→  const todayXp = played.today === today ? played.todayXp + n : n;
   252→   252→  const newLvl = playerLevelFromXp(xp);
   253→   253→  const leveledTo = newLvl > s0.seenPlayerLevel ? newLvl : null;
   254→   254→
   255→   255→  return {
   256→   256→    save: {
   257→   257→      ...played,
   258→   258→      xp,
   259→   259→      todayXp,
   260→   260→      today,
   261→   261→      weekXp: played.leagueWeek === wk ? played.weekXp + n : n,
   262→   262→      leagueWeek: wk,
   263→   263→      seenPlayerLevel: leveledTo ?? played.seenPlayerLevel,
   264→   264→    },
   265→   265→    leveledTo,
   266→   266→  };
   267→   267→}

   269→   269→/** Evaluate one-shot achievements against a snapshot. Pure. */
   270→   270→function checkAchievements(snapshot: LeoSave): { save: LeoSave; unlocked: AchDef[] } {
   271→   271→  const s = snapshot;
   272→   272→  const starLevels = Object.values(s.stars).filter((v) => v > 0).length;
   273→   273→  const chestCount = Object.values(s.chests).filter(Boolean).length;
   274→   274→  const league = leagueOf(s.weekXp);
   275→   275→
   276→   276→  const conditions: Record<string, boolean> = {
   277→   277→    firstStar: starLevels >= 1,
   278→   278→    flawless: Object.values(s.stars).some((v) => v >= 3),
   279→   279→    streak3: s.streak >= 3,
   280→   280→    streak7: s.streak >= 7,
   281→   281→    streak14: s.streak >= 14,
   282→   282→    trail10: starLevels >= 10,
   283→   283→    collector25: starLevels >= 25,
   284→   284→    champion40: starLevels >= 40,
   285→   285→    century: s.totalCorrect >= 100,
   286→   286→    megamind: s.totalCorrect >= 500,
   287→   287→    hunter: chestCount >= 2,
   288→   288→    rising5: playerLevelFromXp(s.xp) >= 5,
   289→   289→    legendLeague: league.index >= 3,
   290→   290→    // Leo Pop
   291→   291→    mergeLeo: (s.mergeDiscoveries[LEO_TIER] ?? 0) >= 1,
   292→   292→    zooAll: Object.keys(s.mergeDiscoveries).filter((k) => Number(k) < ZOO_TARGET).length >= ZOO_TARGET,
   293→   293→  };
   294→   294→
   295→   295→  const unlocked: AchDef[] = [];
   296→   296→  let bonusXp = 0;
   297→   297→  for (const [id, cond] of Object.entries(conditions)) {
   298→   298→    if (cond && !s.ach.includes(id)) {
   299→   299→      const def = ACH_DEFINITIONS[id];
   300→   300→      if (def) {
   301→   301→        unlocked.push(def);
   302→   302→        bonusXp += def.xp;
   303→   303→      }
   304→   304→    }
   305→   305→  }
   306→   306→  if (unlocked.length === 0) return { save: s, unlocked: [] };
   307→   307→  // ids added first → no recursion; XP flows through the same pipe as every
   308→   308→  // other grant (level math AND the daily "Earn XP" quest stay in sync)
   309→   309→  const withIds = { ...s, ach: [...s.ach, ...unlocked.map((a) => a.id)] };
   310→   310→  const added = addXpPure(withIds, bonusXp);
   311→   311→  const quested = bumpQuests(added.save, "xp", bonusXp);
   312→   312→  return { save: quested, unlocked };
   313→   313→}

   315→   315→/* ══════════════════════════ the store ════════════════════════════ */
   316→   316→
   317→   317→export const useLeo = create<LeoStore>()(
   318→   318→  persist(
   319→   319→    (set, get) => ({
   320→   320→      ...DEFAULT_SAVE,
   321→   321→
   322→   322→      tickHearts: () => {
   323→   323→        const s = get();
   324→   324→        if (s.hearts >= MAX_HEARTS) {
   325→   325→          if (s.heartsAt !== 0) set({ heartsAt: Date.now() });
   326→   326→          return;
   327→   327→        }
   328→   328→        const now = Date.now();
   329→   329→        const gained = Math.floor((now - s.heartsAt) / HEART_REGEN_MS);
   330→   330→        if (gained > 0) {
   331→   331→          const hearts = Math.min(MAX_HEARTS, s.hearts + gained);
   332→   332→          set({
   333→   333→            hearts,
   334→   334→            heartsAt: hearts >= MAX_HEARTS ? now : s.heartsAt + gained * HEART_REGEN_MS,
   335→   335→          });
   336→   336→        }
   337→   337→      },
   338→   338→
   339→   339→      loseHeart: () => {
   340→   340→        const s = get();
   341→   341→        if (s.hearts <= 0) return;
   342→   342→        const wasFull = s.hearts >= MAX_HEARTS;
   343→   343→        set({
   344→   344→          hearts: s.hearts - 1,
   345→   345→          heartsAt: wasFull ? Date.now() : s.heartsAt,
   346→   346→        });
   347→   347→      },
   348→   348→
   349→   349→      useRefill: () => {
   350→   350→        const s = get();
   351→   351→        if (s.refills <= 0 || s.hearts >= MAX_HEARTS) return false;
   352→   352→        set({ refills: s.refills - 1, hearts: MAX_HEARTS, heartsAt: Date.now() });
   353→   353→        return true;
   354→   354→      },
   355→   355→
   356→   356→      /**
   357→   357→       * XP always flows through here: it feeds the daily goal ring,
   358→   358→       * the quest board, the weekly league, the streak and the player
   359→   359→       * level — one currency, many meters (the Duolingo trick, §2).
   360→   360→       */
   361→   361→      addXp: (n) => {
   362→   362→        const base = ensureQuestDay(get());
   363→   363→        const quested = bumpQuests(base, "xp", n);
   364→   364→        const { save, leveledTo } = addXpPure(quested, n);
   365→   365→        const { save: finalSave, unlocked } = checkAchievements(save);
   366→   366→        set({
   367→   367→          ...finalSave,
   368→   368→          levelUpPulse: get().levelUpPulse + (leveledTo ? 1 : 0),
   369→   369→          pendingAch: unlocked.length ? [...get().pendingAch, ...unlocked] : get().pendingAch,
   370→   370→        });
   371→   371→      },
   372→   372→
   373→   373→      /** A finished run (any result) counts today as played. */
   374→   374→      noteActivity: () => {
   375→   375→        set(bumpPlayDay(ensureQuestDay(get())));
   376→   376→      },
   377→   377→
   378→   378→      trackAnswer: (correct) => {
   379→   379→        const base = ensureQuestDay(get());
   380→   380→        const quested = correct ? bumpQuests(base, "correct", 1) : base;
   381→   381→        const withCount: LeoSave = correct
   382→   382→          ? { ...quested, totalCorrect: quested.totalCorrect + 1 }
   383→   383→          : quested;
   384→   384→        set(withCount);
   385→   385→        if (correct) {
   386→   386→          const { save, unlocked } = checkAchievements(get());
   387→   387→          if (unlocked.length) {
   388→   388→            set({ ...save, pendingAch: [...get().pendingAch, ...unlocked] });
   389→   389→          }
   390→   390→        }
   391→   391→      },
   392→   392→
   393→   393→      trackCombo: (n) => {
   394→   394→        const base = ensureQuestDay(get());
   395→   395→        set(bumpQuests(base, "maxcombo", n, "max"));
   396→   396→        if (n >= 10 && !get().ach.includes("combo10")) {
   397→   397→          const def = ACH_DEFINITIONS.combo10;
   398→   398→          const withAch = { ...get(), ach: [...get().ach, "combo10"] };
   399→   399→          const { save } = addXpPure(withAch, def.xp);
   400→   400→          set({ ...bumpQuests(save, "xp", def.xp), pendingAch: [...get().pendingAch, def] });
   401→   401→        }
   402→   402→      },
   403→   403→
   404→   404→      trackRush: (hits) => {
   405→   405→        const base = ensureQuestDay(get());
   406→   406→        set(bumpQuests(base, "rushhits", hits, "max"));
   407→   407→        if (hits >= 12 && !get().ach.includes("rushAce")) {
   408→   408→          const def = ACH_DEFINITIONS.rushAce;
   409→   409→          const withAch = { ...get(), ach: [...get().ach, "rushAce"] };
   410→   410→          const { save } = addXpPure(withAch, def.xp);
   411→   411→          set({ ...bumpQuests(save, "xp", def.xp), pendingAch: [...get().pendingAch, def] });
   412→   412→        }
   413→   413→      },
   414→   414→
   415→   415→      /**
   416→   416→       * Records a finished level — ANY finished run (v4): best stars, best
   417→   417→       * score, `done` flag (so the next level always opens), run counter,
   418→   418→       * first-clear quest bumps — then achievement checks. ONE set, built
   419→   419→       * from a single snapshot; nothing can be wiped mid-way any more.
   420→   420→       */
   421→   421→      recordLevel: (levelIdx, stars, score) => {
   422→   422→        const s0 = ensureQuestDay(get());
   423→   423→        const prevStars = s0.stars[levelIdx] ?? 0;
   424→   424→        const prevBest = s0.best[levelIdx] ?? 0;
   425→   425→        const firstClear = prevStars === 0 && stars > 0;
   426→   426→
   427→   427→        let s: LeoSave = {
   428→   428→          ...s0,
   429→   429→          stars: { ...s0.stars, [levelIdx]: Math.max(prevStars, stars) },
   430→   430→          best: { ...s0.best, [levelIdx]: Math.max(prevBest, score) },
   431→   431→          done: { ...s0.done, [levelIdx]: true },
   432→   432→          runs: s0.runs + 1,
   433→   433→        };
   434→   434→        if (firstClear) {
   435→   435→          s = bumpQuests(s, "levels", 1);
   436→   436→          if (stars >= 3) s = bumpQuests(s, "stars3", 1);
   437→   437→        }
   438→   438→        set(s);
   439→   439→
   440→   440→        const { save, unlocked } = checkAchievements(get());
   441→   441→        if (unlocked.length) {
   442→   442→          set({ ...save, pendingAch: [...get().pendingAch, ...unlocked] });
   443→   443→        }
   444→   444→      },
   445→   445→
   446→   446→      claimBoost: () => {
   447→   447→        const s = get();
   448→   448→        const today = dayKey();

   611→   611→      },
   612→   612→      migrate: (persisted, version) => {
   613→   613→        const p = (persisted ?? {}) as Partial<LeoSave>;
   614→   614→        // v2 saves carry XP from the inflated economy (score/12 per run — one
   615→   615→        // run could mint 700+ XP and skip three leagues). Rescale to the v3
   616→   616→        // curve so returning kids land in a sane state without losing progress.
   617→   617→        // (v3+ saves were already rescaled — never touch their XP again.)
   618→   618→        const scale = version < 3 ? 12 : 1;

   632→   632→          hearts: Math.max(p.hearts ?? MAX_HEARTS, 3), // arriving from a fix: be kind
   633→   633→          // v3: boards rebuilt under the exact-3 contract; stars/best survive
   634→   634→          ...freshBoard(dayKey()),
   635→   635→          playDays: p.playDays ?? [],
   636→   636→          ach: p.ach ?? [],
   637→   637→          seenPlayerLevel: playerLevelFromXp(xp),
   638→   638→          comebackDay: "",
   639→   639→          done: { ...done, ...(p.done ?? {}) },
   640→   640→          boostDay: p.boostDay ?? "",
   641→   641→          // v5: the arcade fields (older saves simply start their zoo fresh)
   642→   642→          mergeBest: p.mergeBest ?? 0,
   643→   643→          mergeDiscoveries: p.mergeDiscoveries ?? {},

   645→   645→          mergePlays: p.mergePlays ?? 0,

   647→   647→        };
   648→   648→      },
   649→   649→      /**
   650→   650→       * Hydration merge — THE hydration flag lives here, not in
   651→   651→       * onRehydrateStorage. Why: that callback runs while `create()` is
   652→   652→       * still executing, so referencing `useLeo` inside it dies in the
   653→   653→       * temporal dead zone (an error persist swallows silently). That
   654→   654→       * bug shipped unnoticed for weeks: `_hydrated` never turned true,

   656→   656→       * merge runs for every hydration (stored or fresh) and simply
   657→   657→       * stamps the flag — no store reference needed.
   658→   658→       */
   659→   659→      merge: (persisted, current) =>
   660→   660→        ({
   661→   661→          ...current,
   662→   662→          ...(persisted as Partial<LeoSave>),
   663→   663→          _hydrated: true,
   664→   664→        }) as LeoStore,
   665→   665→      onRehydrateStorage: () => (state) => {

   668→   668→    }
   669→   669→  )
   670→   670→);

   678→   678→/** Convenience selector: hearts display never needs manual math. */
   679→   679→export function useHearts(): { hearts: number; max: number; nextInMin: number } {
   680→   680→  const hearts = useLeo((s) => s.hearts);
   681→   681→  const heartsAt = useLeo((s) => s.heartsAt);
   682→   682→  const tick = useLeo((s) => s.tickHearts);
   683→   683→  // light polling keeps regen honest while the tab is open
   684→   684→  useEffect(() => {
   685→   685→    const id = window.setInterval(tick, 5000);
   686→   686→    return () => window.clearInterval(id);
   687→   687→  }, [tick]);
   688→   688→  if (hearts >= MAX_HEARTS) return { hearts, max: MAX_HEARTS, nextInMin: 0 };
   689→   689→  const nextIn = Math.max(
   690→   690→    0,
   691→   691→    Math.ceil((heartsAt + HEART_REGEN_MS - Date.now()) / 60000)
   692→   692→  );
   693→   693→  return { hearts, max: MAX_HEARTS, nextInMin: nextIn };
   694→   694→}

   696→   696→/** Today's quest board + progress (safe during hydration). */
   697→   697→export function useQuests() {
   698→   698→  const questDay = useLeo((s) => s.questDay);
   699→   699→  const questProgress = useLeo((s) => s.questProgress);
   700→   700→  const questClaimed = useLeo((s) => s.questClaimed);
   701→   701→  const questAllBonusDay = useLeo((s) => s.questAllBonusDay);
   702→   702→  const today = dayKey();
   703→   703→  const board = questsForDay(today);
   704→   704→  const stale = questDay !== today || questProgress.length !== board.length;
   705→   705→  const progress = stale ? board.map(() => 0) : questProgress;
   706→   706→  const claimed = stale ? board.map(() => false) : questClaimed;
   707→   707→  const bonusReady =
   708→   708→    !stale &&
   709→   709→    questAllBonusDay !== today &&
   710→   710→    board.every((q, i) => claimed[i] && (progress[i] ?? 0) >= q.target);
   711→   711→  const allDone = board.every((q, i) => claimed[i] && (progress[i] ?? 0) >= q.target);
   712→   712→  return { board, progress, claimed, bonusReady, allDone, stale, today };
   713→   713→}
