     1→     1→/**
     2→     2→ * Think Leo — Save State v4 (zustand + localStorage)
     3→     3→ * ─────────────────────────────────────────────────
     4→     4→ * Offline-first: the whole save lives on the kid's device.
     5→     5→ * Schema + hearts-regen, streak, quest and achievement math are
     6→     6→ * documented in docs/ALGORITHM.md §6–7 and §13.
     7→     7→ *
     8→     8→ * ENGINE RULES (v3 — after the great wipe-bug of v2):
     9→     9→ *  · Every helper is PURE: it takes a state snapshot and returns a new one.
    10→    10→ *    No helper ever re-reads `get()` mid-transaction — that is exactly what
    11→    11→ *    used to wipe stars the millisecond they were recorded.
    12→    12→ *  · The quest board is ALWAYS the length of questsForDay(today) (3 rows,
    13→    13→ *    distinct events) — a mismatch only happens on a genuine day rollover.
    14→    14→ *  · Any XP grant or finished run counts as "played today": the streak,
    15→    15→ *    playDays strip and daily ring can never disagree with each other.
    16→    16→ *
    17→    17→ * V4 — "levels must always end":
    18→    18→ *  · `done[levelIdx]` records ANY finished run (0 stars or practice too),
    19→    19→ *    so finishing a level always opens the next one;
    20→    20→ *  · `boostDay` powers Leo's courage boost (one free full refill per day),
    21→    21→ *    so a kid with 0 hearts is never stuck waiting;
    22→    22→ *  · heart regen is 8 min (was 15) — kids wait badly.
    23→    23→ *
    24→    24→ * V5 — Leo Pop (the arcade) joins the ONE economy:
    25→    25→ *  · merge XP flows through addXp like every other grant — leagues,
    26→    26→ *    streaks, the daily ring and the quest board all stay in sync;
    27→    27→ *  · the Zoo Book (mergeDiscoveries) is a collection, not a currency;
    28→    28→ *  · the daily egg (mergeEggDay) is the comeback hook for the arcade.
    29→    29→ *
    30→    30→ * V6 — LEO EGGS (the surprise-egg arcade) joins the same ONE economy:
    31→    31→ *  · every hatch pays rarity XP through the same pipe (daily ring,
    32→    32→ *    league, streak, quests) and duplicate stars into its own purse;
    33→    33→ *  · the egg bag is a collection with memory (count + evolved stage);
    34→    34→ *  · the nest (eggNestAt) is the real-time comeback engine — it keeps
    35→    35→ *    incubating while the app is closed;
    36→    36→ *  · the pity counter (eggPity) guarantees rare-or-better every 8
    37→    37→ *    free Spark eggs — losing streaks always end.
    38→    38→ */

   218→   218→/** Distinct egg friends discovered (bag keys that are real critters). */
   219→   219→function eggDistinct(s: LeoSave): number {
   220→   220→  return Object.keys(s.eggBag).filter((k) => CRITTERS_BY_UID[k]).length;
   221→   221→}
   222→   222→
   223→   223→/** A fresh, empty quest board for today. */
   224→   224→function freshBoard(today: string): Pick<LeoSave, "questDay" | "questProgress" | "questClaimed"> {
   225→   225→  const board = questsForDay(today);
   226→   226→  return {
   227→   227→    questDay: today,
   228→   228→    questProgress: board.map(() => 0),
   229→   229→    questClaimed: board.map(() => false),
   230→   230→  };
   231→   231→}
   232→   232→
   233→   233→/** Rebase the snapshot onto today's board if (and only if) it is stale. */
   234→   234→function ensureQuestDay(s: LeoSave): LeoSave {
   235→   235→  const today = dayKey();
   236→   236→  const boardLen = questsForDay(today).length;
   237→   237→  if (s.questDay === today && s.questProgress.length === boardLen) return s;
   238→   238→  return { ...s, ...freshBoard(today) };
   239→   239→}
   240→   240→
   241→   241→/** Bump every quest listening to `event`. `max` mode keeps the highest value. */
   242→   242→function bumpQuests(
   243→   243→  sBase: LeoSave,
   244→   244→  event: string,
   245→   245→  n: number,
   246→   246→  mode: "add" | "max" = "add"
   247→   247→): LeoSave {
   248→   248→  const s = ensureQuestDay(sBase); // preserves every field of sBase
   249→   249→  const board = questsForDay(s.questDay);
   250→   250→  const progress = [...s.questProgress];
   251→   251→  board.forEach((q, i) => {
   252→   252→    if (q.event !== event) return;
   253→   253→    progress[i] = mode === "max" ? Math.max(progress[i] ?? 0, n) : (progress[i] ?? 0) + n;
   254→   254→  });
   255→   255→  return { ...s, questProgress: progress };
   256→   256→}
   257→   257→
   258→   258→/** The day counter: streak, best streak, freezes and the playDays strip. */
   259→   259→function bumpPlayDay(s0: LeoSave): LeoSave {
   260→   260→  const today = dayKey();
   261→   261→  let streak = s0.streak;
   262→   262→  let freezes = s0.freezes;
   263→   263→  let lastPlayDay = s0.lastPlayDay;
   264→   264→
   265→   265→  if (lastPlayDay !== today) {
   266→   266→    // 999 = first-ever play (no yesterday to chain from)
   267→   267→    const gap = lastPlayDay ? daysBetween(lastPlayDay, today) : 999;
   268→   268→    if (gap === 1) {
   269→   269→      streak += 1; // yesterday → today: the streak lives
   270→   270→    } else if (gap === 2 && freezes > 0 && streak > 0) {
   271→   271→      freezes -= 1; // a streak freeze silently saved the streak
   272→   272→      streak += 1;
   273→   273→    } else {
   274→   274→      streak = 1; // fresh start or first ever — kindly begin at 1
   275→   275→    }
   276→   276→    lastPlayDay = today;
   277→   277→  }
   278→   278→  if (streak === 3) freezes += 1; // milestone: earn a freeze
   279→   279→  if (streak === 7) freezes += 2;
   280→   280→
   281→   281→  const playDays = s0.playDays.includes(today)
   282→   282→    ? s0.playDays
   283→   283→    : [...s0.playDays, today].slice(-30);
   284→   284→
   285→   285→  return {
   286→   286→    ...s0,
   287→   287→    streak,
   288→   288→    bestStreak: Math.max(s0.bestStreak, streak),
   289→   289→    freezes,
   290→   290→    lastPlayDay,
   291→   291→    playDays,
   292→   292→    today,
   293→   293→  };
   294→   294→}

   296→   296→/** XP plumbing without side effects on achievements (they call it too). */
   297→   297→function addXpPure(s0: LeoSave, n: number): { save: LeoSave; leveledTo: number | null } {
   298→   298→  const today = dayKey();
   299→   299→  const wk = weekKey();
   300→   300→  const played = bumpPlayDay(s0);
   301→   301→
   302→   302→  const xp = played.xp + n;
   303→   303→  const todayXp = played.today === today ? played.todayXp + n : n;
   304→   304→  const newLvl = playerLevelFromXp(xp);
   305→   305→  const leveledTo = newLvl > s0.seenPlayerLevel ? newLvl : null;
   306→   306→
   307→   307→  return {
   308→   308→    save: {
   309→   309→      ...played,
   310→   310→      xp,
   311→   311→      todayXp,
   312→   312→      today,
   313→   313→      weekXp: played.leagueWeek === wk ? played.weekXp + n : n,
   314→   314→      leagueWeek: wk,
   315→   315→      seenPlayerLevel: leveledTo ?? played.seenPlayerLevel,
   316→   316→    },
   317→   317→    leveledTo,
   318→   318→  };
   319→   319→}

   321→   321→/** Evaluate one-shot achievements against a snapshot. Pure. */
   322→   322→function checkAchievements(snapshot: LeoSave): { save: LeoSave; unlocked: AchDef[] } {
   323→   323→  const s = snapshot;
   324→   324→  const starLevels = Object.values(s.stars).filter((v) => v > 0).length;
   325→   325→  const chestCount = Object.values(s.chests).filter(Boolean).length;
   326→   326→  const league = leagueOf(s.weekXp);
   327→   327→
   328→   328→  const conditions: Record<string, boolean> = {
   329→   329→    firstStar: starLevels >= 1,
   330→   330→    flawless: Object.values(s.stars).some((v) => v >= 3),
   331→   331→    streak3: s.streak >= 3,
   332→   332→    streak7: s.streak >= 7,
   333→   333→    streak14: s.streak >= 14,
   334→   334→    trail10: starLevels >= 10,
   335→   335→    collector25: starLevels >= 25,
   336→   336→    champion40: starLevels >= 40,
   337→   337→    century: s.totalCorrect >= 100,
   338→   338→    megamind: s.totalCorrect >= 500,
   339→   339→    hunter: chestCount >= 2,
   340→   340→    rising5: playerLevelFromXp(s.xp) >= 5,
   341→   341→    legendLeague: league.index >= 3,
   342→   342→    // Leo Pop
   343→   343→    mergeLeo: (s.mergeDiscoveries[LEO_TIER] ?? 0) >= 1,
   344→   344→    zooAll: Object.keys(s.mergeDiscoveries).filter((k) => Number(k) < ZOO_TARGET).length >= ZOO_TARGET,
   345→   345→    // Leo Eggs
   346→   346→    eggFirst: s.eggHatched >= 1,
   347→   347→    eggExplorer: eggDistinct(s) >= 6,
   348→   348→    eggMaster: eggDistinct(s) >= COLLECTION_TARGET,
   349→   349→    eggLegend: CRITTERS.some((c) => (c.rarity === "legend" || c.rarity === "mythic") && (s.eggBag[c.uid]?.count ?? 0) >= 1),
   350→   350→    eggEvolve: Object.values(s.eggBag).some((c) => c.evolved),
   351→   351→  };
   352→   352→
   353→   353→  const unlocked: AchDef[] = [];
   354→   354→  let bonusXp = 0;
   355→   355→  for (const [id, cond] of Object.entries(conditions)) {
   356→   356→    if (cond && !s.ach.includes(id)) {
   357→   357→      const def = ACH_DEFINITIONS[id];
   358→   358→      if (def) {
   359→   359→        unlocked.push(def);
   360→   360→        bonusXp += def.xp;
   361→   361→      }
   362→   362→    }
   363→   363→  }
   364→   364→  if (unlocked.length === 0) return { save: s, unlocked: [] };
   365→   365→  // ids added first → no recursion; XP flows through the same pipe as every
   366→   366→  // other grant (level math AND the daily "Earn XP" quest stay in sync)
   367→   367→  const withIds = { ...s, ach: [...s.ach, ...unlocked.map((a) => a.id)] };
   368→   368→  const added = addXpPure(withIds, bonusXp);
   369→   369→  const quested = bumpQuests(added.save, "xp", bonusXp);
   370→   370→  return { save: quested, unlocked };
   371→   371→}

   374→   374→
   375→   375→export const useLeo = create<LeoStore>()(
   376→   376→  persist(
   377→   377→    (set, get) => ({
   378→   378→      ...DEFAULT_SAVE,
   379→   379→
   380→   380→      tickHearts: () => {
   381→   381→        const s = get();
   382→   382→        if (s.hearts >= MAX_HEARTS) {
   383→   383→          if (s.heartsAt !== 0) set({ heartsAt: Date.now() });
   384→   384→          return;
   385→   385→        }
   386→   386→        const now = Date.now();
   387→   387→        const gained = Math.floor((now - s.heartsAt) / HEART_REGEN_MS);
   388→   388→        if (gained > 0) {
   389→   389→          const hearts = Math.min(MAX_HEARTS, s.hearts + gained);
   390→   390→          set({
   391→   391→            hearts,
   392→   392→            heartsAt: hearts >= MAX_HEARTS ? now : s.heartsAt + gained * HEART_REGEN_MS,
   393→   393→          });
   394→   394→        }
   395→   395→      },
   396→   396→
   397→   397→      loseHeart: () => {
   398→   398→        const s = get();
   399→   399→        if (s.hearts <= 0) return;
   400→   400→        const wasFull = s.hearts >= MAX_HEARTS;
   401→   401→        set({
   402→   402→          hearts: s.hearts - 1,
   403→   403→          heartsAt: wasFull ? Date.now() : s.heartsAt,
   404→   404→        });
   405→   405→      },
   406→   406→
   407→   407→      useRefill: () => {
   408→   408→        const s = get();
   409→   409→        if (s.refills <= 0 || s.hearts >= MAX_HEARTS) return false;
   410→   410→        set({ refills: s.refills - 1, hearts: MAX_HEARTS, heartsAt: Date.now() });
   411→   411→        return true;
   412→   412→      },
   413→   413→
   414→   414→      /**
   415→   415→       * XP always flows through here: it feeds the daily goal ring,
   416→   416→       * the quest board, the weekly league, the streak and the player
   417→   417→       * level — one currency, many meters (the Duolingo trick, §2).
   418→   418→       */
   419→   419→      addXp: (n) => {
   420→   420→        const base = ensureQuestDay(get());
   421→   421→        const quested = bumpQuests(base, "xp", n);
   422→   422→        const { save, leveledTo } = addXpPure(quested, n);
   423→   423→        const { save: finalSave, unlocked } = checkAchievements(save);
   424→   424→        set({
   425→   425→          ...finalSave,
   426→   426→          levelUpPulse: get().levelUpPulse + (leveledTo ? 1 : 0),
   427→   427→          pendingAch: unlocked.length ? [...get().pendingAch, ...unlocked] : get().pendingAch,
   428→   428→        });
   429→   429→      },
   430→   430→
   431→   431→      /** A finished run (any result) counts today as played. */
   432→   432→      noteActivity: () => {
   433→   433→        set(bumpPlayDay(ensureQuestDay(get())));
   434→   434→      },
   435→   435→
   436→   436→      trackAnswer: (correct) => {
   437→   437→        const base = ensureQuestDay(get());
   438→   438→        const quested = correct ? bumpQuests(base, "correct", 1) : base;
   439→   439→        const withCount: LeoSave = correct
   440→   440→          ? { ...quested, totalCorrect: quested.totalCorrect + 1 }
   441→   441→          : quested;

   813→   813→);

   818→   818→  (window as unknown as { __leoStore?: typeof useLeo }).__leoStore = useLeo;

   821→   821→/** Convenience selector: hearts display never needs manual math. */
   822→   822→export function useHearts(): { hearts: number; max: number; nextInMin: number } {
   823→   823→  const hearts = useLeo((s) => s.hearts);
   824→   824→  const heartsAt = useLeo((s) => s.heartsAt);
   825→   825→  const tick = useLeo((s) => s.tickHearts);
   826→   826→  // light polling keeps regen honest while the tab is open
   827→   827→  useEffect(() => {
   828→   828→    const id = window.setInterval(tick, 5000);
   829→   829→    return () => window.clearInterval(id);
   830→   830→  }, [tick]);
   831→   831→  if (hearts >= MAX_HEARTS) return { hearts, max: MAX_HEARTS, nextInMin: 0 };
   832→   832→  const nextIn = Math.max(
   833→   833→    0,
   834→   834→    Math.ceil((heartsAt + HEART_REGEN_MS - Date.now()) / 60000)
   835→   835→  );
   836→   836→  return { hearts, max: MAX_HEARTS, nextInMin: nextIn };
   837→   837→}

   839→   839→/** Today's quest board + progress (safe during hydration). */
   840→   840→export function useQuests() {
   841→   841→  const questDay = useLeo((s) => s.questDay);
   842→   842→  const questProgress = useLeo((s) => s.questProgress);
   843→   843→  const questClaimed = useLeo((s) => s.questClaimed);
   844→   844→  const questAllBonusDay = useLeo((s) => s.questAllBonusDay);
   845→   845→  const today = dayKey();
   846→   846→  const board = questsForDay(today);
   847→   847→  const stale = questDay !== today || questProgress.length !== board.length;
   848→   848→  const progress = stale ? board.map(() => 0) : questProgress;
   849→   849→  const claimed = stale ? board.map(() => false) : questClaimed;
   850→   850→  const bonusReady =
   851→   851→    !stale &&
   852→   852→    questAllBonusDay !== today &&
   853→   853→    board.every((q, i) => claimed[i] && (progress[i] ?? 0) >= q.target);
   854→   854→  const allDone = board.every((q, i) => claimed[i] && (progress[i] ?? 0) >= q.target);
   855→   855→  return { board, progress, claimed, bonusReady, allDone, stale, today };
   856→   856→}

   873→   873→/**
   874→   874→ * The nest, live: a 1s ticker that only runs while the app is open.
   875→   875→ * `ready` means a free egg is waiting in the nest right now — the Home
   876→   876→ * card and the egg game both read this one truth.
   877→   877→ */
   878→   878→export function useEggNest(): NestState & { hydrated: boolean } {
   879→   879→  const eggNestAt = useLeo((s) => s.eggNestAt);
   880→   880→  const hydrated = useLeo((s) => s._hydrated);
   881→   881→  const [now, setNow] = useState(() => Date.now());
   882→   882→  useEffect(() => {
   883→   883→    if (!hydrated) return;
   884→   884→    const id = window.setInterval(() => setNow(Date.now()), 1000);
   885→   885→    return () => window.clearInterval(id);
   886→   886→  }, [hydrated]);
   887→   887→  if (!hydrated) return { ready: false, pct: 0, minsLeft: 0, hydrated: false };
   888→   888→  return { ...nestState(eggNestAt, now), hydrated: true };
   889→   889→}

   899→   899→export function useComeback(): boolean {

   902→   902→  const hydrated = useLeo((s) => s._hydrated);
