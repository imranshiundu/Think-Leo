    53→
    54→type Phase = "intro" | "play" | "reveal" | "rushIntro" | "rush" | "out" | "results";
    55→
    56→interface Results {
    57→  stars: number;
    58→  xp: number;
    59→  accuracy: number;
    60→  total: number;
    61→  isBest: boolean;
    62→  chest: boolean;
    63→  practice: boolean;
    64→  levelUpTo: number | null;
    65→  leagueUpTo: string | null;
    66→  rushHits: number;
    67→}

    71→const OOPS = ["Almost!", "So close!", "No worries —"];
    72→const NO_HEARTS = [
    73→  "Hearts refill over time — brains need naps too!",
    74→  "Out of hearts? Stretch, sip water, come back roaring!",
    75→];
    76→

    93→
    94→export function Play({
    95→  levelIdx,
    96→  onExit,
    97→  onNext,
    98→  hasNext,
    99→}: {
   100→  levelIdx: number;
   101→  onExit: () => void;
   102→  onNext: () => void;
   103→  hasNext: boolean;
   104→}) {
   105→  const world = worldOf(levelIdx);
   106→  const plan = useMemo(() => buildPlan(levelIdx), [levelIdx]);
   107→  const attemptRef = useRef(1);
   108→
   109→  /* snapshot for level-up / league-up detection */
   110→  const xpBeforeRef = useRef(0);
   111→  const leagueBeforeRef = useRef("");
   112→  useEffect(() => {
   113→    const s = useLeo.getState();
   114→    xpBeforeRef.current = s.xp;
   115→    leagueBeforeRef.current = leagueOf(s.weekXp).name;
   116→  }, []);
   117→
   118→  /* ── run state ── */
   119→  const [phase, setPhase] = useState<Phase>("intro");
   120→  const [qi, setQi] = useState(0);
   121→  const [boost, setBoost] = useState(0); // adaptive difficulty offset (-1..+2)
   122→  const [combo, setCombo] = useState(0);
   123→  const [maxCombo, setMaxCombo] = useState(0);
   124→  const [score, setScore] = useState(0);
   125→  const [correctCount, setCorrectCount] = useState(0);
   126→  const [answeredCount, setAnsweredCount] = useState(0);
   127→  const [rightStreak, setRightStreak] = useState(0);
   128→  const [wrongStreak, setWrongStreak] = useState(0);
   129→  const [picked, setPicked] = useState<number | null>(null);
   130→  const [revealOk, setRevealOk] = useState(false);
   131→  const [shakeKey, setShakeKey] = useState(0);
   132→  const [leoMood, setLeoMood] = useState<Mood>("idle");
   133→  const [bubble, setBubble] = useState<string | null>(null);
   134→  const [confettiKey, setConfettiKey] = useState(0);
   135→  /**
   136→   * Practice mode (v4): entered from the out-of-hearts screen — hearts are
   137→   * frozen, XP is halved, and the level ALWAYS ends with a results screen.
   138→   * No kid is ever hard-stuck by the heart economy again.
   139→   */
   140→  const [practice, setPractice] = useState(false);
   141→  /* floating "+120" score pops and combo banners (the candy layer) */
   142→  const [pops, setPops] = useState<{ id: number; pts: number }[]>([]);
   143→  const [banner, setBanner] = useState<{ id: number; text: string } | null>(null);
   144→
   145→  /* rush state */
   146→  const [rushQs, setRushQs] = useState<Question[]>([]);
   147→  const [rushIdx, setRushIdx] = useState(0);
   148→  const [rushCombo, setRushCombo] = useState(0);
   149→  const [rushScore, setRushScore] = useState(0);
   150→  const [rushHits, setRushHits] = useState(0);
   151→  const [rushLeft, setRushLeft] = useState(RUSH_SECONDS);
   152→
   153→  const [results, setResults] = useState<Results | null>(null);
   154→  const recordedRef = useRef(false);
   155→
   156→  const diff = clamp(
   157→    Math.max(1, Math.min(6, 1 + Math.floor((levelIdx - 1) / 8) + boost)),
   158→    1,
   159→    6
   160→  );
   161→  const q = useMemo(
   162→    () =>
   163→      phase === "play" || phase === "reveal"
   164→        ? makeQuestion(plan[qi], diff, levelIdx, qi, attemptRef.current)
   165→        : null,
   166→    [phase, qi, diff, levelIdx, plan]
   167→  );
   168→
   169→  const resetRun = useCallback(
   170→    (fresh: boolean, asPractice = false) => {
   171→      if (fresh) attemptRef.current += 1;
   172→      recordedRef.current = false;
   173→      setPractice(asPractice);
   174→      setPhase("intro");
   175→      setQi(0);
   176→      setBoost(0);
   177→      setCombo(0);
   178→      setMaxCombo(0);
   179→      setScore(0);
   180→      setCorrectCount(0);
   181→      setAnsweredCount(0);
   182→      setRightStreak(0);
   183→      setWrongStreak(0);
   184→      setPicked(null);
   185→      setRushQs([]);
   186→      setRushIdx(0);
   187→      setRushCombo(0);
   188→      setRushScore(0);
   189→      setRushHits(0);
   190→      setResults(null);
   191→      setLeoMood("idle");
   192→      setBubble(null);
   193→    },
   194→    []
   195→  );
   196→
   197→  /* keyboard answering (laptop love) */
   198→  const answerRef = useRef<(i: number) => void>(() => {});
   199→  useEffect(() => {
   200→    const h = (e: KeyboardEvent) => {
   201→      const n = Number(e.key);
   202→      if (phase === "play" && n >= 1 && n <= 4) answerRef.current(n - 1);
   203→    };
   204→    window.addEventListener("keydown", h);
   205→    return () => window.removeEventListener("keydown", h);
   206→  }, [phase]);
   207→
   208→  /* music per screen */
   209→  useEffect(() => {
   210→    sfx.startMusic("play");
   211→    return () => {
   212→      sfx.startMusic("home");
   213→    };
   214→  }, []);
   215→
   216→  /* ── answer handling ─────────────────────────────────────────── */
   217→
   218→  const advance = useCallback(
   219→    (ok: boolean) => {
   220→      const heartsNow = useLeo.getState().hearts;
   221→      if (!ok && heartsNow <= 0 && !practice) {
   222→        setLeoMood("sad");
   223→        setBubble(NO_HEARTS[Math.floor(Math.random() * NO_HEARTS.length)]);
   224→        setPhase("out");
   225→        sfx.sadTrombone();
   226→        return;
   227→      }
   228→      if (qi >= ROUND_COUNT - 1) {
   229→        setLeoMood("wow");
   230→        setBubble("LEO RUSH! Everything ×2 — GO GO GO!");
   231→        setPhase("rushIntro");
   232→        sfx.whoosh();
   233→        sfx.rushGo();
   234→        window.setTimeout(() => {
   235→          try {
   236→            // 200 rapid-fire questions: even the fastest kid cannot outrun the clock
   237→            setRushQs(Array.from({ length: 200 }, (_, i) => makeRushQuestion(diff, i, attemptRef.current)));
   238→            setRushIdx(0);
   239→            setRushLeft(RUSH_SECONDS);
   240→            setPhase("rush");
   241→          } catch {
   242→            // audio/animation can never strand a level — jump straight to results
   243→            setPhase("results");
   244→          }
   245→        }, 1700);
   246→      } else {
   247→        setQi((n) => n + 1);
   248→        setPicked(null);
   249→        setLeoMood("idle");
   250→        setBubble(null);
   251→        setPhase("play");
   252→      }
   253→    },
   254→    [qi, diff, practice]
   255→  );
   256→
   257→  const handleAnswer = useCallback(
   258→    (idx: number | null) => {
   259→      if (!q || phase !== "play") return;
   260→      const fracLeft = clamp(timeLeftRef.current / q.time, 0, 1);
   261→      const ok = idx !== null && idx === q.answer;
   262→      setPicked(idx);
   263→      setRevealOk(ok);
   264→      setAnsweredCount((a) => a + 1);
   265→      setPhase("reveal");
   266→      const store = useLeo.getState();
   267→
   268→      if (ok) {
   269→        const pts = questionPoints(combo + 1, fracLeft);
   270→        setScore((s) => s + pts);
   271→        setCorrectCount((c) => c + 1);
   272→        const nc = combo + 1;
   273→        setCombo(nc);
   274→        setMaxCombo((m) => Math.max(m, nc));
   275→        setRightStreak((r) => {
   276→          const nr = r + 1;
   277→          if (nr % 3 === 0) setBoost((b) => Math.min(2, b + 1)); // Flow Ranger: heat up
   278→          return nr;
   279→        });
   280→        setWrongStreak(0);
   281→        store.trackAnswer(true);
   282→        store.trackCombo(nc);
   283→        sfx.correct();
   284→        if (nc >= 3) window.setTimeout(() => sfx.combo(nc), 140);
   285→        setLeoMood(nc >= 5 ? "excited" : "happy");
   286→        setBubble(nc >= 5 ? UNSTOPPABLE[nc % UNSTOPPABLE.length] : PRAISE[nc % PRAISE.length]);
   287→        /* juice: floating "+points" and milestone banners */
   288→        const popId = Date.now() + Math.random();
   289→        setPops((p) => [...p, { id: popId, pts }]);
   290→        window.setTimeout(() => setPops((p) => p.filter((x) => x.id !== popId)), 950);
   291→        if (nc === 5 || nc === 8 || nc === 10) {
   292→          const bid = Date.now() + Math.random();
   293→          setBanner({ id: bid, text: UNSTOPPABLE[nc % UNSTOPPABLE.length] });
   294→          window.setTimeout(() => setBanner((b) => (b && b.id === bid ? null : b)), 1200);
   295→        }
   296→      } else {
   297→        // v4 kindness: running out of TIME on the learning levels (1-2)
   298→        // never costs a heart — young kids are slow readers, not wrong.
   299→        const gentleTimeout = idx === null && levelIdx <= GENTLE_LEVELS;
   300→        if (!practice && !gentleTimeout) store.loseHeart();
   301→        store.trackAnswer(false);
   302→        setCombo(0);
   303→        setWrongStreak((w) => {
   304→          const nw = w + 1;
   305→          if (nw % 2 === 0) setBoost((b) => Math.max(-1, b - 1)); // Flow Ranger: cool down
   306→          return nw;
   307→        });
   308→        setRightStreak(0);
   309→        sfx.wrong();
   310→        if (!practice && !gentleTimeout) window.setTimeout(() => sfx.heartLost(), 240);
   311→        setShakeKey((k) => k + 1);
   312→        setLeoMood("sad");
   313→        setBubble(
   314→          idx === null
   315→            ? gentleTimeout
   316→              ? "Time's up — no heart lost! Try the next one:"
   317→              : "Time's up! Here's the trick:"
   318→            : `${OOPS[Math.floor(Math.random() * OOPS.length)]} ${q.tip}`
   319→        );
   320→      }
   321→
   322→      window.setTimeout(() => advance(ok), ok ? 900 : 1600);
   323→    },
   324→    [q, phase, combo, advance]
   325→  );
   326→
   327→  answerRef.current = (i: number) => {
   328→    if (!q || phase !== "play") return;
   329→    if (q.kind === "memory") return;
   330→    if (q.options.length && i < q.options.length) handleAnswer(i);
   331→  };
   332→
   333→  /* countdown ring */
   334→  const timeLeftRef = useRef(0);
   335→  const [timeLeft, setTimeLeft] = useState(10);
   336→  useEffect(() => {
   337→    if (phase !== "play" || !q) return;
   338→    const total = q.time;
   339→    setTimeLeft(total);
   340→    timeLeftRef.current = total;
   341→    const start = performance.now();
   342→    let lastTick = Math.ceil(total);
   343→    const id = window.setInterval(() => {
   344→      const left = total - (performance.now() - start) / 1000;
   345→      timeLeftRef.current = Math.max(0, left);
   346→      setTimeLeft(Math.max(0, left));
   347→      const whole = Math.ceil(left);
   348→      if (whole !== lastTick && whole <= 5 && whole > 0) sfx.tick();
   349→      lastTick = whole;
   350→      if (left <= 0) {
   351→        window.clearInterval(id);
   352→        handleAnswer(null);
   353→      }
   354→    }, 100);
   355→    return () => window.clearInterval(id);
   356→  }, [phase, qi, q, handleAnswer]);
   357→
   358→  /* ── LEO RUSH ────────────────────────────────────────────────── */
   359→
   360→  const endRush = useCallback(() => {
   361→    const s = useLeo.getState();
   362→    const total = score + rushScore;
   363→    const accuracy = answeredCount > 0 ? correctCount / answeredCount : 0;
   364→    const stars = starsFor(accuracy);
   365→    const rawXp = stars > 0 ? xpForRun(total, stars) : 0;
   366→    // practice runs pay half XP — the level still ENDS, stars still land,
   367→    // but the honest way to earn big is with hearts on the line
   368→    const xp = practice ? Math.ceil(rawXp / 2) : rawXp;
   369→    const isBest = stars > 0 && !practice && total > (s.best[levelIdx] ?? 0);
   370→    // ANY finished run counts today as played — streaks record on effort,
   371→    // not only on wins (a kid who tried and missed must keep their flame).
   372→    s.noteActivity();
   373→    if (!recordedRef.current) {
   374→      recordedRef.current = true;
   375→      if (rushHits > 0) s.trackRush(rushHits);
   376→      // v4: EVERY finished run is recorded — the `done` flag it sets opens
   377→      // the next level, so a level can never fail to end in progress.
   378→      s.recordLevel(levelIdx, stars, total);
   379→      if (xp > 0) s.addXp(xp);
   380→    }
   381→    const chest = stars > 0 && !practice && chestAfterLevel(levelIdx);
   382→    if (chest) s.grantChest(world.id);
   383→
   384→    const sAfter = useLeo.getState();
   385→    const lvlUp = playerLevelFromXp(sAfter.xp) > playerLevelFromXp(xpBeforeRef.current)
   386→      ? playerLevelFromXp(sAfter.xp)
   387→      : null;
   388→    const leagueUp = leagueOf(sAfter.weekXp).name !== leagueBeforeRef.current
   389→      ? leagueOf(sAfter.weekXp).name
   390→      : null;
   391→
   392→    setResults({ stars, xp, accuracy, total, isBest, chest, practice, levelUpTo: lvlUp, leagueUpTo: leagueUp, rushHits });
   393→    setPhase("results");
   394→    if (stars > 0) {
   395→      sfx.fanfare();
   396→      if (leagueUp) window.setTimeout(() => sfx.leagueUp(), 700);
   397→      setConfettiKey((k) => k + 1);
   398→      setLeoMood(stars === 3 ? "cheer" : "celebrate");
   399→      setBubble(stars === 3 ? "PERFECT RUN! I'm roaring the roof off!" : "Level crushed! More stars next time?");
   400→      for (let i = 0; i < stars; i++) window.setTimeout(() => sfx.star(i), 600 + i * 380);
   401→    } else {
   402→      sfx.sadTrombone();
   403→      setLeoMood("sad");
   404→      setBubble("So close! One more try — I'll cheer for you!");
   405→    }
   406→  }, [score, rushScore, answeredCount, correctCount, rushHits, levelIdx, world.id, practice]);
   407→
   408→  /* ── THE RUSH CLOCK ────────────────────────────────────────────
   409→     endRush is rebuilt on every rush hit (its deps include rushScore
   410→     and rushHits), so this effect MUST NOT depend on it — otherwise
   411→     every correct tap would restart the 12 seconds and the level
   412→     would never, ever end. We call the latest endRush through a ref
   413→     and key the interval on the phase alone. */
   414→  const endRushRef = useRef(endRush);
   415→  endRushRef.current = endRush;
   416→  useEffect(() => {
   417→    if (phase !== "rush") return;
   418→    const start = performance.now();
   419→    const id = window.setInterval(() => {
   420→      const left = RUSH_SECONDS - (performance.now() - start) / 1000;
   421→      setRushLeft(Math.max(0, left));
   422→      if (left <= 0) {
   423→        window.clearInterval(id);
   424→        endRushRef.current();
   425→      }
   426→    }, 100);
   427→    return () => window.clearInterval(id);
   428→  }, [phase]);
   429→
   430→  const rushAnswer = useCallback(
   431→    (idx: number | null) => {
   432→      if (idx === null) return; // no timeout penalty inside the rush
   433→      const rq = rushQs[rushIdx];
   434→      if (!rq) return;
   435→      const ok = idx === rq.answer;
   436→      if (ok) {
   437→        setRushScore((s) => s + 120);
   438→        setRushHits((h) => h + 1);
   439→        setRushCombo((c) => {
   440→          const nc = c + 1;
   441→          sfx.combo(nc + 2);
   442→          return nc;
   443→        });
   444→      } else {
   445→        setRushCombo(0);
   446→        sfx.tap();
   447→      }
   448→      if (rushIdx < rushQs.length - 1) setRushIdx((i) => i + 1);
   449→    },
   450→    [rushQs, rushIdx]
   451→  );
   452→
   453→  const hearts = useLeo((s) => s.hearts);
   454→  const refills = useLeo((s) => s.refills);
   455→  const doRefill = useLeo((s) => s.useRefill);
   456→  const doBoost = useLeo((s) => s.claimBoost);
   457→  const canBoost = useLeo((s) => s.boostDay !== dayKey() && s.hearts < MAX_HEARTS);
   458→
   459→  /* ── render ──────────────────────────────────────────────────── */
   460→
   461→  const qTimeTotal = q?.time ?? 10;
   462→  const danger = phase === "play" && timeLeft <= 3.5;
   463→  const hotCombo = combo >= 5 || (phase === "rush" && rushCombo >= 4);
   464→  const rushQ = phase === "rush" ? rushQs[rushIdx] : null;
   465→  const parade = WORLD_TILES[world.id] ?? WORLD_TILES[0];
   466→
   467→  /* heartbeat under danger */
   468→  useEffect(() => {
   469→    if (!danger) return;
   470→    sfx.pulse();
   471→    const id = window.setInterval(() => sfx.pulse(), 1100);
   472→    return () => window.clearInterval(id);
   473→  }, [danger]);
   474→
   475→  /* music mood follows the phase: rush gets its own driving loop */
   476→  useEffect(() => {
   477→    if (phase === "rush") sfx.startMusic("rush");
   478→    else if (phase === "results" || phase === "out") sfx.startMusic("home");
   479→  }, [phase]);
   480→
   481→  return (
   482→    <div
   483→      className="relative h-screen w-full overflow-hidden"
   484→      style={{ background: `linear-gradient(180deg, ${world.wash[0]}, ${world.wash[1]})` }}
   485→    >
   486→      <QuestWatch />
   487→      {/* living background: world-flavoured cartoons drifting behind the action */}
   488→      {world.id === 2 ? (
   489→        /* Space Hop gets a lazy-loaded 3D sky (three.js) behind everything */
   490→        <SpaceSky className="pointer-events-none absolute inset-0 z-0" opacity={0.55} />
   491→      ) : null}
   492→      <AmbientCartoons world={world.id} count={8} opacity={0.15} />
   493→      {/* heartbeat vignette */}
   494→      {danger && (
   495→        <div
   496→          className="pointer-events-none absolute inset-0 z-10"
   497→          style={{ boxShadow: "inset 0 0 90px rgba(255,60,90,0.45)" }}
   498→        />
   499→      )}
   500→      {hotCombo && <div className="combo-hot pointer-events-none absolute inset-0 z-10" />}
   501→      {/* LEO RUSH: racing light-streaks + pulsing aura */}
   502→      {phase === "rush" && (
   503→        <>
   504→          <div className="rush-lines pointer-events-none absolute inset-0 z-10" />
   505→          <div className="rush-aura pointer-events-none absolute inset-0 z-10" />
   506→        </>
   507→      )}
   508→
   509→      <ConfettiBurst fireKey={confettiKey} />
   510→
   511→      {/* floating +points */}
   512→      <div className="pointer-events-none absolute inset-x-0 top-16 z-30 flex justify-center">
   513→        <AnimatePresence>
   514→          {pops.map((p) => (
   515→            <motion.span

   911→/** Pops "Quest complete!" the moment a daily quest fills up mid-run. */
   912→function QuestWatch() {
   913→  const { board, progress, claimed } = useQuests();
   914→  const seen = useRef<Record<string, boolean>>({});
   915→  const [toast, setToast] = useState<{ label: string; key: number } | null>(null);
   916→
   917→  useEffect(() => {
   918→    board.forEach((q, i) => {
   919→      const done = (progress[i] ?? 0) >= q.target && !claimed[i];
   920→      if (done && !seen.current[q.id]) {
   921→        seen.current[q.id] = true;
   922→        setToast({ label: q.label, key: Date.now() });
   923→        sfx.questDone();
   924→        window.setTimeout(() => setToast(null), 2800);
   925→      }
   926→      if (!done) seen.current[q.id] = false;
   927→    });
   928→  }, [board, progress, claimed]);
   929→
   930→  return (
   931→    <AnimatePresence>
   932→      {toast && (
   933→        <motion.div
   934→          key={toast.key}
   935→          initial={{ y: -60, opacity: 0, scale: 0.9 }}
   936→          animate={{ y: 0, opacity: 1, scale: 1 }}
   937→          exit={{ y: -40, opacity: 0 }}
   938→          transition={{ type: "spring", stiffness: 300, damping: 20 }}
   939→          className="absolute left-1/2 top-16 z-[30] -translate-x-1/2"
   940→        >
   941→          <div className="flex items-center gap-2 rounded-2xl border-[3px] border-leo-sun bg-white px-4 py-2 shadow-[0_6px_0_rgba(214,178,110,0.4)]">
   942→            <PartyPopper width={16} height={16} className="text-leo-sun-deep" />
   943→            <span className="font-display text-sm font-bold text-leo-ink">Quest complete!</span>
   944→            <span className="text-xs text-leo-ink-soft">{toast.label}</span>
   945→          </div>
   946→        </motion.div>
   947→      )}
   948→    </AnimatePresence>
   949→  );
   950→}

   961→function TimerRing({ left, total }: { left: number; total: number }) {
   962→  const r = 16;
   963→  const c = 2 * Math.PI * r;
   964→  const frac = clamp(left / total, 0, 1);
   965→  const low = left <= 5;
   966→  return (
   967→    <div className={`relative h-11 w-11 ${low ? "anim-heartbeat" : ""}`}>
   968→      <svg width="44" height="44" viewBox="0 0 44 44" className="-rotate-90">
   969→        <circle cx="22" cy="22" r={r} fill="#fff" stroke="#F0E0BD" strokeWidth="5" />
   970→        <circle
   971→          cx="22"
   972→          cy="22"
   973→          r={r}
   974→          fill="none"
   975→          stroke={low ? "#ff5a8c" : "#58cc02"}
   976→          strokeWidth="5"
   977→          strokeLinecap="round"
   978→          strokeDasharray={c}
   979→          strokeDashoffset={c * (1 - frac)}

  1252→  const [watching, setWatching] = useState(true);
  1253→  const [inputStep, setInputStep] = useState(0);

  1361→/** Tap-the-target: pop every target in a mixed crowd. Wrong taps just boing. */
  1362→function TapTargetGame({
  1363→  tiles,
  1364→  target,
  1365→  disabled,
  1366→  onResult,
  1367→}: {
  1368→  tiles: StickerId[];
  1369→  target: StickerId;
  1370→  disabled: boolean;
  1371→  onResult: (ok: boolean) => void;
  1372→}) {
  1373→  const total = tiles.filter((t) => t === target).length;
  1374→  const [found, setFound] = useState<number[]>([]);
  1375→  const [miss, setMiss] = useState<number | null>(null);
  1376→  const doneRef = useRef(false);
  1377→
  1378→  const tap = (i: number) => {
  1379→    if (disabled || doneRef.current || found.includes(i)) return;
  1380→    if (tiles[i] === target) {
  1381→      const nf = [...found, i];
  1382→      setFound(nf);
  1383→      sfx.found(nf.length - 1);
  1384→      if (nf.length === total) {
  1385→        doneRef.current = true;
  1386→        window.setTimeout(() => sfx.correct(), 120);
  1387→        window.setTimeout(() => onResult(true), 650);
  1388→      }
  1389→    } else {
  1390→      setMiss(i);
  1391→      sfx.boing();
  1392→      window.setTimeout(() => setMiss((m) => (m === i ? null : m)), 480);
  1393→    }
  1394→  };
  1395→
  1396→  return (
  1397→    <div className="py-1">
  1398→      <div className="mb-2 flex justify-center">
  1399→        <span className="pill gap-1.5 px-3 py-1 text-sm font-bold text-leo-leaf-deep">
  1400→          <Target width={14} height={14} />
  1401→          {found.length} / {total} found
  1402→        </span>
  1403→      </div>
  1404→      <div className="mx-auto flex max-w-md flex-wrap justify-center gap-1.5">
  1405→        {tiles.map((t, i) => {
  1406→          const isFound = found.includes(i);
  1407→          return (
  1408→            <motion.button
  1409→              key={i}
  1410→              type="button"
  1411→              aria-hidden={isFound}
  1412→              tabIndex={-1}
  1413→              onClick={() => tap(i)}
  1414→              animate={
  1415→                isFound
  1416→                  ? { scale: 0, opacity: 0, rotate: 40 }
  1417→                  : miss === i
  1418→                    ? { x: [0, -5, 5, -3, 0] }
  1419→                    : { scale: 1, opacity: 1, rotate: 0 }
  1420→              }
  1421→              transition={{ duration: 0.28 }}
  1422→              whileHover={!isFound ? { scale: 1.1 } : undefined}
  1423→              whileTap={!isFound ? { scale: 0.9 } : undefined}
  1424→              className={`flex h-[52px] w-[52px] cursor-pointer items-center justify-center rounded-xl border-[2.5px] ${
  1425→                isFound ? "border-leo-leaf bg-[#e9ffcf]" : "border-transparent bg-white/70"
  1426→              }`}
  1427→            >
  1428→              <Sticker id={t} size={42} />
  1429→            </motion.button>
  1430→          );
  1431→        })}
  1432→      </div>
  1433→    </div>
  1434→  );
  1435→}

  1437→/** Feed Leo: drop snacks one by one, count out loud, tap DONE when it matches. */
  1438→function FeedLeoGame({
  1439→  snack,
  1440→  want,
  1441→  disabled,
  1442→  onResult,
  1443→}: {
  1444→  snack: StickerId;
  1445→  want: number;
  1446→  disabled: boolean;
  1447→  onResult: (ok: boolean) => void;
  1448→}) {
  1449→  const [bowl, setBowl] = useState(0);
  1450→  const [done, setDone] = useState(false);
  1451→  const perfect = bowl === want;
  1452→  const CAP = want + 4; // room to overshoot, so kids can fix their count
  1453→
  1454→  const drop = () => {
  1455→    if (disabled || done || bowl >= CAP) return;
  1456→    setBowl((b) => b + 1);

  1460→    if (disabled || done) return;
  1461→    setBowl((b) => Math.max(0, b - 1));

  1463→  };
  1464→  const finish = () => {
  1465→    if (disabled || done) return;
  1466→    setDone(true);
  1467→    const ok = bowl === want;
  1468→    window.setTimeout(() => (ok ? sfx.correct() : sfx.wrong()), 80);
  1469→    window.setTimeout(() => onResult(ok), 600);
  1470→  };
  1471→
  1472→  return (
  1473→    <div className="flex flex-col items-center gap-2 py-1">
  1474→      <div className="flex items-end justify-center gap-3">
  1475→        <div className="shrink-0">
  1476→          <Leo mood={perfect ? "happy" : "thinking"} size={96} />
  1477→        </div>
  1478→        {/* Leo's bowl */}
  1479→        <div className="relative flex flex-col items-center">
  1480→          <motion.div
  1481→            key={bowl}
  1482→            initial={{ scale: perfect ? [1, 1.12, 1] : 1 }}
  1483→            animate={{ scale: 1 }}
  1484→            className="relative h-[86px] w-[168px]"
  1485→          >
  1486→            {/* bowl */}
  1487→            <svg viewBox="0 0 168 86" className="absolute inset-0">
