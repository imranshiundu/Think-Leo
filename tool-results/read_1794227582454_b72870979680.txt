     1→/**
     2→ * Think Leo — Save State v4 (zustand + localStorage)
     3→ * ─────────────────────────────────────────────────
     4→ * Offline-first: the whole save lives on the kid's device.
     5→ * Schema + hearts-regen, streak, quest and achievement math are
     6→ * documented in docs/ALGORITHM.md §6–7 and §13.
     7→ *
     8→ * ENGINE RULES (v3 — after the great wipe-bug of v2):
     9→ *  · Every helper is PURE: it takes a state snapshot and returns a new one.
    10→ *    No helper ever re-reads `get()` mid-transaction — that is exactly what
    11→ *    used to wipe stars the millisecond they were recorded.
    12→ *  · The quest board is ALWAYS the length of questsForDay(today) (3 rows,
    13→ *    distinct events) — a mismatch only happens on a genuine day rollover.
    14→ *  · Any XP grant or finished run counts as "played today": the streak,
    15→ *    playDays strip and daily ring can never disagree with each other.
    16→ *
    17→ * V4 — "levels must always end":
    18→ *  · `done[levelIdx]` records ANY finished run (0 stars or practice too),
    19→ *    so finishing a level always opens the next one;
    20→ *  · `boostDay` powers Leo's courage boost (one free full refill per day),
    21→ *    so a kid with 0 hearts is never stuck waiting;
    22→ *  · heart regen is 8 min (was 15) — kids wait badly.
    23→ *
    24→ * V5 — Leo Pop (the arcade) joins the ONE economy:
    25→ *  · merge XP flows through addXp like every other grant — leagues,
    26→ *    streaks, the daily ring and the quest board all stay in sync;
    27→ *  · the Zoo Book (mergeDiscoveries) is a collection, not a currency;
    28→ *  · the daily egg (mergeEggDay) is the comeback hook for the arcade.
    29→ */

   171→/** A fresh, empty quest board for today. */
   172→function freshBoard(today: string): Pick<LeoSave, "questDay" | "questProgress" | "questClaimed"> {
   173→  const board = questsForDay(today);
   174→  return {
   175→    questDay: today,
   176→    questProgress: board.map(() => 0),
   177→    questClaimed: board.map(() => false),
   178→  };
   179→}
   180→
   181→/** Rebase the snapshot onto today's board if (and only if) it is stale. */
   182→function ensureQuestDay(s: LeoSave): LeoSave {
   183→  const today = dayKey();
   184→  const boardLen = questsForDay(today).length;
   185→  if (s.questDay === today && s.questProgress.length === boardLen) return s;
   186→  return { ...s, ...freshBoard(today) };
   187→}
   188→
   189→/** Bump every quest listening to `event`. `max` mode keeps the highest value. */
   190→function bumpQuests(
   191→  sBase: LeoSave,
   192→  event: string,
   193→  n: number,
   194→  mode: "add" | "max" = "add"
   195→): LeoSave {
   196→  const s = ensureQuestDay(sBase); // preserves every field of sBase
   197→  const board = questsForDay(s.questDay);
   198→  const progress = [...s.questProgress];
   199→  board.forEach((q, i) => {
   200→    if (q.event !== event) return;
   201→    progress[i] = mode === "max" ? Math.max(progress[i] ?? 0, n) : (progress[i] ?? 0) + n;
   202→  });
   203→  return { ...s, questProgress: progress };
   204→}
   205→
   206→/** The day counter: streak, best streak, freezes and the playDays strip. */
   207→function bumpPlayDay(s0: LeoSave): LeoSave {
   208→  const today = dayKey();
   209→  let streak = s0.streak;
   210→  let freezes = s0.freezes;
   211→  let lastPlayDay = s0.lastPlayDay;
   212→
   213→  if (lastPlayDay !== today) {
   214→    // 999 = first-ever play (no yesterday to chain from)
   215→    const gap = lastPlayDay ? daysBetween(lastPlayDay, today) : 999;
   216→    if (gap === 1) {
   217→      streak += 1; // yesterday → today: the streak lives
   218→    } else if (gap === 2 && freezes > 0 && streak > 0) {
   219→      freezes -= 1; // a streak freeze silently saved the streak
   220→      streak += 1;
   221→    } else {
   222→      streak = 1; // fresh start or first ever — kindly begin at 1
   223→    }
   224→    lastPlayDay = today;
   225→  }
   226→  if (streak === 3) freezes += 1; // milestone: earn a freeze
   227→  if (streak === 7) freezes += 2;
   228→
   229→  const playDays = s0.playDays.includes(today)
   230→    ? s0.playDays
   231→    : [...s0.playDays, today].slice(-30);
   232→
   233→  return {
   234→    ...s0,
   235→    streak,
   236→    bestStreak: Math.max(s0.bestStreak, streak),
   237→    freezes,
   238→    lastPlayDay,
   239→    playDays,
   240→    today,
   241→  };
   242→}

   244→/** XP plumbing without side effects on achievements (they call it too). */
   245→function addXpPure(s0: LeoSave, n: number): { save: LeoSave; leveledTo: number | null } {
   246→  const today = dayKey();
   247→  const wk = weekKey();
   248→  const played = bumpPlayDay(s0);
   249→
   250→  const xp = played.xp + n;
   251→  const todayXp = played.today === today ? played.todayXp + n : n;
   252→  const newLvl = playerLevelFromXp(xp);
   253→  const leveledTo = newLvl > s0.seenPlayerLevel ? newLvl : null;
   254→
   255→  return {
   256→    save: {
   257→      ...played,
   258→      xp,
   259→      todayXp,
   260→      today,
   261→      weekXp: played.leagueWeek === wk ? played.weekXp + n : n,
   262→      leagueWeek: wk,
   263→      seenPlayerLevel: leveledTo ?? played.seenPlayerLevel,
   264→    },
   265→    leveledTo,
   266→  };
   267→}

   269→/** Evaluate one-shot achievements against a snapshot. Pure. */
   270→function checkAchievements(snapshot: LeoSave): { save: LeoSave; unlocked: AchDef[] } {
   271→  const s = snapshot;
   272→  const starLevels = Object.values(s.stars).filter((v) => v > 0).length;
   273→  const chestCount = Object.values(s.chests).filter(Boolean).length;
   274→  const league = leagueOf(s.weekXp);
   275→
   276→  const conditions: Record<string, boolean> = {
   277→    firstStar: starLevels >= 1,
   278→    flawless: Object.values(s.stars).some((v) => v >= 3),
   279→    streak3: s.streak >= 3,
   280→    streak7: s.streak >= 7,
   281→    streak14: s.streak >= 14,
   282→    trail10: starLevels >= 10,
   283→    collector25: starLevels >= 25,
   284→    champion40: starLevels >= 40,
   285→    century: s.totalCorrect >= 100,
   286→    megamind: s.totalCorrect >= 500,
   287→    hunter: chestCount >= 2,
   288→    rising5: playerLevelFromXp(s.xp) >= 5,
   289→    legendLeague: league.index >= 3,
   290→    // Leo Pop
   291→    mergeLeo: (s.mergeDiscoveries[LEO_TIER] ?? 0) >= 1,
   292→    zooAll: Object.keys(s.mergeDiscoveries).filter((k) => Number(k) < ZOO_TARGET).length >= ZOO_TARGET,
   293→  };
   294→
   295→  const unlocked: AchDef[] = [];
   296→  let bonusXp = 0;
   297→  for (const [id, cond] of Object.entries(conditions)) {
   298→    if (cond && !s.ach.includes(id)) {
   299→      const def = ACH_DEFINITIONS[id];
   300→      if (def) {
   301→        unlocked.push(def);
   302→        bonusXp += def.xp;
   303→      }
   304→    }
   305→  }
   306→  if (unlocked.length === 0) return { save: s, unlocked: [] };
   307→  // ids added first → no recursion; XP flows through the same pipe as every
   308→  // other grant (level math AND the daily "Earn XP" quest stay in sync)
   309→  const withIds = { ...s, ach: [...s.ach, ...unlocked.map((a) => a.id)] };
   310→  const added = addXpPure(withIds, bonusXp);
   311→  const quested = bumpQuests(added.save, "xp", bonusXp);
   312→  return { save: quested, unlocked };
   313→}

   315→/* ══════════════════════════ the store ════════════════════════════ */
   316→
   317→export const useLeo = create<LeoStore>()(
   318→  persist(
   319→    (set, get) => ({
   320→      ...DEFAULT_SAVE,
   321→
   322→      tickHearts: () => {
   323→        const s = get();
   324→        if (s.hearts >= MAX_HEARTS) {
   325→          if (s.heartsAt !== 0) set({ heartsAt: Date.now() });
   326→          return;
   327→        }
   328→        const now = Date.now();
   329→        const gained = Math.floor((now - s.heartsAt) / HEART_REGEN_MS);
   330→        if (gained > 0) {
   331→          const hearts = Math.min(MAX_HEARTS, s.hearts + gained);
   332→          set({
   333→            hearts,
   334→            heartsAt: hearts >= MAX_HEARTS ? now : s.heartsAt + gained * HEART_REGEN_MS,
   335→          });
   336→        }
   337→      },
   338→
   339→      loseHeart: () => {
   340→        const s = get();
   341→        if (s.hearts <= 0) return;
   342→        const wasFull = s.hearts >= MAX_HEARTS;
   343→        set({
   344→          hearts: s.hearts - 1,
   345→          heartsAt: wasFull ? Date.now() : s.heartsAt,
   346→        });
   347→      },
   348→
   349→      useRefill: () => {
   350→        const s = get();
   351→        if (s.refills <= 0 || s.hearts >= MAX_HEARTS) return false;
   352→        set({ refills: s.refills - 1, hearts: MAX_HEARTS, heartsAt: Date.now() });
   353→        return true;
   354→      },
   355→
   356→      /**
   357→       * XP always flows through here: it feeds the daily goal ring,
   358→       * the quest board, the weekly league, the streak and the player
   359→       * level — one currency, many meters (the Duolingo trick, §2).
   360→       */
   361→      addXp: (n) => {
   362→        const base = ensureQuestDay(get());
   363→        const quested = bumpQuests(base, "xp", n);
   364→        const { save, leveledTo } = addXpPure(quested, n);
   365→        const { save: finalSave, unlocked } = checkAchievements(save);
   366→        set({
   367→          ...finalSave,
   368→          levelUpPulse: get().levelUpPulse + (leveledTo ? 1 : 0),
   369→          pendingAch: unlocked.length ? [...get().pendingAch, ...unlocked] : get().pendingAch,
   370→        });
   371→      },
   372→
   373→      /** A finished run (any result) counts today as played. */
   374→      noteActivity: () => {
   375→        set(bumpPlayDay(ensureQuestDay(get())));
   376→      },
   377→
   378→      trackAnswer: (correct) => {
   379→        const base = ensureQuestDay(get());
   380→        const quested = correct ? bumpQuests(base, "correct", 1) : base;
   381→        const withCount: LeoSave = correct
   382→          ? { ...quested, totalCorrect: quested.totalCorrect + 1 }
   383→          : quested;
   384→        set(withCount);
   385→        if (correct) {
   386→          const { save, unlocked } = checkAchievements(get());
   387→          if (unlocked.length) {
   388→            set({ ...save, pendingAch: [...get().pendingAch, ...unlocked] });
   389→          }
   390→        }
   391→      },
   392→
   393→      trackCombo: (n) => {
   394→        const base = ensureQuestDay(get());
   395→        set(bumpQuests(base, "maxcombo", n, "max"));
   396→        if (n >= 10 && !get().ach.includes("combo10")) {
   397→          const def = ACH_DEFINITIONS.combo10;
   398→          const withAch = { ...get(), ach: [...get().ach, "combo10"] };
   399→          const { save } = addXpPure(withAch, def.xp);
   400→          set({ ...bumpQuests(save, "xp", def.xp), pendingAch: [...get().pendingAch, def] });
   401→        }
   402→      },
   403→
   404→      trackRush: (hits) => {
   405→        const base = ensureQuestDay(get());
   406→        set(bumpQuests(base, "rushhits", hits, "max"));
   407→        if (hits >= 12 && !get().ach.includes("rushAce")) {
   408→          const def = ACH_DEFINITIONS.rushAce;
   409→          const withAch = { ...get(), ach: [...get().ach, "rushAce"] };
   410→          const { save } = addXpPure(withAch, def.xp);
   411→          set({ ...bumpQuests(save, "xp", def.xp), pendingAch: [...get().pendingAch, def] });
   412→        }
   413→      },
   414→
   415→      /**
   416→       * Records a finished level — ANY finished run (v4): best stars, best
   417→       * score, `done` flag (so the next level always opens), run counter,
   418→       * first-clear quest bumps — then achievement checks. ONE set, built
   419→       * from a single snapshot; nothing can be wiped mid-way any more.
   420→       */
   421→      recordLevel: (levelIdx, stars, score) => {
   422→        const s0 = ensureQuestDay(get());
   423→        const prevStars = s0.stars[levelIdx] ?? 0;
   424→        const prevBest = s0.best[levelIdx] ?? 0;
   425→        const firstClear = prevStars === 0 && stars > 0;
   426→
   427→        let s: LeoSave = {
   428→          ...s0,
   429→          stars: { ...s0.stars, [levelIdx]: Math.max(prevStars, stars) },
   430→          best: { ...s0.best, [levelIdx]: Math.max(prevBest, score) },
   431→          done: { ...s0.done, [levelIdx]: true },
   432→          runs: s0.runs + 1,
   433→        };
   434→        if (firstClear) {
   435→          s = bumpQuests(s, "levels", 1);
   436→          if (stars >= 3) s = bumpQuests(s, "stars3", 1);
   437→        }
   438→        set(s);
   439→
   440→        const { save, unlocked } = checkAchievements(get());
   441→        if (unlocked.length) {
   442→          set({ ...save, pendingAch: [...get().pendingAch, ...unlocked] });
   443→        }
   444→      },
   445→
   446→      claimBoost: () => {
   447→        const s = get();
   448→        const today = dayKey();
   449→        if (s.boostDay === today || s.hearts >= MAX_HEARTS) return false;
   450→        set({
   451→          ...s,
   452→          boostDay: today,
   453→          hearts: MAX_HEARTS,
   454→          heartsAt: Date.now(),
   455→        });
   456→        get().addXp(BOOST_XP);
   457→        return true;
   458→      },
   459→
   460→      boostReady: () => {
   461→        const s = get();
   462→        return s.boostDay !== dayKey() && s.hearts < MAX_HEARTS;
   463→      },
   464→
   465→      grantChest: (worldId) => {
   466→        const s = get();
   467→        if (s.chests[worldId]) return;
   468→        set({
   469→          chests: { ...s.chests, [worldId]: true },
   470→          refills: s.refills + CHEST_REWARDS.refills,
   471→        });
   472→        get().addXp(CHEST_REWARDS.xp);
   473→      },
   474→
   475→      claimQuest: (i) => {
   476→        const s = ensureQuestDay(get());
   477→        const board = questsForDay(s.questDay);
   478→        const q = board[i];
   479→        if (!q) return false;
   480→        if (s.questClaimed[i] || (s.questProgress[i] ?? 0) < q.target) return false;
   481→        const claimed = [...s.questClaimed];
   482→        claimed[i] = true;
   483→        set({ ...s, questClaimed: claimed });
   484→        get().addXp(q.rewardXp);
   485→        return true;
   486→      },
   487→
   488→      claimQuestBonus: () => {
   489→        const s = ensureQuestDay(get());
   490→        const board = questsForDay(s.questDay);
   491→        const allDone = board.every(
   492→          (q, i) => s.questClaimed[i] && (s.questProgress[i] ?? 0) >= q.target
   493→        );
   494→        if (!allDone || s.questAllBonusDay === s.questDay) return false;
   495→        set({
   496→          ...s,
   497→          questAllBonusDay: s.questDay,
   498→          refills: s.refills + QUEST_ALL_DONE_BONUS.refill,
   499→        });
   500→        get().addXp(QUEST_ALL_DONE_BONUS.xp);
   501→        return true;
   502→      },
   503→
   504→      claimComeback: () => {
   505→        const s = get();
   506→        const today = dayKey();
   507→        if (s.comebackDay === today) return false;
   508→        set({ ...s, comebackDay: today, refills: s.refills + 1 });
   509→        get().addXp(20);
   510→        return true;
   511→      },
   512→
   513→      /* ── Leo Pop (the arcade) ─────────────────────────────────── */
   514→
   515→      trackMerge: (tier, chain) => {
   516→        const s0 = ensureQuestDay(get());
   517→        const disc = { ...s0.mergeDiscoveries };
   518→        const isNew = !disc[tier];
   519→        disc[tier] = (disc[tier] ?? 0) + 1;
   520→
   521→        let s: LeoSave = {
   522→          ...s0,
   523→          mergeDiscoveries: disc,
   524→          totalMerges: s0.totalMerges + 1,
   525→        };
   526→        s = bumpQuests(s, "merges", 1);
   527→        if (isNew) s = bumpQuests(s, "mergeNew", 1);
   528→        if (tier === LEO_TIER) s = bumpQuests(s, "mergeLeo", 1);
   529→        set(s);
   530→
   531→        // direct-unlock achievements (same pattern as combo10 / rushAce)
   532→        const unlock = (id: string, cond: boolean) => {
   533→          if (!cond || get().ach.includes(id)) return;
   534→          const def = ACH_DEFINITIONS[id];
   535→          if (!def) return;
   536→          const withAch: LeoSave = { ...get(), ach: [...get().ach, id] };
   537→          const { save } = addXpPure(withAch, def.xp);
   538→          set({ ...bumpQuests(save, "xp", def.xp), pendingAch: [...get().pendingAch, def] });
   539→        };
   540→        unlock("mergeLeo", tier === LEO_TIER);
   541→        unlock("mergeChain5", chain >= 5);
   542→
   543→        if (isNew) {
   544→          const { save, unlocked } = checkAchievements(get());
   545→          if (unlocked.length) {
   546→            set({ ...save, pendingAch: [...get().pendingAch, ...unlocked] });
   547→          }
   548→        }
   549→        return isNew;
   550→      },
   551→
   552→      recordMergeRun: (score) => {
   553→        const s0 = ensureQuestDay(get());
   554→        set({
   555→          ...s0,
   556→          mergeBest: Math.max(s0.mergeBest, score),
   557→          mergePlays: s0.mergePlays + 1,
   558→        });
   559→        // Pop Star (1500 in one run) — direct unlock like combo10
   560→        if (score >= 1500 && !get().ach.includes("mergeScore")) {
   561→          const def = ACH_DEFINITIONS.mergeScore;
   562→          const withAch: LeoSave = { ...get(), ach: [...get().ach, "mergeScore"] };
   563→          const { save } = addXpPure(withAch, def.xp);
   564→          set({ ...bumpQuests(save, "xp", def.xp), pendingAch: [...get().pendingAch, def] });
   565→        }
   566→      },
   567→
   568→      claimMergeEgg: () => {
   569→        const s = get();
   570→        const today = dayKey();
   571→        if (s.mergeEggDay === today) return false;
   572→        set({ ...s, mergeEggDay: today });
   573→        return true;
   574→      },
   575→
   576→      seeCreature: (tier) => {
   577→        const s0 = ensureQuestDay(get());
   578→        if (s0.mergeDiscoveries[tier]) return false;
   579→        let s: LeoSave = {
   580→          ...s0,
   581→          mergeDiscoveries: { ...s0.mergeDiscoveries, [tier]: 1 },
   582→        };
   583→        s = bumpQuests(s, "mergeNew", 1);
   584→        set(s);
   585→        const { save, unlocked } = checkAchievements(get());
   586→        if (unlocked.length) {
   587→          set({ ...save, pendingAch: [...get().pendingAch, ...unlocked] });
   588→        }
   589→        return true;
   590→      },
   591→
   592→      pendingAch: [],
   593→      dismissAch: () => set({ pendingAch: get().pendingAch.slice(1) }),
   594→      levelUpPulse: 0,
   595→
   596→      toggleSfx: () => set({ soundSfx: !get().soundSfx }),
   597→      toggleMusic: () => set({ soundMusic: !get().soundMusic }),
   598→
   599→      reset: () => set({ ...DEFAULT_SAVE, _hydrated: true, pendingAch: [], levelUpPulse: 0 }),
   600→    }),
   601→    {
   602→      name: "think-leo-save-v1",
   603→      version: 5,
   604→      storage: createJSONStorage(() => localStorage),
   605→      partialize: (s) => {
   606→        const { _hydrated, pendingAch, levelUpPulse, ...rest } = s;
   607→        void _hydrated;
   608→        void pendingAch;
   609→        void levelUpPulse;
   610→        return rest as LeoStore;
   611→      },
   612→      migrate: (persisted, version) => {
   613→        const p = (persisted ?? {}) as Partial<LeoSave>;
   614→        // v2 saves carry XP from the inflated economy (score/12 per run — one
   615→        // run could mint 700+ XP and skip three leagues). Rescale to the v3
   616→        // curve so returning kids land in a sane state without losing progress.
   617→        // (v3+ saves were already rescaled — never touch their XP again.)
   618→        const scale = version < 3 ? 12 : 1;
   619→        const xp = Math.round((p.xp ?? 0) / scale);
   620→        const weekXp = Math.round((p.weekXp ?? 0) / scale);
   621→        // v4: backfill `done` from stars so old saves keep every unlocked door
   622→        const done: Record<number, boolean> = {};
   623→        Object.entries(p.stars ?? {}).forEach(([k, v]) => {
   624→          if (v > 0) done[Number(k)] = true;
   625→        });
   626→        return {
   627→          ...DEFAULT_SAVE,
   628→          ...p,
   629→          xp,
   630→          weekXp,
   631→          todayXp: Math.round((p.todayXp ?? 0) / scale),
   632→          hearts: Math.max(p.hearts ?? MAX_HEARTS, 3), // arriving from a fix: be kind
   633→          // v3: boards rebuilt under the exact-3 contract; stars/best survive
   634→          ...freshBoard(dayKey()),
   635→          playDays: p.playDays ?? [],
   636→          ach: p.ach ?? [],
   637→          seenPlayerLevel: playerLevelFromXp(xp),
   638→          comebackDay: "",
   639→          done: { ...done, ...(p.done ?? {}) },
   640→          boostDay: p.boostDay ?? "",
   641→          // v5: the arcade fields (older saves simply start their zoo fresh)
   642→          mergeBest: p.mergeBest ?? 0,
   643→          mergeDiscoveries: p.mergeDiscoveries ?? {},
   644→          mergeEggDay: p.mergeEggDay ?? "",
   645→          mergePlays: p.mergePlays ?? 0,
   646→          totalMerges: p.totalMerges ?? 0,
   647→        };
   648→      },
   649→      /**
   650→       * Hydration merge — THE hydration flag lives here, not in
   651→       * onRehydrateStorage. Why: that callback runs while `create()` is
   652→       * still executing, so referencing `useLeo` inside it dies in the
   653→       * temporal dead zone (an error persist swallows silently). That
   654→       * bug shipped unnoticed for weeks: `_hydrated` never turned true,
   655→       * so the welcome-back gift and the quick splash never ran. The
   656→       * merge runs for every hydration (stored or fresh) and simply
   657→       * stamps the flag — no store reference needed.
   658→       */
   659→      merge: (persisted, current) =>
   660→        ({
   661→          ...current,
   662→          ...(persisted as Partial<LeoSave>),
   663→          _hydrated: true,
   664→        }) as LeoStore,
   665→      onRehydrateStorage: () => (state) => {
   666→        state?.tickHearts();
   667→      },
   668→    }
   669→  )
   670→);

   678→/** Convenience selector: hearts display never needs manual math. */
   679→export function useHearts(): { hearts: number; max: number; nextInMin: number } {
   680→  const hearts = useLeo((s) => s.hearts);
   681→  const heartsAt = useLeo((s) => s.heartsAt);
   682→  const tick = useLeo((s) => s.tickHearts);
   683→  // light polling keeps regen honest while the tab is open
   684→  useEffect(() => {
   685→    const id = window.setInterval(tick, 5000);
   686→    return () => window.clearInterval(id);
   687→  }, [tick]);
   688→  if (hearts >= MAX_HEARTS) return { hearts, max: MAX_HEARTS, nextInMin: 0 };
   689→  const nextIn = Math.max(
   690→    0,
   691→    Math.ceil((heartsAt + HEART_REGEN_MS - Date.now()) / 60000)
   692→  );
   693→  return { hearts, max: MAX_HEARTS, nextInMin: nextIn };
   694→}

   696→/** Today's quest board + progress (safe during hydration). */
   697→export function useQuests() {
   698→  const questDay = useLeo((s) => s.questDay);
   699→  const questProgress = useLeo((s) => s.questProgress);
   700→  const questClaimed = useLeo((s) => s.questClaimed);
   701→  const questAllBonusDay = useLeo((s) => s.questAllBonusDay);
   702→  const today = dayKey();
   703→  const board = questsForDay(today);
   704→  const stale = questDay !== today || questProgress.length !== board.length;
   705→  const progress = stale ? board.map(() => 0) : questProgress;
   706→  const claimed = stale ? board.map(() => false) : questClaimed;
   707→  const bonusReady =
   708→    !stale &&
   709→    questAllBonusDay !== today &&
   710→    board.every((q, i) => claimed[i] && (progress[i] ?? 0) >= q.target);
   711→  const allDone = board.every((q, i) => claimed[i] && (progress[i] ?? 0) >= q.target);
   712→  return { board, progress, claimed, bonusReady, allDone, stale, today };
   713→}

   730→/** Is a welcome-back gift waiting? (2+ days away, one per comeback) */
   731→export function useComeback(): boolean {
   732→  const lastPlayDay = useLeo((s) => s.lastPlayDay);
   733→  const comebackDay = useLeo((s) => s.comebackDay);
   734→  const hydrated = useLeo((s) => s._hydrated);
   735→  if (!hydrated || !lastPlayDay) return false;
   736→  const today = dayKey();
   737→  return daysBetween(lastPlayDay, today) >= 2 && comebackDay !== today;
   738→}
