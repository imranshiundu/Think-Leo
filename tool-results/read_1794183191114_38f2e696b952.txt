    84→    84→export function Play({
    85→    85→  levelIdx,
    86→    86→  onExit,
    87→    87→  onNext,
    88→    88→  hasNext,
    89→    89→}: {
    90→    90→  levelIdx: number;
    91→    91→  onExit: () => void;
    92→    92→  onNext: () => void;
    93→    93→  hasNext: boolean;
    94→    94→}) {
    95→    95→  const world = worldOf(levelIdx);
    96→    96→  const plan = useMemo(() => buildPlan(levelIdx), [levelIdx]);
    97→    97→  const attemptRef = useRef(1);
    98→    98→
    99→    99→  /* snapshot for level-up / league-up detection */
   100→   100→  const xpBeforeRef = useRef(0);
   101→   101→  const leagueBeforeRef = useRef("");
   102→   102→  useEffect(() => {
   103→   103→    const s = useLeo.getState();
   104→   104→    xpBeforeRef.current = s.xp;
   105→   105→    leagueBeforeRef.current = leagueOf(s.weekXp).name;
   106→   106→  }, []);
   107→   107→
   108→   108→  /* ── run state ── */
   109→   109→  const [phase, setPhase] = useState<Phase>("intro");
   110→   110→  const [qi, setQi] = useState(0);
   111→   111→  const [boost, setBoost] = useState(0); // adaptive difficulty offset (-1..+2)
   112→   112→  const [combo, setCombo] = useState(0);
   113→   113→  const [maxCombo, setMaxCombo] = useState(0);
   114→   114→  const [score, setScore] = useState(0);
   115→   115→  const [correctCount, setCorrectCount] = useState(0);
   116→   116→  const [answeredCount, setAnsweredCount] = useState(0);
   117→   117→  const [rightStreak, setRightStreak] = useState(0);
   118→   118→  const [wrongStreak, setWrongStreak] = useState(0);
   119→   119→  const [picked, setPicked] = useState<number | null>(null);
   120→   120→  const [revealOk, setRevealOk] = useState(false);
   121→   121→  const [shakeKey, setShakeKey] = useState(0);
   122→   122→  const [leoMood, setLeoMood] = useState<Mood>("idle");
   123→   123→  const [bubble, setBubble] = useState<string | null>(null);
   124→   124→  const [confettiKey, setConfettiKey] = useState(0);
   125→   125→
   126→   126→  /* rush state */
   127→   127→  const [rushQs, setRushQs] = useState<Question[]>([]);
   128→   128→  const [rushIdx, setRushIdx] = useState(0);
   129→   129→  const [rushCombo, setRushCombo] = useState(0);
   130→   130→  const [rushScore, setRushScore] = useState(0);
   131→   131→  const [rushHits, setRushHits] = useState(0);
   132→   132→  const [rushLeft, setRushLeft] = useState(RUSH_SECONDS);
   133→   133→
   134→   134→  const [results, setResults] = useState<Results | null>(null);
   135→   135→  const recordedRef = useRef(false);
   136→   136→
   137→   137→  const diff = clamp(
   138→   138→    Math.max(1, Math.min(6, 1 + Math.floor((levelIdx - 1) / 8) + boost)),
   139→   139→    1,
   140→   140→    6
   141→   141→  );
   142→   142→  const q = useMemo(
   143→   143→    () =>
   144→   144→      phase === "play" || phase === "reveal"
   145→   145→        ? makeQuestion(plan[qi], diff, levelIdx, qi, attemptRef.current)
   146→   146→        : null,
   147→   147→    [phase, qi, diff, levelIdx, plan]
   148→   148→  );
   149→   149→
   150→   150→  const resetRun = useCallback(
   151→   151→    (fresh: boolean) => {
   152→   152→      if (fresh) attemptRef.current += 1;
   153→   153→      recordedRef.current = false;
   154→   154→      setPhase("intro");
   155→   155→      setQi(0);
   156→   156→      setBoost(0);
   157→   157→      setCombo(0);
   158→   158→      setMaxCombo(0);
   159→   159→      setScore(0);
   160→   160→      setCorrectCount(0);
   161→   161→      setAnsweredCount(0);
   162→   162→      setRightStreak(0);
   163→   163→      setWrongStreak(0);
   164→   164→      setPicked(null);
   165→   165→      setRushQs([]);
   166→   166→      setRushIdx(0);
   167→   167→      setRushCombo(0);
   168→   168→      setRushScore(0);
   169→   169→      setRushHits(0);
   170→   170→      setResults(null);
   171→   171→      setLeoMood("idle");
   172→   172→      setBubble(null);
   173→   173→    },
   174→   174→    []
   175→   175→  );
   176→   176→
   177→   177→  /* keyboard answering (laptop love) */
   178→   178→  const answerRef = useRef<(i: number) => void>(() => {});
   179→   179→  useEffect(() => {
   180→   180→    const h = (e: KeyboardEvent) => {
   181→   181→      const n = Number(e.key);
   182→   182→      if (phase === "play" && n >= 1 && n <= 4) answerRef.current(n - 1);
   183→   183→    };
   184→   184→    window.addEventListener("keydown", h);
   185→   185→    return () => window.removeEventListener("keydown", h);
   186→   186→  }, [phase]);
   187→   187→
   188→   188→  /* music per screen */
   189→   189→  useEffect(() => {
   190→   190→    sfx.startMusic("play");
   191→   191→    return () => {
   192→   192→      sfx.startMusic("home");
   193→   193→    };
   194→   194→  }, []);
   195→   195→
   196→   196→  /* ── answer handling ─────────────────────────────────────────── */
   197→   197→
   198→   198→  const advance = useCallback(
   199→   199→    (ok: boolean) => {
   200→   200→      const heartsNow = useLeo.getState().hearts;
   201→   201→      if (!ok && heartsNow <= 0) {
   202→   202→        setLeoMood("sad");
   203→   203→        setBubble(NO_HEARTS[Math.floor(Math.random() * NO_HEARTS.length)]);
   204→   204→        setPhase("out");
   205→   205→        sfx.sadTrombone();
   206→   206→        return;
   207→   207→      }
   208→   208→      if (qi >= ROUND_COUNT - 1) {
   209→   209→        setLeoMood("excited");
   210→   210→        setBubble("LEO RUSH! Everything ×2 — GO GO GO!");
   211→   211→        setPhase("rushIntro");
   212→   212→        sfx.whoosh();
   213→   213→        sfx.rushGo();
   214→   214→        window.setTimeout(() => {
   215→   215→          setRushQs(Array.from({ length: 30 }, (_, i) => makeRushQuestion(diff, i, attemptRef.current)));
   216→   216→          setRushIdx(0);
   217→   217→          setRushLeft(RUSH_SECONDS);
   218→   218→          setPhase("rush");
   219→   219→        }, 1700);
   220→   220→      } else {
   221→   221→        setQi((n) => n + 1);
   222→   222→        setPicked(null);
   223→   223→        setLeoMood("idle");
   224→   224→        setBubble(null);
   225→   225→        setPhase("play");
   226→   226→      }
   227→   227→    },
   228→   228→    [qi, diff]
   229→   229→  );
   230→   230→
   231→   231→  const handleAnswer = useCallback(
   232→   232→    (idx: number | null) => {
   233→   233→      if (!q || phase !== "play") return;
   234→   234→      const fracLeft = clamp(timeLeftRef.current / q.time, 0, 1);
   235→   235→      const ok = idx !== null && idx === q.answer;
   236→   236→      setPicked(idx);
   237→   237→      setRevealOk(ok);
   238→   238→      setAnsweredCount((a) => a + 1);
   239→   239→      setPhase("reveal");
   240→   240→      const store = useLeo.getState();
   241→   241→
   242→   242→      if (ok) {
   243→   243→        const pts = questionPoints(combo + 1, fracLeft);
   244→   244→        setScore((s) => s + pts);
   245→   245→        setCorrectCount((c) => c + 1);
   246→   246→        const nc = combo + 1;
   247→   247→        setCombo(nc);
   248→   248→        setMaxCombo((m) => Math.max(m, nc));
   249→   249→        setRightStreak((r) => {
   250→   250→          const nr = r + 1;
   251→   251→          if (nr % 3 === 0) setBoost((b) => Math.min(2, b + 1)); // Flow Ranger: heat up
   252→   252→          return nr;
   253→   253→        });
   254→   254→        setWrongStreak(0);
   255→   255→        store.trackAnswer(true);
   256→   256→        store.trackCombo(nc);
   257→   257→        sfx.correct();
   258→   258→        if (nc >= 3) window.setTimeout(() => sfx.combo(nc), 140);
   259→   259→        setLeoMood(nc >= 5 ? "excited" : "happy");
   260→   260→        setBubble(nc >= 5 ? UNSTOPPABLE[nc % UNSTOPPABLE.length] : PRAISE[nc % PRAISE.length]);
   261→   261→      } else {
   262→   262→        store.loseHeart();
   263→   263→        store.trackAnswer(false);
   264→   264→        setCombo(0);
   265→   265→        setWrongStreak((w) => {
   266→   266→          const nw = w + 1;
   267→   267→          if (nw % 2 === 0) setBoost((b) => Math.max(-1, b - 1)); // Flow Ranger: cool down
   268→   268→          return nw;
   269→   269→        });
   270→   270→        setRightStreak(0);
   271→   271→        sfx.wrong();
   272→   272→        window.setTimeout(() => sfx.heartLost(), 240);
   273→   273→        setShakeKey((k) => k + 1);
   274→   274→        setLeoMood("sad");
   275→   275→        setBubble(idx === null ? "Time's up! Here's the trick:" : `${OOPS[Math.floor(Math.random() * OOPS.length)]} ${q.tip}`);
   276→   276→      }
   277→   277→
   278→   278→      window.setTimeout(() => advance(ok), ok ? 900 : 1600);
   279→   279→    },
   280→   280→    [q, phase, combo, advance]
   281→   281→  );
   282→   282→
   283→   283→  answerRef.current = (i: number) => {
   284→   284→    if (!q || phase !== "play") return;
   285→   285→    if (q.kind === "memory") return;
   286→   286→    if (q.options.length && i < q.options.length) handleAnswer(i);
   287→   287→  };
   288→   288→
   289→   289→  /* countdown ring */
   290→   290→  const timeLeftRef = useRef(0);
   291→   291→  const [timeLeft, setTimeLeft] = useState(10);
   292→   292→  useEffect(() => {
   293→   293→    if (phase !== "play" || !q) return;
   294→   294→    const total = q.time;
   295→   295→    setTimeLeft(total);
   296→   296→    timeLeftRef.current = total;
   297→   297→    const start = performance.now();
   298→   298→    let lastTick = Math.ceil(total);
   299→   299→    const id = window.setInterval(() => {
   300→   300→      const left = total - (performance.now() - start) / 1000;
   301→   301→      timeLeftRef.current = Math.max(0, left);
   302→   302→      setTimeLeft(Math.max(0, left));
   303→   303→      const whole = Math.ceil(left);
   304→   304→      if (whole !== lastTick && whole <= 5 && whole > 0) sfx.tick();
   305→   305→      lastTick = whole;
   306→   306→      if (left <= 0) {
   307→   307→        window.clearInterval(id);
   308→   308→        handleAnswer(null);
   309→   309→      }
   310→   310→    }, 100);
   311→   311→    return () => window.clearInterval(id);
   312→   312→  }, [phase, qi, q, handleAnswer]);
   313→   313→
   314→   314→  /* ── LEO RUSH ────────────────────────────────────────────────── */
   315→   315→
   316→   316→  const endRush = useCallback(() => {
   317→   317→    const s = useLeo.getState();
   318→   318→    const total = score + rushScore;
   319→   319→    const accuracy = answeredCount > 0 ? correctCount / answeredCount : 0;
   320→   320→    const stars = starsFor(accuracy);
   321→   321→    const xp = stars > 0 ? xpForRun(total, stars) : 0;
   322→   322→    const isBest = stars > 0 && total > (s.best[levelIdx] ?? 0);
   323→   323→    if (!recordedRef.current) {
   324→   324→      recordedRef.current = true;
   325→   325→      if (rushHits > 0) s.trackRush(rushHits);
   326→   326→      if (stars > 0) {
   327→   327→        s.recordLevel(levelIdx, stars, total);
   328→   328→        s.addXp(xp);
   329→   329→      }
   330→   330→    }
   331→   331→    const chest = stars > 0 && chestAfterLevel(levelIdx);
   332→   332→    if (chest) s.grantChest(world.id);
   333→   333→
   334→   334→    const sAfter = useLeo.getState();
   335→   335→    const lvlUp = playerLevelFromXp(sAfter.xp) > playerLevelFromXp(xpBeforeRef.current)
   336→   336→      ? playerLevelFromXp(sAfter.xp)
   337→   337→      : null;
   338→   338→    const leagueUp = leagueOf(sAfter.weekXp).name !== leagueBeforeRef.current
   339→   339→      ? leagueOf(sAfter.weekXp).name
   340→   340→      : null;
   341→   341→
   342→   342→    setResults({ stars, xp, accuracy, total, isBest, chest, levelUpTo: lvlUp, leagueUpTo: leagueUp, rushHits });
   343→   343→    setPhase("results");
   344→   344→    if (stars > 0) {
   345→   345→      sfx.fanfare();
   346→   346→      if (leagueUp) window.setTimeout(() => sfx.leagueUp(), 700);
   347→   347→      setConfettiKey((k) => k + 1);
   348→   348→      setLeoMood("celebrate");
   349→   349→      setBubble(stars === 3 ? "PERFECT RUN! I'm roaring the roof off!" : "Level crushed! More stars next time?");
   350→   350→      for (let i = 0; i < stars; i++) window.setTimeout(() => sfx.star(i), 600 + i * 380);
   351→   351→    } else {
   352→   352→      sfx.sadTrombone();
   353→   353→      setLeoMood("sad");
   354→   354→      setBubble("So close! One more try — I'll cheer for you!");
   355→   355→    }
   356→   356→  }, [score, rushScore, answeredCount, correctCount, rushHits, levelIdx, world.id]);
   357→   357→
   358→   358→  useEffect(() => {
   359→   359→    if (phase !== "rush") return;
   360→   360→    const start = performance.now();
   361→   361→    const id = window.setInterval(() => {
   362→   362→      const left = RUSH_SECONDS - (performance.now() - start) / 1000;
   363→   363→      setRushLeft(Math.max(0, left));
   364→   364→      if (left <= 0) {
   365→   365→        window.clearInterval(id);
   366→   366→        endRush();
   367→   367→      }
   368→   368→    }, 100);
   369→   369→    return () => window.clearInterval(id);
   370→   370→  }, [phase, endRush]);
   371→   371→
   372→   372→  const rushAnswer = useCallback(
   373→   373→    (idx: number) => {
   374→   374→      const rq = rushQs[rushIdx];
   375→   375→      if (!rq) return;
   376→   376→      const ok = idx === rq.answer;
   377→   377→      if (ok) {
   378→   378→        setRushScore((s) => s + 120);
   379→   379→        setRushHits((h) => h + 1);
   380→   380→        setRushCombo((c) => {
   381→   381→          const nc = c + 1;
   382→   382→          sfx.combo(nc + 2);
   383→   383→          return nc;
   384→   384→        });
   385→   385→      } else {
   386→   386→        setRushCombo(0);
   387→   387→        sfx.tap();
   388→   388→      }
   389→   389→      if (rushIdx < rushQs.length - 1) setRushIdx((i) => i + 1);
   390→   390→    },
   391→   391→    [rushQs, rushIdx]
   392→   392→  );
   393→   393→
   394→   394→  const hearts = useLeo((s) => s.hearts);
   395→   395→  const refills = useLeo((s) => s.refills);
   396→   396→  const doRefill = useLeo((s) => s.useRefill);
   397→   397→
   398→   398→  /* ── render ──────────────────────────────────────────────────── */
   399→   399→
   400→   400→  const qTimeTotal = q?.time ?? 10;
   401→   401→  const danger = phase === "play" && timeLeft <= 3.5;
   402→   402→  const hotCombo = combo >= 5 || (phase === "rush" && rushCombo >= 4);
   403→   403→  const rushQ = phase === "rush" ? rushQs[rushIdx] : null;
   404→   404→
   405→   405→  /* heartbeat under danger */
   406→   406→  useEffect(() => {
   407→   407→    if (!danger) return;
   408→   408→    sfx.pulse();
   409→   409→    const id = window.setInterval(() => sfx.pulse(), 1100);
   410→   410→    return () => window.clearInterval(id);
   411→   411→  }, [danger]);
   412→   412→
   413→   413→  /* music mood follows the phase: rush gets its own driving loop */
   414→   414→  useEffect(() => {
   415→   415→    if (phase === "rush") sfx.startMusic("rush");
   416→   416→    else if (phase === "results" || phase === "out") sfx.startMusic("home");
   417→   417→  }, [phase]);
   418→   418→
   419→   419→  return (
   420→   420→    <div
   421→   421→      className="relative h-screen w-full overflow-hidden"
   422→   422→      style={{ background: `linear-gradient(180deg, ${world.wash[0]}, ${world.wash[1]})` }}
   423→   423→    >
   424→   424→      <QuestWatch />
   425→   425→      {/* heartbeat vignette */}
   426→   426→      {danger && (
   427→   427→        <div
   428→   428→          className="pointer-events-none absolute inset-0 z-10"
   429→   429→          style={{ boxShadow: "inset 0 0 90px rgba(255,60,90,0.45)" }}
   430→   430→        />
   431→   431→      )}
   432→   432→      {hotCombo && <div className="combo-hot pointer-events-none absolute inset-0 z-10" />}
   433→   433→
   434→   434→      <ConfettiBurst fireKey={confettiKey} />
   435→   435→
   436→   436→      {/* HUD */}
   437→   437→      <div className="absolute inset-x-0 top-0 z-20 flex items-center gap-2 px-4 py-3">
   438→   438→        <button
   439→   439→          type="button"
   440→   440→          onClick={onExit}
   441→   441→          className="pill cursor-pointer px-3 py-1.5 font-display text-sm text-leo-ink hover:bg-leo-cream"
   442→   442→        >
   443→   443→          <X width={14} height={14} /> Map
   444→   444→        </button>
   445→   445→        <div className="pill gap-1.5 px-3 py-1 font-display text-sm text-leo-ink">
   446→   446→          <Sticker id={world.emblem} size={18} />
   447→   447→          Lv {levelIdx}
   448→   448→        </div>

   715→   715→/** Pops "Quest complete!" the moment a daily quest fills up mid-run. */
   716→   716→function QuestWatch() {
   717→   717→  const { board, progress, claimed } = useQuests();
   718→   718→  const seen = useRef<Record<string, boolean>>({});
   719→   719→  const [toast, setToast] = useState<{ label: string; key: number } | null>(null);
   720→   720→
   721→   721→  useEffect(() => {
   722→   722→    board.forEach((q, i) => {
   723→   723→      const done = (progress[i] ?? 0) >= q.target && !claimed[i];
   724→   724→      if (done && !seen.current[q.id]) {
   725→   725→        seen.current[q.id] = true;
   726→   726→        setToast({ label: q.label, key: Date.now() });
   727→   727→        sfx.questDone();
   728→   728→        window.setTimeout(() => setToast(null), 2800);
   729→   729→      }
   730→   730→      if (!done) seen.current[q.id] = false;
   731→   731→    });
   732→   732→  }, [board, progress, claimed]);
   733→   733→
   734→   734→  return (
   735→   735→    <AnimatePresence>
   736→   736→      {toast && (
   737→   737→        <motion.div
   738→   738→          key={toast.key}
   739→   739→          initial={{ y: -60, opacity: 0, scale: 0.9 }}
   740→   740→          animate={{ y: 0, opacity: 1, scale: 1 }}
   741→   741→          exit={{ y: -40, opacity: 0 }}
   742→   742→          transition={{ type: "spring", stiffness: 300, damping: 20 }}
   743→   743→          className="absolute left-1/2 top-16 z-[30] -translate-x-1/2"
   744→   744→        >
   745→   745→          <div className="flex items-center gap-2 rounded-2xl border-[3px] border-leo-sun bg-white px-4 py-2 shadow-[0_6px_0_rgba(214,178,110,0.4)]">
   746→   746→            <PartyPopper width={16} height={16} className="text-leo-sun-deep" />

   974→   974→/* ── memory mini-game (Simon-style) ────────────────────────────── */
   975→   975→
   976→   976→function MemoryPads({
   977→   977→  pads,
   978→   978→  onResult,
   979→   979→  disabled,
   980→   980→}: {
   981→   981→  pads: number[];
   982→   982→  onResult: (ok: boolean) => void;
   983→   983→  disabled: boolean;
   984→   984→}) {
   985→   985→  const [showIdx, setShowIdx] = useState(-1); // which pad is lit during playback
   986→   986→  const [watching, setWatching] = useState(true);
   987→   987→  const [inputStep, setInputStep] = useState(0);
   988→   988→  const [flash, setFlash] = useState<"ok" | "bad" | null>(null);
   989→   989→  const done = useRef(false);
   990→   990→
   991→   991→  useEffect(() => {
   992→   992→    if (!watching) return;
   993→   993→    const timers: number[] = [];
   994→   994→    pads.forEach((p, i) => {
   995→   995→      timers.push(window.setTimeout(() => { setShowIdx(p); sfx.pad(p); }, 500 + i * 700));
   996→   996→      timers.push(window.setTimeout(() => setShowIdx(-1), 500 + i * 700 + 430));
   997→   997→    });
   998→   998→    timers.push(window.setTimeout(() => setWatching(false), 500 + pads.length * 700 + 250));
   999→   999→    return () => timers.forEach((t) => window.clearTimeout(t));
  1000→  1000→  }, [pads, watching]);
  1001→  1001→
  1002→  1002→  const press = (i: number) => {
  1003→  1003→    if (watching || disabled || done.current || flash) return;
  1004→  1004→    sfx.pad(i);
  1005→  1005→    if (pads[inputStep] === i) {
  1006→  1006→      if (inputStep + 1 === pads.length) {
  1007→  1007→        done.current = true;
  1008→  1008→        setFlash("ok");
  1009→  1009→        sfx.correct();
  1010→  1010→        window.setTimeout(() => onResult(true), 550);
  1011→  1011→      } else {
  1012→  1012→        setInputStep((s) => s + 1);
  1013→  1013→      }
  1014→  1014→    } else {
  1015→  1015→      done.current = true;
  1016→  1016→      setFlash("bad");
  1017→  1017→      sfx.wrong();
  1018→  1018→      window.setTimeout(() => onResult(false), 650);
  1019→  1019→    }
  1020→  1020→  };
  1021→  1021→
  1022→  1022→  return (
  1023→  1023→    <div className="flex flex-col items-center gap-3 py-1">
  1024→  1024→      <span className={`flex items-center gap-1.5 font-display text-sm font-bold ${watching ? "text-leo-grape" : "text-leo-leaf-deep"}`}>
  1025→  1025→        {watching ? (
  1026→  1026→          <>
  1027→  1027→            <Eye width={15} height={15} /> WATCH…
  1028→  1028→          </>
  1029→  1029→        ) : flash === "ok" ? (
  1030→  1030→          <>
  1031→  1031→            <PartyPopper width={15} height={15} /> PERFECT!
  1032→  1032→          </>
  1033→  1033→        ) : flash === "bad" ? (
  1034→  1034→          <>
  1035→  1035→            <EyeOff width={15} height={15} /> Oops…
  1036→  1036→          </>
  1037→  1037→        ) : (
  1038→  1038→          <>
  1039→  1039→            <HandIcon size={15} /> YOUR TURN — repeat it!
  1040→  1040→          </>
  1041→  1041→        )}
  1042→  1042→      </span>
  1043→  1043→      <div className="grid grid-cols-2 gap-3">
  1044→  1044→        {PAD_COLORS.map((color, i) => {
  1045→  1045→          const lit = showIdx === i;
  1046→  1046→          return (
  1047→  1047→            <motion.button
  1048→  1048→              key={i}
  1049→  1049→              type="button"
  1050→  1050→              aria-label={`Memory pad ${i + 1}`}
  1051→  1051→              onClick={() => press(i)}
  1052→  1052→              animate={{
  1053→  1053→                scale: lit ? 1.09 : 1,
  1054→  1054→                filter: lit ? "brightness(1.35)" : "brightness(1)",
  1055→  1055→              }}
  1056→  1056→              transition={{ duration: 0.12 }}
  1057→  1057→              className="h-24 w-24 cursor-pointer rounded-2xl border-[3.5px] border-white/90 shadow-[0_6px_0_rgba(0,0,0,0.16)]"
  1058→  1058→              style={{ background: color, opacity: flash === "bad" && pads[inputStep] !== i ? 0.9 : 1 }}
  1059→  1059→            />
  1060→  1060→          );
  1061→  1061→        })}
  1062→  1062→      </div>
  1063→  1063→    </div>
  1064→  1064→  );
  1065→  1065→}
