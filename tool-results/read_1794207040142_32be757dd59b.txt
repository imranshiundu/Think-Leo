     3→/**
     4→ * Think Leo — Play screen: the adrenaline engine
     5→ * ──────────────────────────────────────────────
     6→ * Phase machine:
     7→ *   intro → play ⇄ reveal → (last round) rushIntro → rush → results
     8→ *                          └─ hearts hit 0 → out
     9→ * Juice: combo meter with escalating sounds, countdown ring + heartbeat
    10→ * pulse, danger vignette, screen shake, Leo reactions, score count-up,
    11→ * level-up & league-up banners and the ×2 LEO RUSH finale
    12→ * (Candy-Crush "Sugar Crush" moment). Math in docs/ALGORITHM.md §4–6.
    13→ */
    14→

    16→import { Leo } from "@/components/leo/Leo";
    17→import { ChestArt } from "@/components/leo/badges";
    18→import { Sticker, StickerTile, isStickerId, type StickerId } from "@/components/leo/stickers";
    19→import {
    20→  ChunkyButton,

    28→  TimerIcon,
    29→} from "@/components/game/bits";
    30→import {
    31→  ROUND_COUNT,
    32→  RUSH_SECONDS,
    33→  chestAfterLevel,
    34→  clamp,
    35→  leagueOf,
    36→  playerLevelFromXp,
    37→  questionPoints,

    39→  worldOf,

    41→  type Mood,
    42→} from "@/lib/game";
    43→import { buildPlan, makeQuestion, makeRushQuestion, WORLD_TILES, type Question } from "@/lib/questions";
    44→import { sfx } from "@/lib/sound";

    48→import { useCallback, useEffect, useMemo, useRef, useState } from "react";
    49→
    50→type Phase = "intro" | "play" | "reveal" | "rushIntro" | "rush" | "out" | "results";
    51→
    52→interface Results {
    53→  stars: number;
    54→  xp: number;
    55→  accuracy: number;
    56→  total: number;
    57→  isBest: boolean;
    58→  chest: boolean;
    59→  levelUpTo: number | null;
    60→  leagueUpTo: string | null;
    61→  rushHits: number;
    62→}
    63→
    64→const PRAISE = ["Roarrsome!", "Big brain move!", "Yes! Keep going!", "Pawsome!"];
    65→const UNSTOPPABLE = ["UNSTOPPABLE!", "MEGA COMBO!", "ON FIRE!", "LEO-LEVEL THINKING!"];
    66→const OOPS = ["Almost!", "So close!", "No worries —"];
    67→const NO_HEARTS = [
    68→  "Hearts refill over time — brains need naps too!",
    69→  "Out of hearts? Stretch, sip water, come back roaring!",
    70→];
    71→
    72→const PAD_COLORS = ["#58cc02", "#ff5a8c", "#9b5de5", "#ffb020"];
    73→
    74→const KIND_LABEL: Record<string, string> = {
    75→  count: "Count",
    76→  addsub: "Maths",
    77→  compare: "Compare",
    78→  oddOne: "Odd one",
    79→  sequence: "Sequence",
    80→  multiply: "Tables",
    81→  truefalse: "True/trap",
    82→  memory: "Memory",
    83→};
    84→
    85→export function Play({
    86→  levelIdx,
    87→  onExit,
    88→  onNext,
    89→  hasNext,
    90→}: {
    91→  levelIdx: number;
    92→  onExit: () => void;
    93→  onNext: () => void;
    94→  hasNext: boolean;
    95→}) {
    96→  const world = worldOf(levelIdx);
    97→  const plan = useMemo(() => buildPlan(levelIdx), [levelIdx]);
    98→  const attemptRef = useRef(1);
    99→
   100→  /* snapshot for level-up / league-up detection */
   101→  const xpBeforeRef = useRef(0);
   102→  const leagueBeforeRef = useRef("");
   103→  useEffect(() => {
   104→    const s = useLeo.getState();
   105→    xpBeforeRef.current = s.xp;
   106→    leagueBeforeRef.current = leagueOf(s.weekXp).name;
   107→  }, []);
   108→
   109→  /* ── run state ── */
   110→  const [phase, setPhase] = useState<Phase>("intro");
   111→  const [qi, setQi] = useState(0);
   112→  const [boost, setBoost] = useState(0); // adaptive difficulty offset (-1..+2)
   113→  const [combo, setCombo] = useState(0);
   114→  const [maxCombo, setMaxCombo] = useState(0);
   115→  const [score, setScore] = useState(0);
   116→  const [correctCount, setCorrectCount] = useState(0);
   117→  const [answeredCount, setAnsweredCount] = useState(0);
   118→  const [rightStreak, setRightStreak] = useState(0);
   119→  const [wrongStreak, setWrongStreak] = useState(0);
   120→  const [picked, setPicked] = useState<number | null>(null);
   121→  const [revealOk, setRevealOk] = useState(false);
   122→  const [shakeKey, setShakeKey] = useState(0);
   123→  const [leoMood, setLeoMood] = useState<Mood>("idle");
   124→  const [bubble, setBubble] = useState<string | null>(null);
   125→  const [confettiKey, setConfettiKey] = useState(0);
   126→  /* floating "+120" score pops and combo banners (the candy layer) */
   127→  const [pops, setPops] = useState<{ id: number; pts: number }[]>([]);
   128→  const [banner, setBanner] = useState<{ id: number; text: string } | null>(null);
   129→
   130→  /* rush state */
   131→  const [rushQs, setRushQs] = useState<Question[]>([]);
   132→  const [rushIdx, setRushIdx] = useState(0);
   133→  const [rushCombo, setRushCombo] = useState(0);
   134→  const [rushScore, setRushScore] = useState(0);
   135→  const [rushHits, setRushHits] = useState(0);
   136→  const [rushLeft, setRushLeft] = useState(RUSH_SECONDS);
   137→
   138→  const [results, setResults] = useState<Results | null>(null);
   139→  const recordedRef = useRef(false);
   140→
   141→  const diff = clamp(
   142→    Math.max(1, Math.min(6, 1 + Math.floor((levelIdx - 1) / 8) + boost)),
   143→    1,
   144→    6
   145→  );
   146→  const q = useMemo(
   147→    () =>
   148→      phase === "play" || phase === "reveal"
   149→        ? makeQuestion(plan[qi], diff, levelIdx, qi, attemptRef.current)
   150→        : null,
   151→    [phase, qi, diff, levelIdx, plan]
   152→  );
   153→
   154→  const resetRun = useCallback(
   155→    (fresh: boolean) => {
   156→      if (fresh) attemptRef.current += 1;
   157→      recordedRef.current = false;
   158→      setPhase("intro");
   159→      setQi(0);
   160→      setBoost(0);
   161→      setCombo(0);
   162→      setMaxCombo(0);
   163→      setScore(0);
   164→      setCorrectCount(0);
   165→      setAnsweredCount(0);
   166→      setRightStreak(0);
   167→      setWrongStreak(0);
   168→      setPicked(null);
   169→      setRushQs([]);
   170→      setRushIdx(0);
   171→      setRushCombo(0);
   172→      setRushScore(0);
   173→      setRushHits(0);
   174→      setResults(null);
   175→      setLeoMood("idle");
   176→      setBubble(null);
   177→    },
   178→    []
   179→  );
   180→
   181→  /* keyboard answering (laptop love) */
   182→  const answerRef = useRef<(i: number) => void>(() => {});
   183→  useEffect(() => {
   184→    const h = (e: KeyboardEvent) => {
   185→      const n = Number(e.key);
   186→      if (phase === "play" && n >= 1 && n <= 4) answerRef.current(n - 1);
   187→    };
   188→    window.addEventListener("keydown", h);
   189→    return () => window.removeEventListener("keydown", h);
   190→  }, [phase]);
   191→
   192→  /* music per screen */
   193→  useEffect(() => {
   194→    sfx.startMusic("play");
   195→    return () => {
   196→      sfx.startMusic("home");
   197→    };
   198→  }, []);
   199→
   200→  /* ── answer handling ─────────────────────────────────────────── */
   201→
   202→  const advance = useCallback(
   203→    (ok: boolean) => {
   204→      const heartsNow = useLeo.getState().hearts;
   205→      if (!ok && heartsNow <= 0) {
   206→        setLeoMood("sad");
   207→        setBubble(NO_HEARTS[Math.floor(Math.random() * NO_HEARTS.length)]);
   208→        setPhase("out");
   209→        sfx.sadTrombone();
   210→        return;
   211→      }
   212→      if (qi >= ROUND_COUNT - 1) {
   213→        setLeoMood("excited");
   214→        setBubble("LEO RUSH! Everything ×2 — GO GO GO!");
   215→        setPhase("rushIntro");
   216→        sfx.whoosh();
   217→        sfx.rushGo();
   218→        window.setTimeout(() => {
   219→          // 200 rapid-fire questions: even the fastest kid cannot outrun the clock
   220→          setRushQs(Array.from({ length: 200 }, (_, i) => makeRushQuestion(diff, i, attemptRef.current)));
   221→          setRushIdx(0);
   222→          setRushLeft(RUSH_SECONDS);
   223→          setPhase("rush");
   224→        }, 1700);
   225→      } else {
   226→        setQi((n) => n + 1);
   227→        setPicked(null);
   228→        setLeoMood("idle");
   229→        setBubble(null);
   230→        setPhase("play");
   231→      }
   232→    },
   233→    [qi, diff]
   234→  );
   235→
   236→  const handleAnswer = useCallback(
   237→    (idx: number | null) => {
   238→      if (!q || phase !== "play") return;
   239→      const fracLeft = clamp(timeLeftRef.current / q.time, 0, 1);
   240→      const ok = idx !== null && idx === q.answer;
   241→      setPicked(idx);
   242→      setRevealOk(ok);
   243→      setAnsweredCount((a) => a + 1);
   244→      setPhase("reveal");
   245→      const store = useLeo.getState();
   246→
   247→      if (ok) {
   248→        const pts = questionPoints(combo + 1, fracLeft);
   249→        setScore((s) => s + pts);
   250→        setCorrectCount((c) => c + 1);
   251→        const nc = combo + 1;
   252→        setCombo(nc);
   253→        setMaxCombo((m) => Math.max(m, nc));
   254→        setRightStreak((r) => {
   255→          const nr = r + 1;
   256→          if (nr % 3 === 0) setBoost((b) => Math.min(2, b + 1)); // Flow Ranger: heat up
   257→          return nr;
   258→        });
   259→        setWrongStreak(0);
   260→        store.trackAnswer(true);
   261→        store.trackCombo(nc);
   262→        sfx.correct();
   263→        if (nc >= 3) window.setTimeout(() => sfx.combo(nc), 140);
   264→        setLeoMood(nc >= 5 ? "excited" : "happy");
   265→        setBubble(nc >= 5 ? UNSTOPPABLE[nc % UNSTOPPABLE.length] : PRAISE[nc % PRAISE.length]);
   266→        /* juice: floating "+points" and milestone banners */
   267→        const popId = Date.now() + Math.random();
   268→        setPops((p) => [...p, { id: popId, pts }]);
   269→        window.setTimeout(() => setPops((p) => p.filter((x) => x.id !== popId)), 950);
   270→        if (nc === 5 || nc === 8 || nc === 10) {
   271→          const bid = Date.now() + Math.random();
   272→          setBanner({ id: bid, text: UNSTOPPABLE[nc % UNSTOPPABLE.length] });
   273→          window.setTimeout(() => setBanner((b) => (b && b.id === bid ? null : b)), 1200);
   274→        }
   275→      } else {
   276→        store.loseHeart();
   277→        store.trackAnswer(false);
   278→        setCombo(0);
   279→        setWrongStreak((w) => {
   280→          const nw = w + 1;
   281→          if (nw % 2 === 0) setBoost((b) => Math.max(-1, b - 1)); // Flow Ranger: cool down
   282→          return nw;
   283→        });
   284→        setRightStreak(0);
   285→        sfx.wrong();
   286→        window.setTimeout(() => sfx.heartLost(), 240);
   287→        setShakeKey((k) => k + 1);
   288→        setLeoMood("sad");
   289→        setBubble(idx === null ? "Time's up! Here's the trick:" : `${OOPS[Math.floor(Math.random() * OOPS.length)]} ${q.tip}`);
   290→      }
   291→
   292→      window.setTimeout(() => advance(ok), ok ? 900 : 1600);
   293→    },
   294→    [q, phase, combo, advance]
   295→  );
   296→
   297→  answerRef.current = (i: number) => {
   298→    if (!q || phase !== "play") return;
   299→    if (q.kind === "memory") return;
   300→    if (q.options.length && i < q.options.length) handleAnswer(i);
   301→  };
   302→
   303→  /* countdown ring */
   304→  const timeLeftRef = useRef(0);
   305→  const [timeLeft, setTimeLeft] = useState(10);
   306→  useEffect(() => {
   307→    if (phase !== "play" || !q) return;
   308→    const total = q.time;
   309→    setTimeLeft(total);
   310→    timeLeftRef.current = total;
   311→    const start = performance.now();
   312→    let lastTick = Math.ceil(total);
   313→    const id = window.setInterval(() => {
   314→      const left = total - (performance.now() - start) / 1000;
   315→      timeLeftRef.current = Math.max(0, left);
   316→      setTimeLeft(Math.max(0, left));
   317→      const whole = Math.ceil(left);
   318→      if (whole !== lastTick && whole <= 5 && whole > 0) sfx.tick();
   319→      lastTick = whole;
   320→      if (left <= 0) {
   321→        window.clearInterval(id);
   322→        handleAnswer(null);
   323→      }
   324→    }, 100);
   325→    return () => window.clearInterval(id);
   326→  }, [phase, qi, q, handleAnswer]);
   327→
   328→  /* ── LEO RUSH ────────────────────────────────────────────────── */
   329→
   330→  const endRush = useCallback(() => {
   331→    const s = useLeo.getState();
   332→    const total = score + rushScore;
   333→    const accuracy = answeredCount > 0 ? correctCount / answeredCount : 0;
   334→    const stars = starsFor(accuracy);
   335→    const xp = stars > 0 ? xpForRun(total, stars) : 0;
   336→    const isBest = stars > 0 && total > (s.best[levelIdx] ?? 0);
   337→    // ANY finished run counts today as played — streaks record on effort,
   338→    // not only on wins (a kid who tried and missed must keep their flame).
   339→    s.noteActivity();
   340→    if (!recordedRef.current) {
   341→      recordedRef.current = true;
   342→      if (rushHits > 0) s.trackRush(rushHits);
   343→      if (stars > 0) {
   344→        s.recordLevel(levelIdx, stars, total);
   345→        s.addXp(xp);
   346→      }
   347→    }
   348→    const chest = stars > 0 && chestAfterLevel(levelIdx);
   349→    if (chest) s.grantChest(world.id);
   350→
   351→    const sAfter = useLeo.getState();
   352→    const lvlUp = playerLevelFromXp(sAfter.xp) > playerLevelFromXp(xpBeforeRef.current)
   353→      ? playerLevelFromXp(sAfter.xp)
   354→      : null;
   355→    const leagueUp = leagueOf(sAfter.weekXp).name !== leagueBeforeRef.current
   356→      ? leagueOf(sAfter.weekXp).name
   357→      : null;
   358→
   359→    setResults({ stars, xp, accuracy, total, isBest, chest, levelUpTo: lvlUp, leagueUpTo: leagueUp, rushHits });
   360→    setPhase("results");
   361→    if (stars > 0) {
   362→      sfx.fanfare();
   363→      if (leagueUp) window.setTimeout(() => sfx.leagueUp(), 700);
   364→      setConfettiKey((k) => k + 1);
   365→      setLeoMood("celebrate");
   366→      setBubble(stars === 3 ? "PERFECT RUN! I'm roaring the roof off!" : "Level crushed! More stars next time?");
   367→      for (let i = 0; i < stars; i++) window.setTimeout(() => sfx.star(i), 600 + i * 380);
   368→    } else {
   369→      sfx.sadTrombone();
   370→      setLeoMood("sad");
   371→      setBubble("So close! One more try — I'll cheer for you!");
   372→    }
   373→  }, [score, rushScore, answeredCount, correctCount, rushHits, levelIdx, world.id]);
   374→
   375→  /* ── THE RUSH CLOCK ────────────────────────────────────────────
   376→     endRush is rebuilt on every rush hit (its deps include rushScore
   377→     and rushHits), so this effect MUST NOT depend on it — otherwise
   378→     every correct tap would restart the 12 seconds and the level
   379→     would never, ever end. We call the latest endRush through a ref
   380→     and key the interval on the phase alone. */
   381→  const endRushRef = useRef(endRush);
   382→  endRushRef.current = endRush;
   383→  useEffect(() => {
   384→    if (phase !== "rush") return;
   385→    const start = performance.now();
   386→    const id = window.setInterval(() => {
   387→      const left = RUSH_SECONDS - (performance.now() - start) / 1000;
   388→      setRushLeft(Math.max(0, left));
   389→      if (left <= 0) {
   390→        window.clearInterval(id);
   391→        endRushRef.current();
   392→      }
   393→    }, 100);
   394→    return () => window.clearInterval(id);
   395→  }, [phase]);
   396→
   397→  const rushAnswer = useCallback(
   398→    (idx: number | null) => {
   399→      if (idx === null) return; // no timeout penalty inside the rush
   400→      const rq = rushQs[rushIdx];
   401→      if (!rq) return;
   402→      const ok = idx === rq.answer;
   403→      if (ok) {
   404→        setRushScore((s) => s + 120);
   405→        setRushHits((h) => h + 1);
   406→        setRushCombo((c) => {
   407→          const nc = c + 1;
   408→          sfx.combo(nc + 2);
   409→          return nc;
   410→        });
   411→      } else {
   412→        setRushCombo(0);
   413→        sfx.tap();
   414→      }
   415→      if (rushIdx < rushQs.length - 1) setRushIdx((i) => i + 1);
   416→    },
   417→    [rushQs, rushIdx]
   418→  );
   419→
   420→  const hearts = useLeo((s) => s.hearts);
   421→  const refills = useLeo((s) => s.refills);
   422→  const doRefill = useLeo((s) => s.useRefill);
   423→
   424→  /* ── render ──────────────────────────────────────────────────── */
   425→
   426→  const qTimeTotal = q?.time ?? 10;
   427→  const danger = phase === "play" && timeLeft <= 3.5;
   428→  const hotCombo = combo >= 5 || (phase === "rush" && rushCombo >= 4);
   429→  const rushQ = phase === "rush" ? rushQs[rushIdx] : null;
   430→  const parade = WORLD_TILES[world.id] ?? WORLD_TILES[0];
   431→
   432→  /* heartbeat under danger */
   433→  useEffect(() => {
   434→    if (!danger) return;
   435→    sfx.pulse();
   436→    const id = window.setInterval(() => sfx.pulse(), 1100);
   437→    return () => window.clearInterval(id);
   438→  }, [danger]);
   439→
   440→  /* music mood follows the phase: rush gets its own driving loop */
   441→  useEffect(() => {
   442→    if (phase === "rush") sfx.startMusic("rush");
   443→    else if (phase === "results" || phase === "out") sfx.startMusic("home");
   444→  }, [phase]);
   445→
   446→  return (
   447→    <div
   448→      className="relative h-screen w-full overflow-hidden"
   449→      style={{ background: `linear-gradient(180deg, ${world.wash[0]}, ${world.wash[1]})` }}
   450→    >
   451→      <QuestWatch />
   452→      {/* living background: world-flavoured cartoons drifting behind the action */}
   453→      <AmbientCartoons world={world.id} count={8} opacity={0.15} />
   454→      {/* heartbeat vignette */}
   455→      {danger && (
   456→        <div
   457→          className="pointer-events-none absolute inset-0 z-10"
   458→          style={{ boxShadow: "inset 0 0 90px rgba(255,60,90,0.45)" }}
   459→        />
   460→      )}
   461→      {hotCombo && <div className="combo-hot pointer-events-none absolute inset-0 z-10" />}
   462→      {/* LEO RUSH: racing light-streaks + pulsing aura */}
   463→      {phase === "rush" && (
   464→        <>
   465→          <div className="rush-lines pointer-events-none absolute inset-0 z-10" />
   466→          <div className="rush-aura pointer-events-none absolute inset-0 z-10" />
   467→        </>
   468→      )}
   469→
   470→      <ConfettiBurst fireKey={confettiKey} />
   471→
   472→      {/* floating +points */}
   473→      <div className="pointer-events-none absolute inset-x-0 top-16 z-30 flex justify-center">
   474→        <AnimatePresence>
   475→          {pops.map((p) => (
   476→            <motion.span
   477→              key={p.id}
   478→              initial={{ y: 18, opacity: 0, scale: 0.5 }}
   479→              animate={{ y: -44, opacity: [0, 1, 1, 0], scale: [0.5, 1.25, 1, 0.9] }}
   480→              exit={{ opacity: 0 }}
   481→              transition={{ duration: 0.9, ease: "easeOut" }}
   482→              className="font-display text-3xl font-bold text-leo-leaf-deep drop-shadow-[0_2px_0_rgba(255,255,255,0.9)]"
   483→            >
   484→              +{p.pts}
   485→            </motion.span>
   486→          ))}
   487→        </AnimatePresence>
   488→      </div>
   489→
   490→      {/* combo milestone banner sweeping across the screen */}
   491→      <AnimatePresence>
   492→        {banner && (
   493→          <motion.div
   494→            key={banner.id}
   495→            initial={{ x: "-120%", rotate: -3 }}
   496→            animate={{ x: "0%", rotate: -3 }}
   497→            exit={{ x: "130%", rotate: -3 }}
   498→            transition={{ type: "spring", stiffness: 210, damping: 20 }}
   499→            className="pointer-events-none absolute left-0 top-1/3 z-40"
   500→          >
   501→            <div className="rounded-r-2xl border-[3px] border-white bg-gradient-to-r from-leo-sun via-leo-berry to-leo-grape px-6 py-2 shadow-[0_6px_0_rgba(0,0,0,0.15)]">
   502→              <span className="font-display text-2xl font-bold text-white text-outline-sm">{banner.text}</span>
   503→            </div>
   504→          </motion.div>
   505→        )}
   506→      </AnimatePresence>
   507→
   508→      {/* HUD */}
   509→      <div className="absolute inset-x-0 top-0 z-20 flex items-center gap-2 px-4 py-3">
   510→        <button
   511→          type="button"
   512→          onClick={onExit}
   513→          className="pill cursor-pointer px-3 py-1.5 font-display text-sm text-leo-ink hover:bg-leo-cream"
   514→        >
   515→          <X width={14} height={14} /> Map

   829→/** Pops "Quest complete!" the moment a daily quest fills up mid-run. */
   830→function QuestWatch() {
   831→  const { board, progress, claimed } = useQuests();
   832→  const seen = useRef<Record<string, boolean>>({});
   833→  const [toast, setToast] = useState<{ label: string; key: number } | null>(null);
   834→
   835→  useEffect(() => {
   836→    board.forEach((q, i) => {
   837→      const done = (progress[i] ?? 0) >= q.target && !claimed[i];
   838→      if (done && !seen.current[q.id]) {
   839→        seen.current[q.id] = true;
   840→        setToast({ label: q.label, key: Date.now() });
   841→        sfx.questDone();
   842→        window.setTimeout(() => setToast(null), 2800);
   843→      }
   844→      if (!done) seen.current[q.id] = false;
   845→    });
   846→  }, [board, progress, claimed]);
   847→
   848→  return (
   849→    <AnimatePresence>
   850→      {toast && (
   851→        <motion.div
   852→          key={toast.key}
   853→          initial={{ y: -60, opacity: 0, scale: 0.9 }}
   854→          animate={{ y: 0, opacity: 1, scale: 1 }}
   855→          exit={{ y: -40, opacity: 0 }}
   856→          transition={{ type: "spring", stiffness: 300, damping: 20 }}
   857→          className="absolute left-1/2 top-16 z-[30] -translate-x-1/2"
   858→        >
   859→          <div className="flex items-center gap-2 rounded-2xl border-[3px] border-leo-sun bg-white px-4 py-2 shadow-[0_6px_0_rgba(214,178,110,0.4)]">
   860→            <PartyPopper width={16} height={16} className="text-leo-sun-deep" />
   861→            <span className="font-display text-sm font-bold text-leo-ink">Quest complete!</span>
   862→            <span className="text-xs text-leo-ink-soft">{toast.label}</span>
   863→          </div>
   864→        </motion.div>
   865→      )}
   866→    </AnimatePresence>
   867→  );
   868→}

   879→function TimerRing({ left, total }: { left: number; total: number }) {
   880→  const r = 16;
   881→  const c = 2 * Math.PI * r;
   882→  const frac = clamp(left / total, 0, 1);
   883→  const low = left <= 5;
   884→  return (
   885→    <div className={`relative h-11 w-11 ${low ? "anim-heartbeat" : ""}`}>
   886→      <svg width="44" height="44" viewBox="0 0 44 44" className="-rotate-90">
   887→        <circle cx="22" cy="22" r={r} fill="#fff" stroke="#F0E0BD" strokeWidth="5" />
   888→        <circle
   889→          cx="22"
   890→          cy="22"
   891→          r={r}
   892→          fill="none"
   893→          stroke={low ? "#ff5a8c" : "#58cc02"}
   894→          strokeWidth="5"
   895→          strokeLinecap="round"
   896→          strokeDasharray={c}
   897→          strokeDashoffset={c * (1 - frac)}
   898→        />
   899→      </svg>
   900→      <span className={`absolute inset-0 flex items-center justify-center font-display text-sm font-bold ${low ? "text-leo-berry-deep" : "text-leo-ink"}`}>
   901→        {Math.ceil(left)}
   902→      </span>
   903→    </div>
   904→  );

  1002→        <div className="mx-auto flex max-w-md flex-wrap justify-center gap-1.5 py-2">
  1003→          {q.tiles.map((t, i) => (
  1004→            <StickerTile key={i} id={t} size={54} delay={i * 0.03} />
  1005→          ))}
  1006→        </div>
  1007→      )}
  1008→
  1009→      {q.kind === "sequence" && q.seq && (
  1010→        <div className="flex flex-wrap justify-center gap-2 py-2">
  1011→          {q.seq.map((s, i) => (
  1012→            <span
  1013→              key={i}
  1014→              className={`flex h-14 min-w-14 items-center justify-center rounded-xl border-[3px] px-2 font-display text-2xl font-bold ${
  1015→                s === "?" ? "border-leo-sun bg-leo-cream text-leo-sun-deep" : "border-leo-line bg-white text-leo-ink"
  1016→              }`}
  1017→            >
  1018→              {s}
  1019→            </span>
  1020→          ))}
  1021→        </div>
  1022→      )}
  1023→
  1024→      {q.kind === "memory" && q.pads && (
  1025→        <MemoryPads pads={q.pads} onResult={(ok) => onAnswer(ok ? q.answer : (q.answer + 1) % 4)} disabled={reveal} />
  1026→      )}
  1027→
  1028→      {/* options */}
  1029→      {q.kind !== "memory" && q.options.length > 0 && (
  1030→        <div
  1031→          className={`mt-3 grid gap-2.5 ${
  1032→            q.options.length <= 2
  1033→              ? "grid-cols-2"
  1034→              : q.kind === "oddOne"
  1035→                ? "grid-cols-4"
  1036→                : q.options.every((o) => o.length <= 3)
  1037→                  ? "grid-cols-4"
  1038→                  : "grid-cols-2"
  1039→          }`}
  1040→        >
  1041→          {q.options.map((opt, i) => {
  1042→            const isCorrect = reveal && i === q.answer;
  1043→            const isWrongPick = reveal && picked === i && i !== q.answer;
  1044→            const plate = q.groupOptions?.[i];
  1045→            const isArt = q.kind === "oddOne" && isStickerId(opt);
  1046→            return (
  1047→              <motion.button
  1048→                key={i}
  1049→                type="button"
  1050→                disabled={reveal}
  1051→                onClick={() => onAnswer(i)}
  1052→                data-state={isCorrect ? "correct" : isWrongPick ? "wrong" : undefined}
  1053→                whileHover={!reveal ? { y: -3 } : undefined}
  1054→                whileTap={!reveal ? { scale: 0.96 } : undefined}
  1055→                className={`answer-bubble flex min-h-[64px] cursor-pointer items-center justify-center px-3 py-3 ${
  1056→                  isArt
  1057→                    ? ""
  1058→                    : q.kind === "truefalse"
  1059→                      ? "font-display text-2xl font-bold"
  1060→                      : "font-display text-2xl font-bold text-leo-ink"
  1061→                } ${q.kind === "truefalse" ? (i === 0 ? "bg-[#e9ffcf]" : "bg-[#ffe4ec]") : ""}`}
  1062→              >
  1063→                {plate ? (
  1064→                  <span className="flex flex-col items-center gap-1">
  1065→                    <span className="flex max-w-[200px] flex-wrap justify-center gap-0.5">
  1066→                      {plate.map((t, j) => (
  1067→                        <Sticker key={j} id={t} size={24} />
  1068→                      ))}
  1069→                    </span>
  1070→                    <span className="font-display text-lg font-bold text-leo-ink">{opt}</span>
  1071→                  </span>
  1072→                ) : isArt ? (
  1073→                  <Sticker id={opt as StickerId} size={52} />
  1074→                ) : (
  1075→                  opt
  1076→                )}
  1077→              </motion.button>
  1078→            );
  1079→          })}
  1080→        </div>
  1081→      )}
  1082→
  1083→      {rushMode && null}
  1084→    </div>
  1085→  );
  1086→}

  1088→/* ── memory mini-game (Simon-style) ────────────────────────────── */
  1089→
  1090→function MemoryPads({
  1091→  pads,
  1092→  onResult,
  1093→  disabled,
  1094→}: {
  1095→  pads: number[];
  1096→  onResult: (ok: boolean) => void;
  1097→  disabled: boolean;
  1098→}) {
  1099→  const [showIdx, setShowIdx] = useState(-1); // which pad is lit during playback
  1100→  const [watching, setWatching] = useState(true);
  1101→  const [inputStep, setInputStep] = useState(0);
  1102→  const [flash, setFlash] = useState<"ok" | "bad" | null>(null);
  1103→  const done = useRef(false);
  1104→
  1105→  useEffect(() => {
  1106→    if (!watching) return;
  1107→    const timers: number[] = [];
  1108→    pads.forEach((p, i) => {
  1109→      timers.push(window.setTimeout(() => { setShowIdx(p); sfx.pad(p); }, 500 + i * 700));
  1110→      timers.push(window.setTimeout(() => setShowIdx(-1), 500 + i * 700 + 430));
  1111→    });
  1112→    timers.push(window.setTimeout(() => setWatching(false), 500 + pads.length * 700 + 250));
  1113→    return () => timers.forEach((t) => window.clearTimeout(t));
  1114→  }, [pads, watching]);
  1115→
  1116→  const press = (i: number) => {
  1117→    if (watching || disabled || done.current || flash) return;
  1118→    sfx.pad(i);
  1119→    if (pads[inputStep] === i) {
  1120→      if (inputStep + 1 === pads.length) {
  1121→        done.current = true;
  1122→        setFlash("ok");
  1123→        sfx.correct();
  1124→        window.setTimeout(() => onResult(true), 550);
  1125→      } else {
  1126→        setInputStep((s) => s + 1);
  1127→      }
  1128→    } else {
  1129→      done.current = true;
  1130→      setFlash("bad");
  1131→      sfx.wrong();
  1132→      window.setTimeout(() => onResult(false), 650);
  1133→    }
  1134→  };
  1135→
  1136→  return (
  1137→    <div className="flex flex-col items-center gap-3 py-1">
  1138→      <span className={`flex items-center gap-1.5 font-display text-sm font-bold ${watching ? "text-leo-grape" : "text-leo-leaf-deep"}`}>
  1139→        {watching ? (
  1140→          <>
  1141→            <Eye width={15} height={15} /> WATCH…
  1142→          </>
  1143→        ) : flash === "ok" ? (
  1144→          <>
  1145→            <PartyPopper width={15} height={15} /> PERFECT!
  1146→          </>
  1147→        ) : flash === "bad" ? (
  1148→          <>
  1149→            <EyeOff width={15} height={15} /> Oops…
  1150→          </>
  1151→        ) : (
  1152→          <>
  1153→            <HandIcon size={15} /> YOUR TURN — repeat it!
  1154→          </>
  1155→        )}
  1156→      </span>
  1157→      <div className="grid grid-cols-2 gap-3">
  1158→        {PAD_COLORS.map((color, i) => {
  1159→          const lit = showIdx === i;
  1160→          return (
  1161→            <motion.button
  1162→              key={i}
  1163→              type="button"
  1164→              aria-label={`Memory pad ${i + 1}`}
  1165→              onClick={() => press(i)}
  1166→              animate={{
  1167→                scale: lit ? 1.09 : 1,
  1168→                filter: lit ? "brightness(1.35)" : "brightness(1)",
  1169→              }}
  1170→              transition={{ duration: 0.12 }}
  1171→              className="h-24 w-24 cursor-pointer rounded-2xl border-[3.5px] border-white/90 shadow-[0_6px_0_rgba(0,0,0,0.16)]"
  1172→              style={{ background: color, opacity: flash === "bad" && pads[inputStep] !== i ? 0.9 : 1 }}
  1173→            />
  1174→          );
  1175→        })}
  1176→      </div>
  1177→    </div>
  1178→  );
  1179→}
