     1→/**
     2→ * Think Leo — Save State v4 (zustand + localStorage)
     3→ * ─────────────────────────────────────────────────
     4→ * Offline-first: the whole save lives on the kid's device.
     5→ * Schema + hearts-regen, streak, quest and achievement math are
     6→ * documented in docs/ALGORITHM.md §6–7 and §13.
     7→ *
     8→ * ENGINE RULES (v3 — after the great wipe-bug of v2):
     9→ *  · Every helper is PURE: it takes a state snapshot and returns a new one.
    10→ *    No helper ever re-reads `get()` mid-transaction — that is exactly what
    11→ *    used to wipe stars the millisecond they were recorded.
    12→ *  · The quest board is ALWAYS the length of questsForDay(today) (3 rows,
    13→ *    distinct events) — a mismatch only happens on a genuine day rollover.
    14→ *  · Any XP grant or finished run counts as "played today": the streak,
    15→ *    playDays strip and daily ring can never disagree with each other.
    16→ *
    17→ * V4 — "levels must always end":
    18→ *  · `done[levelIdx]` records ANY finished run (0 stars or practice too),
    19→ *    so finishing a level always opens the next one;
    20→ *  · `boostDay` powers Leo's courage boost (one free full refill per day),
    21→ *    so a kid with 0 hearts is never stuck waiting;
    22→ *  · heart regen is 8 min (was 15) — kids wait badly.
    23→ *
    24→ * V5 — Leo Pop (the arcade) joins the ONE economy:
    25→ *  · merge XP flows through addXp like every other grant — leagues,
    26→ *    streaks, the daily ring and the quest board all stay in sync;
    27→ *  · the Zoo Book (mergeDiscoveries) is a collection, not a currency;
    28→ *  · the daily egg (mergeEggDay) is the comeback hook for the arcade.
    29→ *
    30→ * V6 — LEO EGGS (the surprise-egg arcade) joins the same ONE economy:
    31→ *  · every hatch pays rarity XP through the same pipe (daily ring,
    32→ *    league, streak, quests) and duplicate stars into its own purse;
    33→ *  · the egg bag is a collection with memory (count + evolved stage);
    34→ *  · the nest (eggNestAt) is the real-time comeback engine — it keeps
    35→ *    incubating while the app is closed;
    36→ *  · the pity counter (eggPity) guarantees rare-or-better every 8
    37→ *    free Spark eggs — losing streaks always end.
    38→ */

   218→/** Distinct egg friends discovered (bag keys that are real critters). */
   219→function eggDistinct(s: LeoSave): number {
   220→  return Object.keys(s.eggBag).filter((k) => CRITTERS_BY_UID[k]).length;
   221→}
   222→
   223→/** A fresh, empty quest board for today. */
   224→function freshBoard(today: string): Pick<LeoSave, "questDay" | "questProgress" | "questClaimed"> {
   225→  const board = questsForDay(today);
   226→  return {
   227→    questDay: today,
   228→    questProgress: board.map(() => 0),
   229→    questClaimed: board.map(() => false),
   230→  };
   231→}
   232→
   233→/** Rebase the snapshot onto today's board if (and only if) it is stale. */
   234→function ensureQuestDay(s: LeoSave): LeoSave {
   235→  const today = dayKey();
   236→  const boardLen = questsForDay(today).length;
   237→  if (s.questDay === today && s.questProgress.length === boardLen) return s;
   238→  return { ...s, ...freshBoard(today) };
   239→}
   240→
   241→/** Bump every quest listening to `event`. `max` mode keeps the highest value. */
   242→function bumpQuests(
   243→  sBase: LeoSave,
   244→  event: string,
   245→  n: number,
   246→  mode: "add" | "max" = "add"
   247→): LeoSave {
   248→  const s = ensureQuestDay(sBase); // preserves every field of sBase
   249→  const board = questsForDay(s.questDay);
   250→  const progress = [...s.questProgress];
   251→  board.forEach((q, i) => {
   252→    if (q.event !== event) return;
   253→    progress[i] = mode === "max" ? Math.max(progress[i] ?? 0, n) : (progress[i] ?? 0) + n;
   254→  });
   255→  return { ...s, questProgress: progress };
   256→}
   257→
   258→/** The day counter: streak, best streak, freezes and the playDays strip. */
   259→function bumpPlayDay(s0: LeoSave): LeoSave {
   260→  const today = dayKey();
   261→  let streak = s0.streak;
   262→  let freezes = s0.freezes;
   263→  let lastPlayDay = s0.lastPlayDay;
   264→
   265→  if (lastPlayDay !== today) {
   266→    // 999 = first-ever play (no yesterday to chain from)
   267→    const gap = lastPlayDay ? daysBetween(lastPlayDay, today) : 999;
   268→    if (gap === 1) {
   269→      streak += 1; // yesterday → today: the streak lives
   270→    } else if (gap === 2 && freezes > 0 && streak > 0) {
   271→      freezes -= 1; // a streak freeze silently saved the streak
   272→      streak += 1;
   273→    } else {
   274→      streak = 1; // fresh start or first ever — kindly begin at 1
   275→    }
   276→    lastPlayDay = today;
   277→  }
   278→  if (streak === 3) freezes += 1; // milestone: earn a freeze
   279→  if (streak === 7) freezes += 2;
   280→
   281→  const playDays = s0.playDays.includes(today)
   282→    ? s0.playDays
   283→    : [...s0.playDays, today].slice(-30);
   284→
   285→  return {
   286→    ...s0,
   287→    streak,
   288→    bestStreak: Math.max(s0.bestStreak, streak),
   289→    freezes,
   290→    lastPlayDay,
   291→    playDays,
   292→    today,
   293→  };
   294→}

   296→/** XP plumbing without side effects on achievements (they call it too). */
   297→function addXpPure(s0: LeoSave, n: number): { save: LeoSave; leveledTo: number | null } {
   298→  const today = dayKey();
   299→  const wk = weekKey();
   300→  const played = bumpPlayDay(s0);
   301→
   302→  const xp = played.xp + n;
   303→  const todayXp = played.today === today ? played.todayXp + n : n;
   304→  const newLvl = playerLevelFromXp(xp);
   305→  const leveledTo = newLvl > s0.seenPlayerLevel ? newLvl : null;
   306→
   307→  return {
   308→    save: {
   309→      ...played,
   310→      xp,
   311→      todayXp,
   312→      today,
   313→      weekXp: played.leagueWeek === wk ? played.weekXp + n : n,
   314→      leagueWeek: wk,
   315→      seenPlayerLevel: leveledTo ?? played.seenPlayerLevel,
   316→    },
   317→    leveledTo,
   318→  };
   319→}

   321→/** Evaluate one-shot achievements against a snapshot. Pure. */
   322→function checkAchievements(snapshot: LeoSave): { save: LeoSave; unlocked: AchDef[] } {
   323→  const s = snapshot;
   324→  const starLevels = Object.values(s.stars).filter((v) => v > 0).length;
   325→  const chestCount = Object.values(s.chests).filter(Boolean).length;
   326→  const league = leagueOf(s.weekXp);
   327→
   328→  const conditions: Record<string, boolean> = {
   329→    firstStar: starLevels >= 1,
   330→    flawless: Object.values(s.stars).some((v) => v >= 3),
   331→    streak3: s.streak >= 3,
   332→    streak7: s.streak >= 7,
   333→    streak14: s.streak >= 14,
   334→    trail10: starLevels >= 10,
   335→    collector25: starLevels >= 25,
   336→    champion40: starLevels >= 40,
   337→    century: s.totalCorrect >= 100,
   338→    megamind: s.totalCorrect >= 500,
   339→    hunter: chestCount >= 2,
   340→    rising5: playerLevelFromXp(s.xp) >= 5,
   341→    legendLeague: league.index >= 3,
   342→    // Leo Pop
   343→    mergeLeo: (s.mergeDiscoveries[LEO_TIER] ?? 0) >= 1,
   344→    zooAll: Object.keys(s.mergeDiscoveries).filter((k) => Number(k) < ZOO_TARGET).length >= ZOO_TARGET,
   345→    // Leo Eggs
   346→    eggFirst: s.eggHatched >= 1,
   347→    eggExplorer: eggDistinct(s) >= 6,
   348→    eggMaster: eggDistinct(s) >= COLLECTION_TARGET,
   349→    eggLegend: CRITTERS.some((c) => (c.rarity === "legend" || c.rarity === "mythic") && (s.eggBag[c.uid]?.count ?? 0) >= 1),
   350→    eggEvolve: Object.values(s.eggBag).some((c) => c.evolved),
   351→  };
   352→
   353→  const unlocked: AchDef[] = [];
   354→  let bonusXp = 0;
   355→  for (const [id, cond] of Object.entries(conditions)) {
   356→    if (cond && !s.ach.includes(id)) {
   357→      const def = ACH_DEFINITIONS[id];
   358→      if (def) {
   359→        unlocked.push(def);
   360→        bonusXp += def.xp;
   361→      }
   362→    }
   363→  }
   364→  if (unlocked.length === 0) return { save: s, unlocked: [] };
   365→  // ids added first → no recursion; XP flows through the same pipe as every
   366→  // other grant (level math AND the daily "Earn XP" quest stay in sync)
   367→  const withIds = { ...s, ach: [...s.ach, ...unlocked.map((a) => a.id)] };
   368→  const added = addXpPure(withIds, bonusXp);
   369→  const quested = bumpQuests(added.save, "xp", bonusXp);
   370→  return { save: quested, unlocked };
   371→}

   375→export const useLeo = create<LeoStore>()(
   376→  persist(
   377→    (set, get) => ({
   378→      ...DEFAULT_SAVE,
   379→
   380→      tickHearts: () => {
   381→        const s = get();
   382→        if (s.hearts >= MAX_HEARTS) {
   383→          if (s.heartsAt !== 0) set({ heartsAt: Date.now() });
   384→          return;
   385→        }
   386→        const now = Date.now();
   387→        const gained = Math.floor((now - s.heartsAt) / HEART_REGEN_MS);
   388→        if (gained > 0) {
   389→          const hearts = Math.min(MAX_HEARTS, s.hearts + gained);
   390→          set({
   391→            hearts,
   392→            heartsAt: hearts >= MAX_HEARTS ? now : s.heartsAt + gained * HEART_REGEN_MS,
   393→          });
   394→        }
   395→      },
   396→
   397→      loseHeart: () => {
   398→        const s = get();
   399→        if (s.hearts <= 0) return;
   400→        const wasFull = s.hearts >= MAX_HEARTS;
   401→        set({
   402→          hearts: s.hearts - 1,
   403→          heartsAt: wasFull ? Date.now() : s.heartsAt,
   404→        });
   405→      },
   406→
   407→      useRefill: () => {
   408→        const s = get();
   409→        if (s.refills <= 0 || s.hearts >= MAX_HEARTS) return false;
   410→        set({ refills: s.refills - 1, hearts: MAX_HEARTS, heartsAt: Date.now() });
   411→        return true;
   412→      },
   413→
   414→      /**
   415→       * XP always flows through here: it feeds the daily goal ring,
   416→       * the quest board, the weekly league, the streak and the player
   417→       * level — one currency, many meters (the Duolingo trick, §2).
   418→       */
   419→      addXp: (n) => {
   420→        const base = ensureQuestDay(get());
   421→        const quested = bumpQuests(base, "xp", n);
   422→        const { save, leveledTo } = addXpPure(quested, n);
   423→        const { save: finalSave, unlocked } = checkAchievements(save);
   424→        set({
   425→          ...finalSave,
   426→          levelUpPulse: get().levelUpPulse + (leveledTo ? 1 : 0),
   427→          pendingAch: unlocked.length ? [...get().pendingAch, ...unlocked] : get().pendingAch,
   428→        });
   429→      },
   430→
   431→      /** A finished run (any result) counts today as played. */
   432→      noteActivity: () => {
   433→        set(bumpPlayDay(ensureQuestDay(get())));
   434→      },
   435→
   436→      trackAnswer: (correct) => {
   437→        const base = ensureQuestDay(get());
   438→        const quested = correct ? bumpQuests(base, "correct", 1) : base;
   439→        const withCount: LeoSave = correct
   440→          ? { ...quested, totalCorrect: quested.totalCorrect + 1 }
   441→          : quested;
   442→        set(withCount);
   443→        if (correct) {
   444→          const { save, unlocked } = checkAchievements(get());
   445→          if (unlocked.length) {
   446→            set({ ...save, pendingAch: [...get().pendingAch, ...unlocked] });
   447→          }
   448→        }
   449→      },
   450→
   451→      trackCombo: (n) => {
   452→        const base = ensureQuestDay(get());
   453→        set(bumpQuests(base, "maxcombo", n, "max"));
   454→        if (n >= 10 && !get().ach.includes("combo10")) {
   455→          const def = ACH_DEFINITIONS.combo10;
   456→          const withAch = { ...get(), ach: [...get().ach, "combo10"] };
   457→          const { save } = addXpPure(withAch, def.xp);
   458→          set({ ...bumpQuests(save, "xp", def.xp), pendingAch: [...get().pendingAch, def] });
   459→        }
   460→      },
   461→
   462→      trackRush: (hits) => {
   463→        const base = ensureQuestDay(get());
   464→        set(bumpQuests(base, "rushhits", hits, "max"));
   465→        if (hits >= 12 && !get().ach.includes("rushAce")) {
   466→          const def = ACH_DEFINITIONS.rushAce;
   467→          const withAch = { ...get(), ach: [...get().ach, "rushAce"] };
   468→          const { save } = addXpPure(withAch, def.xp);
   469→          set({ ...bumpQuests(save, "xp", def.xp), pendingAch: [...get().pendingAch, def] });
   470→        }
   471→      },
   472→
   473→      /**
   474→       * Records a finished level — ANY finished run (v4): best stars, best
   475→       * score, `done` flag (so the next level always opens), run counter,
   476→       * first-clear quest bumps — then achievement checks. ONE set, built
   477→       * from a single snapshot; nothing can be wiped mid-way any more.
   478→       */
   479→      recordLevel: (levelIdx, stars, score) => {
   480→        const s0 = ensureQuestDay(get());
   481→        const prevStars = s0.stars[levelIdx] ?? 0;
   482→        const prevBest = s0.best[levelIdx] ?? 0;
   483→        const firstClear = prevStars === 0 && stars > 0;
   484→
   485→        let s: LeoSave = {
   486→          ...s0,
   487→          stars: { ...s0.stars, [levelIdx]: Math.max(prevStars, stars) },
   488→          best: { ...s0.best, [levelIdx]: Math.max(prevBest, score) },
   489→          done: { ...s0.done, [levelIdx]: true },
   490→          runs: s0.runs + 1,
   491→        };
   492→        if (firstClear) {
   493→          s = bumpQuests(s, "levels", 1);
   494→          if (stars >= 3) s = bumpQuests(s, "stars3", 1);
   495→        }
   496→        set(s);
   497→
   498→        const { save, unlocked } = checkAchievements(get());
   499→        if (unlocked.length) {
   500→          set({ ...save, pendingAch: [...get().pendingAch, ...unlocked] });
   501→        }
   502→      },
   503→
   504→      claimBoost: () => {
   505→        const s = get();
   506→        const today = dayKey();
   507→        if (s.boostDay === today || s.hearts >= MAX_HEARTS) return false;
   508→        set({
   509→          ...s,
   510→          boostDay: today,
   511→          hearts: MAX_HEARTS,
   512→          heartsAt: Date.now(),
   513→        });
   514→        get().addXp(BOOST_XP);
   515→        return true;
   516→      },
   517→
   518→      boostReady: () => {
   519→        const s = get();
   520→        return s.boostDay !== dayKey() && s.hearts < MAX_HEARTS;
   521→      },
   522→
   523→      grantChest: (worldId) => {
   524→        const s = get();
   525→        if (s.chests[worldId]) return;
   526→        set({

   656→       * (through the same addXp pipe as every other grant) and the
   657→       * egg quests. Achievements are checked once, at the end.

   659→      trackEgg: (plan, opts) => {

   662→        const cur = s0.eggBag[uid];

   664→          ...s0.eggBag,

   675→          // pity guards only the free Spark egg — paid eggs are their own guarantee
   676→          eggPity: plan.kind === "spark" ? (plan.pityReset ? 0 : s0.eggPity + 1) : s0.eggPity,
   677→          // the nest immediately starts incubating the next free egg
   678→          eggNestAt: opts.nest ? Date.now() : s0.eggNestAt,
   679→        };
   680→        s = bumpQuests(s, "eggs", 1);
   681→        if (plan.isNew) s = bumpQuests(s, "eggNew", 1);
   682→        if (isRarePlus(plan.rarity)) s = bumpQuests(s, "eggRare", 1);

   684→        s = bumpQuests(xped, "xp", xp);

   690→        }

   737→    {
   738→      name: "think-leo-save-v1",
   739→      version: 6,
   740→      storage: createJSONStorage(() => localStorage),
   741→      partialize: (s) => {
   742→        const { _hydrated, pendingAch, levelUpPulse, ...rest } = s;
   743→        void _hydrated;
   744→        void pendingAch;
   745→        void levelUpPulse;
   746→        return rest as LeoStore;
   747→      },
   748→      migrate: (persisted, version) => {
   749→        const p = (persisted ?? {}) as Partial<LeoSave>;
   750→        // v2 saves carry XP from the inflated economy (score/12 per run — one
   751→        // run could mint 700+ XP and skip three leagues). Rescale to the v3
   752→        // curve so returning kids land in a sane state without losing progress.
   753→        // (v3+ saves were already rescaled — never touch their XP again.)
   754→        const scale = version < 3 ? 12 : 1;
   755→        const xp = Math.round((p.xp ?? 0) / scale);
   756→        const weekXp = Math.round((p.weekXp ?? 0) / scale);

   758→        const done: Record<number, boolean> = {};
   759→        Object.entries(p.stars ?? {}).forEach(([k, v]) => {
   760→          if (v > 0) done[Number(k)] = true;
   761→        });

   763→          ...DEFAULT_SAVE,
   764→          ...p,
   765→          xp,
   766→          weekXp,
   767→          todayXp: Math.round((p.todayXp ?? 0) / scale),
   768→          hearts: Math.max(p.hearts ?? MAX_HEARTS, 3), // arriving from a fix: be kind
   769→          // v3: boards rebuilt under the exact-3 contract; stars/best survive
   770→          ...freshBoard(dayKey()),
   771→          playDays: p.playDays ?? [],
   772→          ach: p.ach ?? [],
   773→          seenPlayerLevel: playerLevelFromXp(xp),
   774→          comebackDay: "",
   775→          done: { ...done, ...(p.done ?? {}) },
   776→          boostDay: p.boostDay ?? "",
   777→          // v5: the arcade fields (older saves simply start their zoo fresh)
   778→          mergeBest: p.mergeBest ?? 0,
   779→          mergeDiscoveries: p.mergeDiscoveries ?? {},
   780→          mergeEggDay: p.mergeEggDay ?? "",
   781→          mergePlays: p.mergePlays ?? 0,
   782→          totalMerges: p.totalMerges ?? 0,
   783→          // v6: the egg fields (a warm nest + a starter purse for everyone)
   784→          eggStars: p.eggStars ?? 10,
   785→          eggBag: p.eggBag ?? {},
   786→          eggHatched: p.eggHatched ?? 0,
   787→          eggPity: p.eggPity ?? 0,
   788→          eggNestAt: p.eggNestAt ?? 0,
   789→          eggRoyalDay: p.eggRoyalDay ?? "",
   790→        };
   791→      },
   792→      /**
   793→       * Hydration merge — THE hydration flag lives here, not in
   794→       * onRehydrateStorage. Why: that callback runs while `create()` is
   795→       * still executing, so referencing `useLeo` inside it dies in the
   796→       * temporal dead zone (an error persist swallows silently). That
   797→       * bug shipped unnoticed for weeks: `_hydrated` never turned true,
   798→       * so the welcome-back gift and the quick splash never ran. The
   799→       * merge runs for every hydration (stored or fresh) and simply
   800→       * stamps the flag — no store reference needed.
   801→       */
   802→      merge: (persisted, current) =>
   803→        ({
   804→          ...current,
   805→          ...(persisted as Partial<LeoSave>),
   806→          _hydrated: true,
   807→        }) as LeoStore,
   808→      onRehydrateStorage: () => (state) => {
   809→        state?.tickHearts();
   810→      },
   811→    }
   812→  )
   813→);

   821→/** Convenience selector: hearts display never needs manual math. */
   822→export function useHearts(): { hearts: number; max: number; nextInMin: number } {
   823→  const hearts = useLeo((s) => s.hearts);
   824→  const heartsAt = useLeo((s) => s.heartsAt);
   825→  const tick = useLeo((s) => s.tickHearts);
   826→  // light polling keeps regen honest while the tab is open
   827→  useEffect(() => {
   828→    const id = window.setInterval(tick, 5000);
   829→    return () => window.clearInterval(id);
   830→  }, [tick]);
   831→  if (hearts >= MAX_HEARTS) return { hearts, max: MAX_HEARTS, nextInMin: 0 };
   832→  const nextIn = Math.max(
   833→    0,
   834→    Math.ceil((heartsAt + HEART_REGEN_MS - Date.now()) / 60000)
   835→  );
   836→  return { hearts, max: MAX_HEARTS, nextInMin: nextIn };
   837→}

   839→/** Today's quest board + progress (safe during hydration). */
   840→export function useQuests() {
   841→  const questDay = useLeo((s) => s.questDay);
   842→  const questProgress = useLeo((s) => s.questProgress);
   843→  const questClaimed = useLeo((s) => s.questClaimed);
   844→  const questAllBonusDay = useLeo((s) => s.questAllBonusDay);
   845→  const today = dayKey();
   846→  const board = questsForDay(today);
   847→  const stale = questDay !== today || questProgress.length !== board.length;
   848→  const progress = stale ? board.map(() => 0) : questProgress;
   849→  const claimed = stale ? board.map(() => false) : questClaimed;
   850→  const bonusReady =
   851→    !stale &&
   852→    questAllBonusDay !== today &&
   853→    board.every((q, i) => claimed[i] && (progress[i] ?? 0) >= q.target);
   854→  const allDone = board.every((q, i) => claimed[i] && (progress[i] ?? 0) >= q.target);
   855→  return { board, progress, claimed, bonusReady, allDone, stale, today };
   856→}

   874→ * The nest, live: a 1s ticker that only runs while the app is open.
   875→ * `ready` means a free egg is waiting in the nest right now — the Home
   876→ * card and the egg game both read this one truth.

   878→export function useEggNest(): NestState & { hydrated: boolean } {

   880→  const hydrated = useLeo((s) => s._hydrated);

   888→  return { ...nestState(eggNestAt, now), hydrated: true };
