     1→"use client";
     2→
     3→/**
     4→ * Think Leo — Play screen: the adrenaline engine
     5→ * ──────────────────────────────────────────────
     6→ * Phase machine:
     7→ *   intro → play ⇄ reveal → (last round) rushIntro → rush → results
     8→ *                          └─ hearts hit 0 → out
     9→ * Juice: combo meter with escalating sounds, countdown ring, heartbeat
    10→ * vignette, screen shake, Leo reactions, and the ×2 LEO RUSH finale
    11→ * (Candy-Crush "Sugar Crush" moment). Math in docs/ALGORITHM.md §4–6.
    12→ */
    13→
    14→import { Leo } from "@/components/leo/Leo";
    15→import { ChunkyButton, ConfettiBurst, HeartPill, PawProgress, StarRow } from "@/components/game/bits";
    16→import {
    17→  BASE_POINTS,
    18→  RUSH_SECONDS,
    19→  ROUND_COUNT,
    20→  chestAfterLevel,
    21→  clamp,
    22→  questionPoints,
    23→  questionTime,
    24→  starsFor,
    25→  worldOf,
    26→  xpForRun,
    27→  type Mood,
    28→} from "@/lib/game";
    29→import { buildPlan, makeQuestion, makeRushQuestion, type Question } from "@/lib/questions";
    30→import { sfx } from "@/lib/sound";
    31→import { useLeo } from "@/lib/store";
    32→import { AnimatePresence, motion } from "framer-motion";
    33→import { useCallback, useEffect, useMemo, useRef, useState } from "react";
    34→
    35→type Phase = "intro" | "play" | "reveal" | "rushIntro" | "rush" | "out" | "results";
    36→
    37→interface Results {
    38→  stars: number;
    39→  xp: number;
    40→  accuracy: number;
    41→  total: number;
    42→  isBest: boolean;
    43→  chest: boolean;
    44→}
    45→
    46→const PRAISE = ["Roarrsome!", "Big brain move!", "Yes! Keep going!", "Pawsome!"];
    47→const UNSTOPPABLE = ["UNSTOPPABLE!", "MEGA COMBO!", "ON FIRE! 🔥", "LEO-LEVEL THINKING!"];
    48→const OOPS = ["Almost!", "So close!", "No worries —"];
    49→const NO_HEARTS = [
    50→  "Hearts refill over time — brains need naps too!",
    51→  "Out of hearts? Stretch, sip water, come back roaring!",
    52→];
    53→
    54→const PAD_COLORS = ["#58cc02", "#ff5a8c", "#9b5de5", "#ffb020"];
    55→
    56→export function Play({
    57→  levelIdx,
    58→  onExit,
    59→  onNext,
    60→  hasNext,
    61→}: {
    62→  levelIdx: number;
    63→  onExit: () => void;
    64→  onNext: () => void;
    65→  hasNext: boolean;
    66→}) {
    67→  const world = worldOf(levelIdx);
    68→  const plan = useMemo(() => buildPlan(levelIdx), [levelIdx]);
    69→  const attemptRef = useRef(1);
    70→
    71→  /* ── run state ── */
    72→  const [phase, setPhase] = useState<Phase>("intro");
    73→  const [qi, setQi] = useState(0);
    74→  const [boost, setBoost] = useState(0); // adaptive difficulty offset (-1..+2)
    75→  const [combo, setCombo] = useState(0);
    76→  const [maxCombo, setMaxCombo] = useState(0);
    77→  const [score, setScore] = useState(0);
    78→  const [correctCount, setCorrectCount] = useState(0);
    79→  const [answeredCount, setAnsweredCount] = useState(0);
    80→  const [rightStreak, setRightStreak] = useState(0);
    81→  const [wrongStreak, setWrongStreak] = useState(0);
    82→  const [picked, setPicked] = useState<number | null>(null);
    83→  const [revealOk, setRevealOk] = useState(false);
    84→  const [shakeKey, setShakeKey] = useState(0);
    85→  const [leoMood, setLeoMood] = useState<Mood>("idle");
    86→  const [bubble, setBubble] = useState<string | null>(null);
    87→  const [confettiKey, setConfettiKey] = useState(0);
    88→
    89→  /* rush state */
    90→  const [rushQs, setRushQs] = useState<Question[]>([]);
    91→  const [rushIdx, setRushIdx] = useState(0);
    92→  const [rushCombo, setRushCombo] = useState(0);
    93→  const [rushScore, setRushScore] = useState(0);
    94→  const [rushLeft, setRushLeft] = useState(RUSH_SECONDS);
    95→
    96→  const [results, setResults] = useState<Results | null>(null);
    97→  const recordedRef = useRef(false);
    98→
    99→  const diff = clamp(
   100→    Math.max(1, Math.min(6, 1 + Math.floor((levelIdx - 1) / 8) + boost)),
   101→    1,
   102→    6
   103→  );
   104→  const q = useMemo(
   105→    () =>
   106→      phase === "play" || phase === "reveal"
   107→        ? makeQuestion(plan[qi], diff, levelIdx, qi, attemptRef.current)
   108→        : null,
   109→    [phase, qi, diff, levelIdx, plan]
   110→  );
   111→
   112→  const resetRun = useCallback(
   113→    (fresh: boolean) => {
   114→      if (fresh) attemptRef.current += 1;
   115→      recordedRef.current = false;
   116→      setPhase("intro");
   117→      setQi(0);
   118→      setBoost(0);
   119→      setCombo(0);
   120→      setMaxCombo(0);
   121→      setScore(0);
   122→      setCorrectCount(0);
   123→      setAnsweredCount(0);
   124→      setRightStreak(0);
   125→      setWrongStreak(0);
   126→      setPicked(null);
   127→      setRushQs([]);
   128→      setRushIdx(0);
   129→      setRushCombo(0);
   130→      setRushScore(0);
   131→      setResults(null);
   132→      setLeoMood("idle");
   133→      setBubble(null);
   134→    },
   135→    []
   136→  );
   137→
   138→  /* keyboard answering (laptop love) */
   139→  const answerRef = useRef<(i: number) => void>(() => {});
   140→  useEffect(() => {
   141→    const h = (e: KeyboardEvent) => {
   142→      const n = Number(e.key);
   143→      if (phase === "play" && n >= 1 && n <= 4) answerRef.current(n - 1);
   144→    };
   145→    window.addEventListener("keydown", h);
   146→    return () => window.removeEventListener("keydown", h);
   147→  }, [phase]);
   148→
   149→  /* music per screen */
   150→  useEffect(() => {
   151→    sfx.startMusic("play");
   152→    return () => {
   153→      sfx.startMusic("home");
   154→    };
   155→  }, []);
   156→
   157→  /* ── answer handling ─────────────────────────────────────────── */
   158→
   159→  const advance = useCallback(
   160→    (ok: boolean) => {
   161→      const heartsNow = useLeo.getState().hearts;
   162→      if (!ok && heartsNow <= 0) {
   163→        setLeoMood("sad");
   164→        setBubble(NO_HEARTS[Math.floor(Math.random() * NO_HEARTS.length)]);
   165→        setPhase("out");
   166→        sfx.sadTrombone();
   167→        return;
   168→      }
   169→      if (qi >= ROUND_COUNT - 1) {
   170→        setLeoMood("excited");
   171→        setBubble("LEO RUSH! Everything ×2 — GO GO GO!");
   172→        setPhase("rushIntro");
   173→        sfx.whoosh();
   174→        window.setTimeout(() => {
   175→          setRushQs(Array.from({ length: 30 }, (_, i) => makeRushQuestion(diff, i, attemptRef.current)));
   176→          setRushIdx(0);
   177→          setRushLeft(RUSH_SECONDS);
   178→          setPhase("rush");
   179→        }, 1700);
   180→      } else {
   181→        setQi((n) => n + 1);
   182→        setPicked(null);
   183→        setLeoMood("idle");
   184→        setBubble(null);
   185→        setPhase("play");
   186→      }
   187→    },
   188→    [qi, diff]
   189→  );
   190→
   191→  const handleAnswer = useCallback(
   192→    (idx: number | null) => {
   193→      if (!q || phase !== "play") return;
   194→      const fracLeft = clamp(timeLeftRef.current / q.time, 0, 1);
   195→      const ok = idx !== null && idx === q.answer;
   196→      setPicked(idx);
   197→      setRevealOk(ok);
   198→      setAnsweredCount((a) => a + 1);
   199→      setPhase("reveal");
   200→
   201→      if (ok) {
   202→        const pts = questionPoints(combo + 1, fracLeft);
   203→        setScore((s) => s + pts);
   204→        setCorrectCount((c) => c + 1);
   205→        const nc = combo + 1;
   206→        setCombo(nc);
   207→        setMaxCombo((m) => Math.max(m, nc));
   208→        setRightStreak((r) => {
   209→          const nr = r + 1;
   210→          if (nr % 3 === 0) setBoost((b) => Math.min(2, b + 1)); // Flow Ranger: heat up
   211→          return nr;
   212→        });
   213→        setWrongStreak(0);
   214→        sfx.correct();
   215→        if (nc >= 3) window.setTimeout(() => sfx.combo(nc), 140);
   216→        setLeoMood(nc >= 5 ? "excited" : "happy");
   217→        setBubble(nc >= 5 ? UNSTOPPABLE[nc % UNSTOPPABLE.length] : PRAISE[nc % PRAISE.length]);
   218→      } else {
   219→        const s = useLeo.getState();
   220→        s.loseHeart();
   221→        setCombo(0);
   222→        setWrongStreak((w) => {
   223→          const nw = w + 1;
   224→          if (nw % 2 === 0) setBoost((b) => Math.max(-1, b - 1)); // Flow Ranger: cool down
   225→          return nw;
   226→        });
   227→        setRightStreak(0);
   228→        sfx.wrong();
   229→        window.setTimeout(() => sfx.heartLost(), 240);
   230→        setShakeKey((k) => k + 1);
   231→        setLeoMood("sad");
   232→        setBubble(idx === null ? "Time's up! Here's the trick:" : `${OOPS[Math.floor(Math.random() * OOPS.length)]} ${q.tip}`);
   233→      }
   234→
   235→      window.setTimeout(() => advance(ok), ok ? 900 : 1600);
   236→    },
   237→    [q, phase, combo, advance]
   238→  );
   239→
   240→  answerRef.current = (i: number) => {
   241→    if (!q || phase !== "play") return;
   242→    if (q.kind === "memory") return;
   243→    if (q.options.length && i < q.options.length) handleAnswer(i);
   244→  };
   245→
   246→  /* countdown ring */
   247→  const timeLeftRef = useRef(0);
   248→  const [timeLeft, setTimeLeft] = useState(10);
   249→  useEffect(() => {
   250→    if (phase !== "play" || !q) return;
   251→    const total = q.time;
   252→    setTimeLeft(total);
   253→    timeLeftRef.current = total;
   254→    const start = performance.now();
   255→    let lastTick = Math.ceil(total);
   256→    const id = window.setInterval(() => {
   257→      const left = total - (performance.now() - start) / 1000;
   258→      timeLeftRef.current = Math.max(0, left);
   259→      setTimeLeft(Math.max(0, left));
   260→      const whole = Math.ceil(left);
   261→      if (whole !== lastTick && whole <= 5 && whole > 0) sfx.tick();
   262→      lastTick = whole;
   263→      if (left <= 0) {
   264→        window.clearInterval(id);
   265→        handleAnswer(null);
   266→      }
   267→    }, 100);
   268→    return () => window.clearInterval(id);
   269→  }, [phase, qi]);
   270→
   271→  /* ── LEO RUSH ────────────────────────────────────────────────── */
   272→
   273→  const endRush = useCallback(() => {
   274→    const s = useLeo.getState();
   275→    const total = score + rushScore;
   276→    const accuracy = answeredCount > 0 ? correctCount / answeredCount : 0;
   277→    const stars = starsFor(accuracy);
   278→    const xp = stars > 0 ? xpForRun(total, stars) : 0;
   279→    const isBest = stars > 0 && total > (s.best[levelIdx] ?? 0);
   280→    if (!recordedRef.current) {
   281→      recordedRef.current = true;
   282→      if (stars > 0) {
   283→        s.recordLevel(levelIdx, stars, total);
   284→        s.addXp(xp);
   285→      }
   286→    }
   287→    const chest = stars > 0 && chestAfterLevel(levelIdx);
   288→    if (chest) s.grantChest(world.id);
   289→    setResults({ stars, xp, accuracy, total, isBest, chest });
   290→    setPhase("results");
   291→    if (stars > 0) {
   292→      sfx.fanfare();
   293→      setConfettiKey((k) => k + 1);
   294→      setLeoMood("celebrate");
   295→      setBubble(stars === 3 ? "PERFECT RUN! I'm roaring the roof off!" : "Level crushed! More stars next time?");
   296→      for (let i = 0; i < stars; i++) window.setTimeout(() => sfx.star(i), 600 + i * 380);
   297→    } else {
   298→      sfx.sadTrombone();
   299→      setLeoMood("sad");
   300→      setBubble("So close! One more try — I'll cheer for you!");
   301→    }
   302→  }, [score, rushScore, answeredCount, correctCount, levelIdx, world.id]);
   303→
   304→  useEffect(() => {
   305→    if (phase !== "rush") return;
   306→    const start = performance.now();
   307→    const id = window.setInterval(() => {
   308→      const left = RUSH_SECONDS - (performance.now() - start) / 1000;
   309→      setRushLeft(Math.max(0, left));
   310→      if (left <= 0) {
   311→        window.clearInterval(id);
   312→        endRush();
   313→      }
   314→    }, 100);
   315→    return () => window.clearInterval(id);
   316→  }, [phase, endRush]);
   317→
   318→  const rushAnswer = useCallback(
   319→    (idx: number) => {
   320→      const rq = rushQs[rushIdx];
   321→      if (!rq) return;
   322→      const ok = idx === rq.answer;
   323→      if (ok) {
   324→        setRushScore((s) => s + 120);
   325→        setRushCombo((c) => {
   326→          const nc = c + 1;
   327→          sfx.combo(nc + 2);
   328→          return nc;
   329→        });
   330→      } else {
   331→        setRushCombo(0);
   332→        sfx.tap();
   333→      }
   334→      if (rushIdx < rushQs.length - 1) setRushIdx((i) => i + 1);
   335→    },
   336→    [rushQs, rushIdx]
   337→  );
   338→
   339→  const hearts = useLeo((s) => s.hearts);
   340→  const refills = useLeo((s) => s.refills);
   341→  const doRefill = useLeo((s) => s.useRefill);
   342→
   343→  /* ── render ──────────────────────────────────────────────────── */
   344→
   345→  const qTimeTotal = q?.time ?? 10;
   346→  const danger = phase === "play" && timeLeft <= 3.5;
   347→  const hotCombo = combo >= 5 || (phase === "rush" && rushCombo >= 4);
   348→  const rushQ = phase === "rush" ? rushQs[rushIdx] : null;
   349→
   350→  return (
   351→    <div
   352→      className="relative h-screen w-full overflow-hidden"
   353→      style={{ background: `linear-gradient(180deg, ${world.wash[0]}, ${world.wash[1]})` }}
   354→    >
   355→      {/* heartbeat vignette */}
   356→      {danger && (
   357→        <div
   358→          className="pointer-events-none absolute inset-0 z-10"
   359→          style={{ boxShadow: "inset 0 0 90px rgba(255,60,90,0.45)" }}
   360→        />
   361→      )}
   362→      {hotCombo && <div className="combo-hot pointer-events-none absolute inset-0 z-10" />}
   363→
   364→      <ConfettiBurst fireKey={confettiKey} />
   365→
   366→      {/* HUD */}
   367→      <div className="absolute inset-x-0 top-0 z-20 flex items-center gap-2 px-4 py-3">
   368→        <button
   369→          type="button"
   370→          onClick={onExit}
   371→          className="pill cursor-pointer px-3 py-1.5 font-display text-sm text-leo-ink hover:bg-leo-cream"
   372→        >
   373→          ✕ Map
   374→        </button>
   375→        <div className="pill px-3 py-1 font-display text-sm text-leo-ink">
   376→          {world.emoji} Lv {levelIdx}
   377→        </div>
   378→        <div className="ml-auto flex items-center gap-2">
   379→          <AnimatePresence mode="popLayout">
   380→            {combo >= 2 && (phase === "play" || phase === "reveal") && (
   381→              <motion.div
   382→                key={combo}
   383→                initial={{ scale: 0.4, rotate: -8, opacity: 0 }}
   384→                animate={{ scale: 1, rotate: 0, opacity: 1 }}
   385→                exit={{ scale: 0.6, opacity: 0 }}
   386→                className="pill px-3 py-1"
   387→                style={{ background: "#fff3c4", borderColor: "#ffb020" }}
   388→              >
   389→                <span className="font-display text-sm font-bold text-leo-sun-deep">×{combo} COMBO</span>
   390→              </motion.div>
   391→            )}
   392→          </AnimatePresence>
   393→          <div className="pill px-3 py-1 font-display text-sm font-bold text-leo-ink">{score + rushScore} pts</div>
   394→          <HeartPill compact />
   395→        </div>
   396→      </div>
   397→
   398→      {/* phases */}
   399→      <AnimatePresence mode="wait">
   400→        {phase === "intro" && (
   401→          <IntroCard
   402→            key="intro"
   403→            levelIdx={levelIdx}
   404→            plan={plan}
   405→            hearts={hearts}
   406→            refills={refills}
   407→            onStart={() => {
   408→              sfx.click();
   409→              setLeoMood("excited");
   410→              setBubble("8 puzzles. Beat the clock. ROAR!");
   411→              window.setTimeout(() => setBubble(null), 1600);
   412→              setPhase("play");
   413→            }}
   414→            onRefill={() => {
   415→              if (doRefill()) {
   416→                sfx.flame();
   417→                setBubble("Hearts full — LET'S GO!");
   418→              }
   419→            }}
   420→            onExit={onExit}
   421→          />
   422→        )}
   423→
   424→        {(phase === "play" || phase === "reveal") && q && (
   425→          <motion.section
   426→            key={`q-${qi}-${shakeKey}`}
   427→            initial={{ opacity: 0, y: 24 }}
   428→            animate={{ opacity: 1, y: 0 }}
   429→            exit={{ opacity: 0, y: -16 }}
   430→            className="absolute inset-0 z-[15] flex flex-col items-center justify-center px-6 pt-14"
   431→          >
   432→            {/* progress + timer */}
   433→            <div className="mb-3 flex w-full max-w-2xl items-center justify-between">
   434→              <PawProgress current={qi} total={ROUND_COUNT} />
   435→              <TimerRing left={timeLeft} total={qTimeTotal} />
   436→            </div>
   437→
   438→            {/* Leo + question card */}
   439→            <div className="flex w-full max-w-2xl items-end gap-3">
   440→              <div className="hidden shrink-0 md:block">
   441→                <Leo mood={leoMood} size={130} bubble={bubble} />
   442→              </div>
   443→              <div className={`leo-card w-full p-5 ${shakeKey % 2 === 1 && phase === "reveal" && !revealOk ? "anim-shake" : ""}`}>
   444→                <h2 className="text-center font-display text-xl font-bold text-leo-ink">{q.prompt}</h2>
   445→                <div className="mt-4">
   446→                  <QuestionStage q={q} reveal={phase === "reveal"} picked={picked} onAnswer={handleAnswer} />
   447→                </div>
   448→              </div>
   449→            </div>
   450→
   451→            {/* mobile Leo line */}
   452→            <p className="mt-3 font-display text-sm font-semibold text-leo-ink md:hidden">
   453→              {bubble ?? ""}
   454→            </p>
   455→          </motion.section>
   456→        )}
   457→
   458→        {phase === "rushIntro" && (
   459→          <motion.div
   460→            key="rushintro"
   461→            className="absolute inset-0 z-[25] flex flex-col items-center justify-center"
   462→            initial={{ opacity: 0 }}
   463→            animate={{ opacity: 1 }}
   464→            exit={{ opacity: 0 }}
   465→          >
   466→            <motion.div
   467→              initial={{ scale: 0.2, rotate: -14 }}
   468→              animate={{ scale: [0.2, 1.25, 1], rotate: [-14, 4, 0] }}
   469→              transition={{ duration: 0.7 }}
   470→              className="font-display text-7xl font-bold"
   471→            >
   472→              <span className="bg-gradient-to-r from-leo-sun via-leo-berry to-leo-grape bg-clip-text text-transparent">
   473→                LEO RUSH!
   474→              </span>
   475→            </motion.div>
   476→            <motion.div
   477→              initial={{ y: 20, opacity: 0 }}
   478→              animate={{ y: 0, opacity: 1 }}
   479→              transition={{ delay: 0.3 }}
   480→              className="mt-2 font-display text-xl text-leo-ink"
   481→            >
   482→              Everything ×2 · no hearts lost · GO!
   483→            </motion.div>
   484→            <div className="mt-4">
   485→              <Leo mood="excited" size={150} />
   486→            </div>
   487→          </motion.div>
   488→        )}
   489→
   490→        {phase === "rush" && rushQ && (
   491→          <motion.section
   492→            key={`rush-${rushIdx}`}
   493→            initial={{ opacity: 0, scale: 0.92 }}
   494→            animate={{ opacity: 1, scale: 1 }}
   495→            exit={{ opacity: 0, scale: 1.06 }}
   496→            transition={{ duration: 0.12 }}
   497→            className="absolute inset-0 z-[15] flex flex-col items-center justify-center px-6 pt-10"
   498→          >
   499→            <div className="mb-2 flex w-full max-w-xl items-center justify-between">
   500→              <span className="pill px-3 py-1 font-display text-sm font-bold text-leo-berry-deep">
   501→                ×2 RUSH
   502→              </span>
   503→              <span
   504→                className={`pill px-3 py-1 font-display text-lg font-bold ${
   505→                  rushLeft <= 4 ? "text-leo-berry-deep anim-heartbeat" : "text-leo-ink"
   506→                }`}
   507→              >
   508→                ⏱ {rushLeft.toFixed(1)}s
   509→              </span>
   510→              {rushCombo >= 2 && (
   511→                <span className="pill px-3 py-1 font-display text-sm font-bold text-leo-sun-deep">
   512→                  streak {rushCombo}
   513→                </span>
   514→              )}
   515→            </div>
   516→            <div className="leo-card w-full max-w-xl p-5">
   517→              <h2 className="text-center font-display text-xl font-bold text-leo-ink">{rushQ.prompt}</h2>
   518→              <div className="mt-4">
   519→                <QuestionStage q={rushQ} reveal={false} picked={null} onAnswer={rushAnswer} rushMode />
   520→              </div>
   521→            </div>
   522→            <div className="mt-3 flex items-center gap-3">
   523→              <Leo mood="excited" size={92} />
   524→              <div className="font-display text-3xl font-bold text-leo-ink">{rushScore}</div>
   525→            </div>
   526→          </motion.section>
   527→        )}
   528→
   529→        {phase === "out" && (
   530→          <motion.div
   531→            key="out"
   532→            initial={{ opacity: 0, y: 20 }}

   624→function TimerRing({ left, total }: { left: number; total: number }) {
   625→  const r = 16;
   626→  const c = 2 * Math.PI * r;
   627→  const frac = clamp(left / total, 0, 1);
   628→  const low = left <= 5;
   629→  return (
   630→    <div className={`relative h-11 w-11 ${low ? "anim-heartbeat" : ""}`}>
   631→      <svg width="44" height="44" viewBox="0 0 44 44" className="-rotate-90">
   632→        <circle cx="22" cy="22" r={r} fill="#fff" stroke="#F0E0BD" strokeWidth="5" />
   633→        <circle
   634→          cx="22"
   635→          cy="22"
   636→          r={r}
   637→          fill="none"
   638→          stroke={low ? "#ff5a8c" : "#58cc02"}
   639→          strokeWidth="5"
   640→          strokeLinecap="round"
   641→          strokeDasharray={c}
   642→          strokeDashoffset={c * (1 - frac)}
   643→        />
   644→      </svg>
   645→      <span className={`absolute inset-0 flex items-center justify-center font-display text-sm font-bold ${low ? "text-leo-berry-deep" : "text-leo-ink"}`}>
   646→        {Math.ceil(left)}
   647→      </span>
   648→    </div>
   649→  );
   650→}

   731→function QuestionStage({
   732→  q,
   733→  reveal,
   734→  picked,
   735→  onAnswer,
   736→  rushMode = false,
   737→}: {
   738→  q: Question;
   739→  reveal: boolean;
   740→  picked: number | null;
   741→  onAnswer: (i: number | null) => void;
   742→  rushMode?: boolean;
   743→}) {
   744→  const bigDisplay = q.display ?? null;
   745→
   746→  return (
   747→    <div>
   748→      {bigDisplay && (
   749→        <div className="py-2 text-center font-display text-5xl font-bold tracking-wide text-leo-ink">
   750→          {bigDisplay}
   751→        </div>
   752→      )}
   753→
   754→      {q.kind === "count" && q.emojis && (
   755→        <div className="mx-auto flex max-w-md flex-wrap justify-center gap-1.5 py-2 text-4xl">
   756→          {q.emojis.map((e, i) => (
   757→            <motion.span key={i} initial={{ scale: 0 }} animate={{ scale: 1 }} transition={{ delay: i * 0.03 }}>
   758→              {e}
   759→            </motion.span>
   760→          ))}
   761→        </div>
   762→      )}
   763→
   764→      {q.kind === "sequence" && q.seq && (
   765→        <div className="flex flex-wrap justify-center gap-2 py-2">
   766→          {q.seq.map((s, i) => (
   767→            <span
   768→              key={i}
   769→              className={`flex h-14 min-w-14 items-center justify-center rounded-xl border-[3px] px-2 font-display text-2xl font-bold ${
   770→                s === "?" ? "border-leo-sun bg-leo-cream text-leo-sun-deep" : "border-leo-line bg-white text-leo-ink"
   771→              }`}
   772→            >
   773→              {s}
   774→            </span>
   775→          ))}
   776→        </div>
   777→      )}
   778→
   779→      {q.kind === "memory" && q.pads && (
   780→        <MemoryPads pads={q.pads} onResult={(ok) => onAnswer(ok ? q.answer : (q.answer + 1) % 4)} disabled={reveal} />
   781→      )}
   782→
   783→      {/* options */}
   784→      {q.kind !== "memory" && q.options.length > 0 && (
   785→        <div
   786→          className={`mt-3 grid gap-2.5 ${
   787→            q.options.length <= 2
   788→              ? "grid-cols-2"
   789→              : q.options.every((o) => o.length <= 3)
   790→                ? "grid-cols-4"
   791→                : "grid-cols-2"
   792→          }`}
   793→        >
   794→          {q.options.map((opt, i) => {
   795→            const isCorrect = reveal && i === q.answer;
   796→            const isWrongPick = reveal && picked === i && i !== q.answer;
   797→            const plate = q.groupOptions?.[i];
   798→            return (
   799→              <motion.button
   800→                key={i}
   801→                type="button"
   802→                disabled={reveal}
   803→                onClick={() => onAnswer(i)}
   804→                data-state={isCorrect ? "correct" : isWrongPick ? "wrong" : undefined}
   805→                whileHover={!reveal ? { y: -3 } : undefined}
   806→                whileTap={!reveal ? { scale: 0.96 } : undefined}
   807→                className={`answer-bubble flex min-h-[64px] cursor-pointer items-center justify-center px-3 py-3 ${
   808→                  q.kind === "oddOne" ? "text-4xl" : q.kind === "truefalse" ? "font-display text-2xl font-bold" : "font-display text-2xl font-bold text-leo-ink"
   809→                } ${q.kind === "truefalse" ? (i === 0 ? "bg-[#e9ffcf]" : "bg-[#ffe4ec]") : ""}`}
   810→              >
   811→                {plate ? (
   812→                  <span className="flex flex-col items-center gap-1">
   813→                    <span className="flex max-w-[180px] flex-wrap justify-center gap-0.5 text-xl">
   814→                      {plate.map((e, j) => (
   815→                        <span key={j}>{e}</span>
   816→                      ))}
   817→                    </span>
   818→                    <span className="font-display text-lg font-bold text-leo-ink">{opt}</span>
   819→                  </span>
   820→                ) : (
   821→                  opt
   822→                )}
   823→              </motion.button>
   824→            );
   825→          })}
   826→        </div>
   827→      )}
   828→
   829→      {rushMode && null}
   830→    </div>
   831→  );
   832→}

   834→/* ── memory mini-game (Simon-style) ────────────────────────────── */
   835→
   836→function MemoryPads({
   837→  pads,
   838→  onResult,
   839→  disabled,
   840→}: {
   841→  pads: number[];
   842→  onResult: (ok: boolean) => void;
   843→  disabled: boolean;
   844→}) {
   845→  const [showIdx, setShowIdx] = useState(-1); // which pad is lit during playback
   846→  const [watching, setWatching] = useState(true);
   847→  const [inputStep, setInputStep] = useState(0);
   848→  const [flash, setFlash] = useState<"ok" | "bad" | null>(null);
   849→  const done = useRef(false);
   850→
   851→  useEffect(() => {
   852→    if (!watching) return;
   853→    const timers: number[] = [];
   854→    pads.forEach((p, i) => {
   855→      timers.push(window.setTimeout(() => { setShowIdx(p); sfx.pad(p); }, 500 + i * 700));
   856→      timers.push(window.setTimeout(() => setShowIdx(-1), 500 + i * 700 + 430));
   857→    });
   858→    timers.push(window.setTimeout(() => setWatching(false), 500 + pads.length * 700 + 250));
   859→    return () => timers.forEach((t) => window.clearTimeout(t));
   860→  }, [pads, watching]);
   861→
   862→  const press = (i: number) => {
   863→    if (watching || disabled || done.current || flash) return;
   864→    sfx.pad(i);
   865→    if (pads[inputStep] === i) {
   866→      if (inputStep + 1 === pads.length) {
   867→        done.current = true;
   868→        setFlash("ok");
   869→        sfx.correct();
   870→        window.setTimeout(() => onResult(true), 550);
   871→      } else {
   872→        setInputStep((s) => s + 1);
   873→      }
   874→    } else {
   875→      done.current = true;
   876→      setFlash("bad");
   877→      sfx.wrong();
   878→      window.setTimeout(() => onResult(false), 650);
   879→    }
   880→  };
   881→
   882→  return (
   883→    <div className="flex flex-col items-center gap-3 py-1">
   884→      <span className={`font-display text-sm font-bold ${watching ? "text-leo-grape" : "text-leo-leaf-deep"}`}>
   885→        {watching ? "👀 WATCH…" : flash === "ok" ? "🎉 PERFECT!" : flash === "bad" ? "Oops…" : "🖐 YOUR TURN — repeat it!"}
   886→      </span>
   887→      <div className="grid grid-cols-2 gap-3">
   888→        {PAD_COLORS.map((color, i) => {
   889→          const lit = showIdx === i;
   890→          return (
   891→            <motion.button
   892→              key={i}
   893→              type="button"
   894→              aria-label={`Memory pad ${i + 1}`}
   895→              onClick={() => press(i)}
   896→              animate={{
   897→                scale: lit ? 1.09 : 1,
   898→                filter: lit ? "brightness(1.35)" : "brightness(1)",
   899→              }}
   900→              transition={{ duration: 0.12 }}
   901→              className="h-24 w-24 cursor-pointer rounded-2xl border-[3.5px] border-white/90 shadow-[0_6px_0_rgba(0,0,0,0.16)]"
   902→              style={{ background: color, opacity: flash === "bad" && pads[inputStep] !== i ? 0.9 : 1 }}
   903→            />
   904→          );
   905→        })}
   906→      </div>
   907→    </div>
   908→  );
   909→}
